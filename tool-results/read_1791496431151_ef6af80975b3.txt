    67→function StatCard({
    68→  icon,
    69→  label,
    70→  value,
    71→  subtitle,
    72→  tint = 'emerald',
    73→}: {
    74→  icon: ReactNode
    75→  label: string
    76→  value: ReactNode
    77→  subtitle?: ReactNode
    78→  tint?: 'emerald' | 'teal' | 'amber' | 'rose' | 'zinc'
    79→}) {
    80→  const tints = {
    81→    emerald: 'text-emerald-600 dark:text-emerald-400 shadow-emerald-500/20',
    82→    teal: 'text-teal-600 dark:text-teal-400 shadow-teal-500/20',
    83→    amber: 'text-amber-600 dark:text-amber-400 shadow-amber-500/20',
    84→    rose: 'text-rose-600 dark:text-rose-400 shadow-rose-500/20',
    85→    zinc: 'text-zinc-500 dark:text-zinc-400',
    86→  } as const
    87→  return (
    88→    <Panel className="card-sheen glass-hover group relative overflow-hidden p-4">
    89→      {/* corner sheen for the liquid-glass look */}
    90→      <div
    91→        aria-hidden
    92→        className="pointer-events-none absolute -right-8 -top-8 h-24 w-24 rounded-full bg-primary/5 blur-2xl transition-opacity duration-300 group-hover:bg-primary/10"
    93→      />
    94→      <div className="flex items-center justify-between gap-2">
    95→        <span className="text-xs font-medium text-muted-foreground">{label}</span>
    96→        <span
    97→          className={cn(
    98→            'glass-inset flex h-7 w-7 shrink-0 items-center justify-center rounded-lg shadow-sm [&_svg]:size-3.5',
    99→            tints[tint]
   100→          )}
   101→        >
   102→          {icon}
   103→        </span>
   104→      </div>
   105→      <p className="mt-2 text-2xl font-semibold tabular-nums tracking-tight">{value}</p>
   106→      {subtitle ? (
   107→        <p className="mt-0.5 text-xs text-muted-foreground">{subtitle}</p>
   108→      ) : null}
   109→    </Panel>
   110→  )
   111→}
