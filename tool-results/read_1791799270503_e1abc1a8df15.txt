     1→// Simulated Windows agent. Connects to the hub over socket.io exactly like a
     2→// real agent would (register → heartbeat → request/reply) and implements every
     3→// agent-side request type from the SentraRMM contract.
     4→
     5→import { io, type Socket } from "socket.io-client";
     6→import type { DeviceMetrics } from "../../../src/lib/types";
     7→import { CONFIG, nextApi } from "../config";
     8→import { clamp, fmtBytes, jitter, pick, postJson, rndInt, round1, round4 } from "../util";
     9→import { runCommand, type ExecContext } from "./exec";
    10→import { runCommandCmd } from "./cmd";
    11→import { svgDesktop, svgPoster, svgWebcam, toDataUri } from "./media";
    12→import { ProcTable } from "./proc";
    13→import { SimRecorder } from "./record";
    14→import type { DeviceDef } from "./devices";
    15→import { seedDeviceFs, type Vfs } from "./vfs";
    16→
    17→interface RequestMsg {
    18→  requestId: string;
    19→  type: string;
    20→  payload?: Record<string, unknown>;
    21→}
    22→
    23→interface MediaCapture {
    24→  assetId: string;
    25→  kind: string;
    26→  quality: string;
    27→  withMic: boolean;
    28→  startedAt: number;
    29→  bytesPerSec: number;
    30→  timer: ReturnType<typeof setTimeout> | null;
    31→}
    32→
    33→interface NetCapture {
    34→  captureId: string;
    35→  iface: string;
    36→  filterBpf: string | null;
    37→  packets: number;
    38→  bytesCaptured: number;
    39→  timer: ReturnType<typeof setInterval> | null;
    40→}
    41→
    42→interface LiveStream {
    43→  seq: number;
    44→  fps: number;
    45→  startedAt: number;
    46→  timer: ReturnType<typeof setInterval> | null;
    47→}

    49→/** v0.3 multi-source stream (any number may run simultaneously). */
    50→interface SimStream {
    51→  streamId: string;
    52→  kind: "screen" | "camera" | "mic";
    53→  sourceId: string;
    54→  fps: number;
    55→  width: number;
    56→  height: number;
    57→  seq: number;
    58→  timer: ReturnType<typeof setInterval> | null;
    59→  startedAt: number;
    60→}
    61→
    62→const str = (v: unknown): string => (typeof v === "string" ? v : "");
    63→
    64→export class SimAgent {
    65→  readonly vfs: Vfs;
    66→  private readonly procTable: ProcTable;
    67→  private readonly socket: Socket;
    68→  private readonly bootTime: number;
    69→  private heartbeatTimer: ReturnType<typeof setInterval> | null = null;
    70→  private lastMetrics: DeviceMetrics | null = null;
    71→  private cpuHistory: number[] = [];
    72→  private readonly media = new Map<string, MediaCapture>();
    73→  private readonly net = new Map<string, NetCapture>();
    74→  private live: LiveStream | null = null;
    75→  private readonly streams = new Map<string, SimStream>();
    76→  private recorder: SimRecorder | null = null;
    77→  private cmdCwd: string;
    78→  private stopped = false;
    79→
    80→  constructor(readonly def: DeviceDef) {
    81→    this.vfs = seedDeviceFs(def);
    82→    this.procTable = new ProcTable(def);
    83→    this.bootTime = Date.now() - def.bootDaysAgo * 86_400_000 - rndInt(0, 3_600_000);
    84→    this.cmdCwd = `C:\\Users\\${def.userName}`;
    85→    this.socket = io(`http://127.0.0.1:${CONFIG.port}`, {
    86→      path: "/",
    87→      reconnectionDelayMax: 5000,
    88→    });
    89→  }
    90→
    91→  start(): void {
    92→    const s = this.socket;
    93→    s.on("connect", () => {
    94→      this.log(`connected to hub (socket ${s.id})`);
    95→      s.emit(
    96→        "agent:register",
    97→        { info: { ...this.def.info, captures: this.def.captures } },
    98→        (ack?: { ok?: boolean }) => {
    99→          if (ack?.ok) this.log("registered (captures: monitors/cameras/mics/shells reported)");
   100→          else this.log("registration rejected by hub");
   101→        },
   102→      );
   103→      this.ensureHeartbeat();
   104→    });
   105→    s.on("disconnect", (reason: string) => {
   106→      this.log(`disconnected from hub (${reason}) — will reconnect`);
   107→      // v0.3 contract: agents stop streaming when they disconnect. Recordings
   108→      // keep running and buffer chunks until the hub is reachable again.
   109→      this.stopAllStreams();
   110→    });
   111→    s.on("connect_error", (err: Error) => {
   112→      this.log(`hub connect error: ${err.message} — retrying`);
   113→    });
   114→    s.on("agent:request", (msg: RequestMsg) => {
   115→      void this.onRequest(msg);
   116→    });
   117→  }
   118→
   119→  stop(): void {
   120→    this.stopped = true;
   121→    if (this.heartbeatTimer) {
   122→      clearInterval(this.heartbeatTimer);
   123→      this.heartbeatTimer = null;
   124→    }
   125→    for (const cap of this.media.values()) {
   126→      if (cap.timer) clearTimeout(cap.timer);
   127→    }
   128→    this.media.clear();
   129→    for (const cap of this.net.values()) {
   130→      if (cap.timer) clearInterval(cap.timer);
   131→    }
   132→    this.net.clear();
   133→    if (this.live?.timer) clearInterval(this.live.timer);
   134→    this.live = null;
   135→    this.stopAllStreams();
   136→    if (this.recorder && !this.recorder.isDone()) this.recorder.cancel();
   137→    this.recorder = null;
   138→    this.socket.disconnect();
   139→  }
   140→
   141→  // ── telemetry ─────────────────────────────────────────────────────────
   142→
   143→  private ensureHeartbeat(): void {
   144→    if (this.heartbeatTimer || this.stopped) return;
   145→    this.heartbeatTimer = setInterval(() => this.beat(), CONFIG.heartbeatSec * 1000);
   146→  }
   147→
   148→  private beat(): void {
   149→    if (this.stopped || !this.socket.connected) return;
   150→    const metrics = this.buildMetrics();
   151→    this.socket.emit("agent:heartbeat", {
   152→      agentId: this.def.info.agentId,
   153→      metrics,
   154→      location: this.locationFix(),
   155→    });
   156→    this.log(
   157→      `heartbeat cpu=${metrics.cpuPercent}% mem=${metrics.memoryPercent}% disk=${metrics.diskPercent}% net=↑${metrics.netUpKBps}KBps ↓${metrics.netDownKBps}KBps`,
   158→    );
   159→  }
   160→
   161→  private buildMetrics(): DeviceMetrics {
   162→    const d = this.def;
   163→    const m: DeviceMetrics = {
   164→      deviceId: d.info.agentId,
   165→      cpuPercent: round1(clamp(jitter(d.cpuBase, d.cpuSpread), 3, 85)),
   166→      memoryPercent: round1(clamp(jitter(d.memBase, d.memSpread), 8, 96)),
   167→      diskPercent: d.diskPercent,
   168→      uptimeSec: Math.floor((Date.now() - this.bootTime) / 1000),
   169→      activeUsers: pick(d.activeUsers),
   170→      netUpKBps: Math.max(1, Math.round(jitter(d.netUpBase, d.netUpSpread))),
   171→      netDownKBps: Math.max(1, Math.round(jitter(d.netDownBase, d.netDownSpread))),
   172→      capturedAt: new Date().toISOString(),
   173→    };
   174→    this.lastMetrics = m;
   175→    this.cpuHistory.push(m.cpuPercent);
   176→    if (this.cpuHistory.length > 64) this.cpuHistory.shift();
   177→    return m;
   178→  }
   179→
   180→  private execCtx(): ExecContext {
   181→    const uptimeSec = Math.floor((Date.now() - this.bootTime) / 1000);
   182→    return {
   183→      def: this.def,
   184→      procTable: this.procTable,
   185→      vfs: this.vfs,
   186→      uptimeSec,
   187→      cpuPercent: this.lastMetrics?.cpuPercent ?? this.def.cpuBase,
   188→      memPercent: this.lastMetrics?.memoryPercent ?? this.def.memBase,
   189→      cwd: this.cmdCwd,
   190→    };
   191→  }
   192→
   193→  // ── request handling ──────────────────────────────────────────────────
   194→
   195→  private async onRequest(msg: RequestMsg): Promise<void> {
   196→    if (this.stopped || !msg || typeof msg.requestId !== "string" || typeof msg.type !== "string") return;
   197→    const payload = (msg.payload ?? {}) as Record<string, unknown>;
   198→    try {
   199→      const data = await this.handle(msg.type, payload);
   200→      this.socket.emit("agent:reply", { requestId: msg.requestId, ok: true, data });
   201→    } catch (err) {
   202→      const error = err instanceof Error ? err.message : String(err);
   203→      this.socket.emit("agent:reply", { requestId: msg.requestId, ok: false, error });
   204→      this.log(`${msg.type} failed: ${error}`);
   205→    }
   206→  }
   207→
   208→  private async handle(type: string, p: Record<string, unknown>): Promise<unknown> {
   209→    switch (type) {
   210→      case "exec": {
   211→        const command = str(p.command);
   212→        const wanted = str(p.shell) || "powershell";
   213→        const effective = wanted === "cmd" ? "cmd" : "powershell"; // sims always have powershell
   214→        const ctx = this.execCtx();
   215→        const r = effective === "cmd" ? await runCommandCmd(command, ctx) : await runCommand(command, ctx);
   216→        this.cmdCwd = ctx.cwd; // cmd `cd` persists across commands
   217→        this.log(
   218→          `exec [${effective}] '${command}' ${r.exitCode === 0 ? "ok" : `exit ${r.exitCode}`} (${r.durationMs}ms)`,
   219→        );
   220→        return r;
   221→      }
   222→      case "fs:list":
   223→        return this.vfs.list(str(p.path));
   224→      case "fs:download":
   225→        return this.vfs.download(str(p.path));
   226→      case "fs:upload":
   227→        return this.vfs.upload(str(p.path), str(p.name), str(p.contentBase64));
   228→      case "fs:delete":
   229→        return this.vfs.remove(str(p.path));
   230→      case "fs:rename":
   231→        return this.vfs.rename(str(p.path), str(p.newName));
   232→      case "fs:mkdir":
   233→        return this.vfs.mkdir(str(p.path));
   234→      case "ps:list":
   235→        return { processes: this.procTable.list(), metrics: this.buildMetrics() };
   236→      case "ps:action": {
   237→        const res = this.procTable.action(Number(p.pid), str(p.action));
   238→        this.log(`ps:action ${str(p.action)} pid=${p.pid} ok`);
   239→        return res;
   240→      }
   241→      case "media:screenshot":
   242→        return this.screenshot();
   243→      case "media:record":
   244→        return this.startMediaRecord(p);
   245→      case "media:stop":
   246→        return this.stopMediaRecord(str(p.assetId));
   247→      case "net:start":
   248→        return this.startNetCapture(p);
   249→      case "net:stop":
   250→        return this.stopNetCapture(str(p.captureId));
   251→      case "location:fix":
   252→        return this.locationFix();
   253→      case "screen:live:start":
   254→        return this.startLiveStream(p);
   255→      case "screen:live:stop":
   256→        return this.stopLiveStream();
   257→      case "stream:start":
   258→        return this.startStream(p);
   259→      case "stream:stop":
   260→        return this.stopStream(str(p.streamId));
   261→      case "record:schedule":
   262→        return this.scheduleRecording(p);
   263→      case "record:cancel":
   264→        return this.cancelRecording(str(p.recordingId));
   265→      default:
   266→        throw new Error(`Unknown request type '${type}'`);
   267→    }
   268→  }

   270→  // ── media ─────────────────────────────────────────────────────────────
   271→
   272→  private screenshot(): { dataUri: string; sizeBytes: number } {
   273→    const m = this.lastMetrics;
   274→    const svg = svgDesktop({
   275→      hostname: this.def.info.hostname,
   276→      ip: this.def.info.localIp,
   277→      osLabel: `${this.def.info.osName} ${this.def.info.osVersion}`,
   278→      agentVersion: this.def.info.agentVersion,
   279→      cpu: m?.cpuPercent ?? this.def.cpuBase,
   280→      mem: m?.memoryPercent ?? this.def.memBase,
   281→      disk: this.def.diskPercent,
   282→      // jittered per call → consecutive live-view frames differ (net meters + clock seconds)
   283→      netUpKBps: Math.max(4, Math.round(jitter(this.def.netUpBase, this.def.netUpSpread))),
   284→      netDownKBps: Math.max(16, Math.round(jitter(this.def.netDownBase, this.def.netDownSpread))),
   285→      uptimeSec: Math.floor((Date.now() - this.bootTime) / 1000),
   286→      cpuHistory: this.cpuHistory,
   287→      palette: this.def.palette,
   288→      accent: this.def.accent,
   289→      now: new Date(),
   290→    });
   291→    return { dataUri: toDataUri(svg), sizeBytes: Buffer.byteLength(svg, "utf8") };
   292→  }

   294→  private poster(cap: Pick<MediaCapture, "assetId" | "kind" | "quality">, elapsedSec: number): string {
   295→    return toDataUri(
   296→      svgPoster({
   297→        hostname: this.def.info.hostname,
   298→        kind: cap.kind,
   299→        quality: cap.quality,
   300→        assetId: cap.assetId,
   301→        elapsedSec,
   302→        palette: this.def.palette,
   303→        accent: this.def.accent,
   304→        camera: cap.kind === "camera_stream" || cap.kind === "call",
   305→        mic: cap.kind === "mic_record",
   306→      }),
   307→    );
   308→  }

   310→  private startMediaRecord(p: Record<string, unknown>): { started: true } {
   311→    const assetId = str(p.assetId) || `asset-${Date.now()}`;
   312→    const kind = str(p.kind) || "screen_record";
   313→    const quality = str(p.quality) || "720p";
   314→    const withMic = p.withMic === true;
   315→    const durationSec = typeof p.durationSec === "number" ? p.durationSec : 0;
   316→    if (this.media.has(assetId)) throw new Error(`Capture '${assetId}' is already running`);
   317→
   318→    const bytesPerSec =
   319→      kind === "mic_record"
   320→        ? 16_000
   321→        : kind === "camera_stream" || kind === "call"
   322→          ? 260_000
   323→          : quality === "1080p"
   324→            ? 760_000
   325→            : quality === "480p"
   326→              ? 190_000
   327→              : quality === "source"
   328→                ? 1_150_000
   329→                : 390_000;
   330→
   331→    const cap: MediaCapture = { assetId, kind, quality, withMic, startedAt: Date.now(), bytesPerSec, timer: null };
   332→    this.media.set(assetId, cap);
   333→    this.emitEvent("media:update", { id: assetId, status: "capturing", kind, quality, withMic });
   334→    this.log(`media:record ${assetId} capturing (${kind}, ${quality}, ${durationSec > 0 ? `${durationSec}s` : "indefinite"})`);
   335→
   336→    if (durationSec > 0) {
   337→      cap.timer = setTimeout(() => {
   338→        void this.finishMedia(cap, "available");
   339→      }, Math.min(durationSec, 3600) * 1000);
   340→    }
   341→    return { started: true };
   342→  }
   343→
   344→  private stopMediaRecord(assetId: string): { ok: true } {
   345→    const cap = this.media.get(assetId);
   346→    if (!cap) {
   347→      this.log(`media:stop ${assetId}: no active capture (already finished)`);
   348→      return { ok: true };
   349→    }
   350→    void this.finishMedia(cap, "stopped");
   351→    return { ok: true };
   352→  }
   353→
   354→  private async finishMedia(cap: MediaCapture, status: "available" | "stopped"): Promise<void> {
   355→    if (cap.timer) {
   356→      clearTimeout(cap.timer);
   357→      cap.timer = null;
   358→    }
   359→    if (!this.media.has(cap.assetId) || this.stopped) return;
   360→    this.media.delete(cap.assetId);
   361→    const elapsed = (Date.now() - cap.startedAt) / 1000;
   362→    const sizeBytes = Math.max(24_000, Math.round(cap.bytesPerSec * elapsed * (0.85 + Math.random() * 0.3)));
   363→    const dataRef = this.poster(cap, elapsed);
   364→    this.log(`media:record ${cap.assetId} ${status} (${fmtBytes(sizeBytes)}, ${Math.round(elapsed)}s)`);
   365→    this.emitEvent("media:update", { id: cap.assetId, status, sizeBytes });
   366→    await postJson(
   367→      nextApi("/api/internal/media/result"),
   368→      { assetId: cap.assetId, status, sizeBytes, dataRef },
   369→      { "x-hub-token": CONFIG.hubToken },
   370→    );
   371→  }

   373→  // ── live screen stream ────────────────────────────────────────────────
   374→
   375→  private startLiveStream(p: Record<string, unknown>): { started: true; fps: number } {
   376→    if (this.live) throw new Error("A live screen stream is already running");
   377→    const fps = Math.min(3, Math.max(0.5, typeof p.fps === "number" && p.fps > 0 ? p.fps : 1));
   378→    const stream: LiveStream = { seq: 0, fps, startedAt: Date.now(), timer: null };
   379→    this.live = stream;
   380→    const interval = Math.round(1000 / fps);
   381→    stream.timer = setInterval(() => {
   382→      if (!this.live || this.stopped || !this.socket.connected) return;
   383→      stream.seq += 1;
   384→      const shot = this.screenshot();
   385→      this.socket.emit("agent:event", {
   386→        name: "screen:frame",
   387→        payload: {
   388→          agentId: this.def.info.agentId,
   389→          dataUri: shot.dataUri,
   390→          ts: Date.now(),
   391→          seq: stream.seq,
   392→          width: 1600,
   393→          height: 900,
   394→        },
   395→      });
   396→    }, interval);
   397→    this.log(`screen:live started (${fps} fps)`);
   398→    return { started: true, fps };
   399→  }
   400→
   401→  private stopLiveStream(): { frames: number } {
   402→    const stream = this.live;
   403→    if (!stream) return { frames: 0 };
   404→    if (stream.timer) clearInterval(stream.timer);
   405→    this.live = null;
   406→    this.log(`screen:live stopped (${stream.seq} frames streamed)`);
   407→    return { frames: stream.seq };
   408→  }

   410→  // ── v0.3 multi-source streams ─────────────────────────────────────────

   412→  private num(v: unknown): number | null {
   413→    return typeof v === "number" && Number.isFinite(v) ? v : null;
   414→  }
   415→
   416→  private startStream(p: Record<string, unknown>): { started: true; meta: Record<string, unknown> } {
   417→    const streamId = str(p.streamId);
   418→    if (!streamId) throw new Error("streamId is required");
   419→    const kind = str(p.kind);
   420→    if (kind !== "screen" && kind !== "camera" && kind !== "mic") {
   421→      throw new Error("kind must be one of: screen, camera, mic");
   422→    }
   423→    if (this.streams.has(streamId)) throw new Error(`Stream '${streamId}' is already running`);
   424→    const caps = this.def.captures;
   425→
   426→    let stream: SimStream;
   427→    let meta: Record<string, unknown>;
   428→    if (kind === "screen") {
   429→      const sourceId = str(p.sourceId) || caps.monitors[0]?.id || "monitor:0";
   430→      const idx = caps.monitors.findIndex((m) => m.id === sourceId);
   431→      if (idx < 0) {
   432→        throw new Error(
   433→          `Unknown monitor '${sourceId}' — available: ${caps.monitors.map((m) => m.id).join(", ") || "none"}`,
   434→        );
   435→      }
   436→      const mon = caps.monitors[idx]!;
   437→      const fps = clamp(this.num(p.fps) ?? 2, 0.5, 5);
   438→      stream = { streamId, kind, sourceId, fps, width: mon.width, height: mon.height, seq: 0, timer: null, startedAt: Date.now() };
   439→      meta = { width: mon.width, height: mon.height, fps, sourceId, name: mon.name, primary: mon.primary };
   440→    } else if (kind === "camera") {
   441→      const sourceId = str(p.sourceId) || caps.cameras[0]?.id || "camera:0";
   442→      const idx = caps.cameras.findIndex((c) => c.id === sourceId);
   443→      if (idx < 0) {
   444→        throw new Error(`Unknown camera '${sourceId}' — available: ${caps.cameras.map((c) => c.id).join(", ") || "none"}`);
   445→      }
   446→      const fps = clamp(this.num(p.fps) ?? 10, 1, 15);
   447→      stream = { streamId, kind, sourceId, fps, width: 1280, height: 720, seq: 0, timer: null, startedAt: Date.now() };
   448→      meta = { width: 1280, height: 720, fps, sourceId, name: caps.cameras[idx]!.name };
   449→    } else {
   450→      const sourceId = str(p.sourceId) || caps.mics[0]?.id || "mic:0";
   451→      const idx = caps.mics.findIndex((m) => m.id === sourceId);
   452→      if (idx < 0) {
   453→        throw new Error(`Unknown mic '${sourceId}' — available: ${caps.mics.map((m) => m.id).join(", ") || "none"}`);
   454→      }
   455→      stream = { streamId, kind, sourceId, fps: 8, width: 0, height: 0, seq: 0, timer: null, startedAt: Date.now() };
   456→      meta = { fps: 8, sourceId, name: caps.mics[idx]!.name };
   457→    }
   458→
   459→    this.streams.set(streamId, stream);
   460→    const interval = Math.round(1000 / stream.fps);
   461→    stream.timer = setInterval(() => this.emitStreamFrame(stream), interval);
   462→    this.log(`stream:start ${kind} '${stream.sourceId}' @${stream.fps}fps (streamId ${streamId.slice(0, 8)}…, ${this.streams.size} active)`);
   463→    return { started: true, meta };
   464→  }
   465→
   466→  private emitStreamFrame(stream: SimStream): void {
   467→    if (this.stopped || !this.socket.connected) return;
   468→    stream.seq += 1;
   469→    const frame: Record<string, unknown> = {
   470→      agentId: this.def.info.agentId,
   471→      streamId: stream.streamId,
   472→      kind: stream.kind,
   473→      ts: Date.now(),
   474→      seq: stream.seq,
   475→    };
   476→    if (stream.kind === "screen") {
   477→      // render at the monitor's advertised resolution so frames match meta
   478→      const shot = this.renderScreenSource(stream.sourceId, stream.width, stream.height);
   479→      frame.dataUri = shot.dataUri;
   480→      frame.width = shot.width;
   481→      frame.height = shot.height;
   482→    } else if (stream.kind === "camera") {
   483→      const idx = Math.max(0, this.def.captures.cameras.findIndex((c) => c.id === stream.sourceId));
   484→      const shot = this.renderCameraSource(stream.sourceId, idx);
   485→      frame.dataUri = shot.dataUri;
   486→      frame.width = 1280;
   487→      frame.height = 720;
   488→    } else {
   489→      frame.level = this.micLevel();
   490→    }
   491→    this.socket.emit("agent:event", { name: "stream:frame", payload: frame });
   492→  }
   493→
   494→  private stopStream(streamId: string): { frames: number } {
   495→    const stream = this.streams.get(streamId);
   496→    if (!stream) return { frames: 0 };
   497→    if (stream.timer) clearInterval(stream.timer);
   498→    this.streams.delete(streamId);
   499→    this.log(`stream:stop ${stream.kind} '${stream.sourceId}' (${stream.seq} frames streamed, ${this.streams.size} active)`);
   500→    return { frames: stream.seq };
   501→  }

   553→  /** Mic input level 0.05..0.9 (sinusoid + noise), 8 Hz cadence. */
   554→  private micLevel(): number {
   555→    const t = Date.now() / 1000;
   556→    const base = 0.475 + 0.32 * Math.sin(t * 2.2) + 0.1 * Math.sin(t * 5.7);
   557→    const noise = (Math.random() * 2 - 1) * 0.08;
   558→    return Number(clamp(base + noise, 0.05, 0.9).toFixed(3));
   559→  }

   561→  // ── v0.3 scheduled recordings ─────────────────────────────────────────
   562→
   563→  private scheduleRecording(p: Record<string, unknown>): { accepted: true; state: "armed" } {
   564→    const recordingId = str(p.recordingId);
   565→    if (!recordingId) throw new Error("recordingId is required");
   566→    if (this.recorder && !this.recorder.isDone()) {
   567→      throw new Error("A recording is already active on this device");
   568→    }
   569→    const passphrase = str(p.passphrase);
   570→    if (!passphrase) throw new Error("passphrase is required");
   571→    const durationSec = clamp(Math.round(this.num(p.durationSec) ?? 0), 5, 120); // sim hard cap: 120s
   572→    const list = (v: unknown): string[] | undefined =>
   573→      Array.isArray(v) && v.length > 0 ? v.filter((x): x is string => typeof x === "string") : undefined;
   574→    const sources = {
   575→      ...(list(p.monitors) ? { monitors: list(p.monitors) } : {}),
   576→      ...(list(p.cameras) ? { cameras: list(p.cameras) } : {}),
   577→      ...(list(p.mics) ? { mics: list(p.mics) } : {}),
   578→    };
   579→    const label = str(p.label) || null;
   580→    this.log(
   581→      `record:schedule ${recordingId} armed (${durationSec}s) — on-device notice shown, indicator will be visible while recording`,
   582→    );
   583→    this.recorder = new SimRecorder({
   584→      socket: this.socket,
   585→      agentId: this.def.info.agentId,
   586→      hostname: this.def.info.hostname,
   587→      recordingId,
   588→      durationSec,
   589→      passphrase,
   590→      label,
   591→      sources,
   592→      renderScreen: (sourceId) => this.renderScreenSource(sourceId, 1280, 720),
   593→      renderCamera: (sourceId) =>
   594→        this.renderCameraSource(sourceId, Math.max(0, this.def.captures.cameras.findIndex((c) => c.id === sourceId))),
   595→      micLevel: () => this.micLevel(),
   596→      log: (msg) => this.log(msg),
   597→    });
   598→    return { accepted: true, state: "armed" };
   599→  }
   600→
   601→  private cancelRecording(recordingId: string): { ok: true; state: string } {
   602→    if (!this.recorder || this.recorder.isDone()) {
   603→      this.log(`record:cancel ${recordingId}: no active recording`);
   604→      return { ok: true, state: "idle" };
   605→    }
   606→    this.recorder.cancel();
   607→    return { ok: true, state: "canceled" };
   608→  }

   610→  // ── network capture ───────────────────────────────────────────────────
   611→
   612→  private startNetCapture(p: Record<string, unknown>): { started: true } {
   613→    const captureId = str(p.captureId) || `cap-${Date.now()}`;
   614→    if (this.net.has(captureId)) throw new Error(`Capture '${captureId}' is already running`);
   615→    const cap: NetCapture = {
   616→      captureId,
   617→      iface: str(p.iface) || "Ethernet",
   618→      filterBpf: str(p.filterBpf) || null,
   619→      packets: 0,
   620→      bytesCaptured: 0,
   621→      timer: null,
   622→    };
   623→    this.net.set(captureId, cap);
   624→    cap.timer = setInterval(() => {
   625→      cap.packets += rndInt(20, 80);
   626→      cap.bytesCaptured += rndInt(2_048, 61_440);
   627→      this.emitEvent("network:update", {
   628→        id: captureId,
   629→        status: "running",
   630→        packets: cap.packets,
   631→        bytesCaptured: cap.bytesCaptured,
   632→      });
   633→      void postJson(
   634→        nextApi("/api/internal/network/stats"),
   635→        { captureId, packets: cap.packets, bytesCaptured: cap.bytesCaptured },
   636→        { "x-hub-token": CONFIG.hubToken },
   637→      );
   638→    }, 3000);
   639→    this.log(`net:capture ${captureId} started on ${cap.iface}${cap.filterBpf ? ` (bpf: ${cap.filterBpf})` : ""}`);
   640→    return { started: true };
   641→  }
   642→
   643→  private stopNetCapture(captureId: string): { packets: number; bytesCaptured: number } {
   644→    const cap = this.net.get(captureId);
   645→    if (!cap) throw new Error(`No active capture '${captureId}'`);
   646→    if (cap.timer) {
   647→      clearInterval(cap.timer);
   648→      cap.timer = null;
   649→    }
   650→    this.net.delete(captureId);
   651→    this.log(`net:capture ${captureId} stopped (${cap.packets} packets, ${fmtBytes(cap.bytesCaptured)})`);
   652→    this.emitEvent("network:update", {
   653→      id: captureId,
   654→      status: "stopped",
   655→      packets: cap.packets,
   656→      bytesCaptured: cap.bytesCaptured,
   657→    });
   658→    return { packets: cap.packets, bytesCaptured: cap.bytesCaptured };
   659→  }
