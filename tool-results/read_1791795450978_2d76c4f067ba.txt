   129→function fmtElapsed(ms: number): string {
   130→  const s = Math.max(0, Math.floor(ms / 1000))
   131→  const h = Math.floor(s / 3600)
   132→  const m = Math.floor((s % 3600) / 60)
   133→  const sec = s % 60
   134→  const pad = (n: number) => String(n).padStart(2, '0')
   135→  return h > 0 ? `${h}:${pad(m)}:${pad(sec)}` : `${m}:${pad(sec)}`
   136→}
   137→
   138→function fmtDurationLabel(durationSec: number | null): string {
   139→  if (!durationSec) return 'indefinite'
   140→  if (durationSec < 60) return `${durationSec}s`
   141→  if (durationSec < 3600) return `${Math.floor(durationSec / 60)} min`
   142→  return `${Math.floor(durationSec / 3600)} h`
   143→}

   147→/**
   148→ * Live screen viewer — one-way socket stream (`screen:frame`).
   149→ * Frames are rendered straight from the store; nothing is persisted.
   150→ */
   151→function LiveViewDialog({
   152→  device,
   153→  open,
   154→  onOpenChange,
   155→  now,
   156→}: {
   157→  device: Device | null
   158→  open: boolean
   159→  onOpenChange: (o: boolean) => void
   160→  now: number
   161→}) {
   162→  const liveFrame = useDash((s) => s.liveFrame)
   163→  const clearLiveFrame = useDash((s) => s.setLiveFrame)
   164→  const { data: devices } = useDevices()
   165→  const [fps, setFps] = useState(0)
   166→  const frameTimes = useRef<number[]>([])
   167→  const startedAt = useRef<number>(Date.now())
   168→
   169→  const frame = useMemo(() => {
   170→    if (!liveFrame || !device) return null
   171→    if (liveFrame.agentId === device.id) return liveFrame
   172→    const matchesAgent = (devices ?? []).some(
   173→      (d) => d.id === device.id && d.agentId === liveFrame.agentId,
   174→    )
   175→    return matchesAgent ? liveFrame : null
   176→  }, [liveFrame, device, devices])
   177→
   178→  // record frame arrivals (one effect per frame, no timers)
   179→  useEffect(() => {
   180→    if (!open || frame?.seq == null) return
   181→    const t = Date.now()
   182→    frameTimes.current = [...frameTimes.current, t].filter((x) => t - x <= 4000)
   183→  }, [open, frame?.seq])
   184→
   185→  // FPS meter — independent 1s ticker over the trailing 4s window
   186→  useEffect(() => {
   187→    if (!open) return
   188→    const iv = setInterval(() => {
   189→      const t = Date.now()
   190→      frameTimes.current = frameTimes.current.filter((x) => t - x <= 4000)
   191→      setFps(Math.round((frameTimes.current.length / 4) * 10) / 10)
   192→    }, 1000)
   193→    return () => clearInterval(iv)
   194→  }, [open])
   195→
   196→  useEffect(() => {
   197→    if (open) startedAt.current = Date.now()
   198→  }, [open])
   199→
   200→  const stop = useMutation({
   201→    mutationFn: () =>
   202→      api<{ ok: true }>('/api/media/live', {
   203→        method: 'POST',
   204→        body: JSON.stringify({ deviceId: device?.id, action: 'stop' }),
   205→      }),
   206→    onSettled: () => clearLiveFrame(null),
   207→  })
   208→
   209→  const handleClose = (o: boolean) => {
   210→    if (!o && open) stop.mutate()
   211→    onOpenChange(o)
   212→  }
   213→
   214→  return (
   215→    <Dialog open={open} onOpenChange={handleClose}>
   216→      <DialogContent className="glass-strong sm:max-w-4xl">
   217→        <DialogHeader>
   218→          <DialogTitle className="flex items-center gap-2">
   219→            <span className="relative flex h-2.5 w-2.5">
   220→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-rose-500 opacity-70" />
   221→              <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-rose-500" />
   222→            </span>
   223→            Live view — {device?.hostname ?? 'device'}
   224→          </DialogTitle>
   225→          <DialogDescription>
   226→            Realtime one-way stream over the agent socket · never persisted · encrypted transport
   227→          </DialogDescription>
   228→        </DialogHeader>
   229→
   230→        <div className="overflow-hidden rounded-xl ring-1 ring-border/60">
   231→          {frame ? (
   232→            <img
   233→              src={frame.dataUri}
   234→              alt={`Live screen of ${device?.hostname ?? 'device'}`}
   235→              className="w-full object-contain"
   236→            />
   237→          ) : (
   238→            <div className="terminal-glass flex h-72 w-full flex-col items-center justify-center gap-3">
   239→              <Loader2 className="h-6 w-6 animate-spin text-primary" />
   240→              <p className="text-sm text-muted-foreground">Waiting for first frame…</p>
   241→            </div>
   242→          )}
   243→        </div>
   244→
   245→        <div className="flex flex-wrap items-center justify-between gap-3 text-xs text-muted-foreground">
   246→          <div className="flex flex-wrap items-center gap-2">
   247→            <Badge variant="outline" className="border-rose-500/40 bg-rose-500/10 text-rose-600 dark:text-rose-400">
   248→              LIVE
   249→            </Badge>
   250→            <Badge variant="secondary" className="font-mono text-[10px]">{fps} fps</Badge>
   251→            <Badge variant="secondary" className="font-mono text-[10px]">
   252→              frame #{frame?.seq ?? 0}
   253→            </Badge>
   254→            <Badge variant="secondary" className="font-mono text-[10px]">1600×900</Badge>
   255→          </div>
   256→          <div className="flex items-center gap-3">
   257→            <span className="font-mono">{fmtElapsed(now - startedAt.current)} elapsed</span>
   258→            <Button
   259→              size="sm"
   260→              variant="destructive"
   261→              className="h-9"
   262→              disabled={stop.isPending}
   263→              onClick={() => handleClose(false)}
   264→            >
   265→              <Square className="h-3.5 w-3.5" /> Stop stream
   266→            </Button>
   267→          </div>
   268→        </div>
   269→      </DialogContent>
   270→    </Dialog>
   271→  )
   272→}
