   146→function fmtElapsed(ms: number): string {
   147→  const s = Math.max(0, Math.floor(ms / 1000))
   148→  const h = Math.floor(s / 3600)
   149→  const m = Math.floor((s % 3600) / 60)
   150→  const sec = s % 60
   151→  const pad = (n: number) => String(n).padStart(2, '0')
   152→  return h > 0 ? `${h}:${pad(m)}:${pad(sec)}` : `${m}:${pad(sec)}`
   153→}
   154→
   155→function fmtDurationLabel(durationSec: number | null): string {
   156→  if (!durationSec) return 'indefinite'
   157→  if (durationSec < 60) return `${durationSec}s`
   158→  if (durationSec < 3600) return `${Math.floor(durationSec / 60)} min`
   159→  return `${Math.floor(durationSec / 3600)} h`
   160→}

   219→/**
   220→ * Animated input-level meter driven by `stream:frame` level (0..1, ~8 Hz).
   221→ * The fill eases via CSS transition; a peak-hold marker decays slowly so
   222→ * transient spikes stay readable.
   223→ */
   224→function MicLevelMeter({ level }: { level: number }) {
   225→  const peakRef = useRef(0)
   226→  const [peak, setPeak] = useState(0)
   227→
   228→  useEffect(() => {
   229→    // rise instantly, decay ~0.02 per frame (~8 Hz updates)
   230→    const next = Math.max(level, peakRef.current - 0.02)
   231→    peakRef.current = next
   232→    setPeak(next)
   233→  }, [level])
   234→
   235→  const pct = Math.min(100, Math.max(0, Math.round(level * 100)))
   236→  const peakPct = Math.min(99, Math.max(1, Math.round(peak * 100)))
   237→
   238→  return (
   239→    <div className="flex flex-col items-center justify-center gap-5 px-6 py-12">
   240→      <div className="glass-inset flex h-14 w-14 items-center justify-center rounded-2xl">
   241→        <Mic
   242→          className={cn(
   243→            'h-6 w-6 transition-colors duration-200',
   244→            pct > 4 ? 'text-rose-500' : 'text-muted-foreground',
   245→          )}
   246→          aria-hidden
   247→        />
   248→      </div>
   249→      <div
   250→        role="meter"
   251→        aria-valuemin={0}
   252→        aria-valuemax={100}
   253→        aria-valuenow={pct}
   254→        aria-label="Microphone input level"
   255→        className="glass-inset relative h-3.5 w-full max-w-sm overflow-hidden rounded-full"
   256→      >
   257→        <div
   258→          className="h-full rounded-full bg-gradient-to-r from-emerald-500 via-teal-400 to-rose-400 transition-[width] duration-150 ease-out"
   259→          style={{ width: `${Math.max(2, pct)}%` }}
   260→        />
   261→        {/* peak-hold marker */}
   262→        <span
   263→          aria-hidden
   264→          className="absolute top-0 h-full w-0.5 rounded-full bg-foreground/60 transition-[left] duration-150 ease-out"
   265→          style={{ left: `calc(${peakPct}% - 1px)` }}
   266→        />
   267→      </div>
   268→      <p className="font-mono text-xs tabular-nums text-muted-foreground">
   269→        {pct}% input level · live
   270→      </p>
   271→    </div>
   272→  )
   273→}

   277→function StreamPanel({
   278→  stream,
   279→  now,
   280→  onStop,
   281→  stopping,
   282→}: {
   283→  stream: ActiveStream
   284→  now: number
   285→  onStop: (streamId: string) => void
   286→  stopping: boolean
   287→}) {
   288→  const frame = useDash((s) => s.streamFrames[stream.streamId])
   289→  const { data: devices } = useDevices()
   290→  const hostname =
   291→    devices?.find((d) => d.id === stream.deviceId)?.hostname ?? stream.agentId
   292→
   293→  // measured fps — trailing 5s window of frame arrivals, 1s ticker
   294→  const timesRef = useRef<number[]>([])
   295→  const [measuredFps, setMeasuredFps] = useState(0)
   296→  const seq = frame?.seq ?? 0
   297→
   298→  useEffect(() => {
   299→    if (seq === 0) return
   300→    const t = Date.now()
   301→    timesRef.current = [...timesRef.current, t].filter((x) => t - x <= 5000)
   302→  }, [seq, frame?.at])
   303→
   304→  useEffect(() => {
   305→    const iv = setInterval(() => {
   306→      const t = Date.now()
   307→      timesRef.current = timesRef.current.filter((x) => t - x <= 5000)
   308→      setMeasuredFps(Math.round((timesRef.current.length / 5) * 10) / 10)
   309→    }, 1000)
   310→    return () => clearInterval(iv)
   311→  }, [])
   312→
   313→  const meta = STREAM_KIND_META[stream.kind]
   314→  const Icon = meta.icon
   315→  const hasImage = stream.kind !== 'mic'
   316→
   317→  return (
   318→    <motion.div
   319→      layout
   320→      variants={{
   321→        hidden: { opacity: 0, y: 14, scale: 0.98 },
   322→        show: {
   323→          opacity: 1,
   324→          y: 0,
   325→          scale: 1,
   326→          transition: { duration: 0.28, ease: 'easeOut' },
   327→        },
   328→      }}
   329→      exit={{ opacity: 0, scale: 0.96, transition: { duration: 0.18 } }}
   330→      className="glass glass-hover lift flex flex-col overflow-hidden rounded-2xl"
   331→    >
   332→      {/* header */}
   333→      <div className="flex items-center gap-2.5 px-3.5 pb-2 pt-3">
   334→        <span className="relative flex h-2.5 w-2.5 shrink-0" aria-hidden>
   335→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-rose-500 opacity-70" />
   336→          <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-rose-500" />
   337→        </span>
   338→        <div className="glass-inset flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-primary">
   339→          <Icon className="h-3.5 w-3.5" aria-hidden />
   340→        </div>
   341→        <div className="min-w-0 flex-1">
   342→          <p className="truncate text-sm font-semibold leading-tight">{stream.sourceName}</p>
   343→          <p className="truncate text-[11px] text-muted-foreground">
   344→            {meta.label} · {hostname}
   345→          </p>
   346→        </div>
   347→        <Badge
   348→          variant="outline"
   349→          className="shrink-0 border-rose-500/40 bg-rose-500/10 text-[10px] font-bold tracking-wider text-rose-600 dark:text-rose-400"
   350→        >
   351→          LIVE
   352→        </Badge>
   353→      </div>
   354→
   355→      {/* body */}
   356→      {hasImage ? (
   357→        frame?.dataUri ? (
   358→          <img
   359→            src={frame.dataUri}
   360→            alt={`Live ${stream.kind} stream from ${stream.sourceName}`}
   361→            className="max-h-56 w-full bg-black/50 object-contain sm:max-h-64"
   362→            decoding="async"
   363→          />
   364→        ) : (
   365→          <div className="flex h-40 w-full flex-col items-center justify-center gap-2 bg-black/30">
   366→            <Loader2 className="h-5 w-5 animate-spin text-emerald-400" aria-hidden />
   367→            <p className="text-xs text-zinc-400">waiting for first frame…</p>
   368→          </div>
   369→        )
   370→      ) : (
   371→        <MicLevelMeter level={frame?.level ?? 0} />
   372→      )}
   373→
   374→      {/* footer */}
   375→      <div className="flex flex-wrap items-center gap-1.5 px-3.5 py-2.5 text-[11px] text-muted-foreground">
   376→        <Badge variant="secondary" className="font-mono text-[10px] tabular-nums">
   377→          {measuredFps > 0 ? `${measuredFps.toFixed(1)} fps` : `${stream.fps || '—'} fps target`}
   378→        </Badge>
   379→        <Badge variant="secondary" className="font-mono text-[10px] tabular-nums">
   380→          #{seq}
   381→        </Badge>
   382→        <Badge variant="secondary" className="font-mono text-[10px] tabular-nums">
   383→          {fmtElapsed(now - stream.startedAt)}
   384→        </Badge>
   385→        {hasImage && stream.quality ? (
   386→          <Badge variant="secondary" className="font-mono text-[10px]">
   387→            {stream.quality}
   388→          </Badge>
   389→        ) : null}
   390→        <Button
   391→          size="sm"
   392→          variant="outline"
   393→          className="ml-auto h-8 shrink-0 border-rose-500/40 text-rose-600 hover:bg-rose-500/10 hover:text-rose-700 dark:text-rose-400 dark:hover:text-rose-300"
   394→          disabled={stopping}
   395→          onClick={() => onStop(stream.streamId)}
   396→        >
   397→          {stopping ? (
   398→            <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden />
   399→          ) : (
   400→            <Square className="h-3.5 w-3.5" aria-hidden />

   402→          Stop

   563→  const startedAt = useRef<number>(Date.now())
   564→
   565→  const frame = useMemo(() => {
   566→    if (!liveFrame || !device) return null
   567→    if (liveFrame.agentId === device.id) return liveFrame
   568→    const matchesAgent = (devices ?? []).some(
   569→      (d) => d.id === device.id && d.agentId === liveFrame.agentId,
   570→    )
   571→    return matchesAgent ? liveFrame : null
   572→  }, [liveFrame, device, devices])
   573→
   574→  // record frame arrivals (one effect per frame, no timers)
   575→  useEffect(() => {
   576→    if (!open || frame?.seq == null) return
   577→    const t = Date.now()
   578→    frameTimes.current = [...frameTimes.current, t].filter((x) => t - x <= 4000)
   579→  }, [open, frame?.seq])
   580→
   581→  // FPS meter — independent 1s ticker over the trailing 4s window
   582→  useEffect(() => {
   583→    if (!open) return
   584→    const iv = setInterval(() => {
   585→      const t = Date.now()
   586→      frameTimes.current = frameTimes.current.filter((x) => t - x <= 4000)
   587→      setFps(Math.round((frameTimes.current.length / 4) * 10) / 10)
   588→    }, 1000)
   589→    return () => clearInterval(iv)
   590→  }, [open])
   591→
   592→  useEffect(() => {
   593→    if (open) startedAt.current = Date.now()
   594→  }, [open])
   595→
   596→  const stop = useMutation({
   597→    mutationFn: () =>
   598→      api<{ ok: true }>('/api/media/live', {
   599→        method: 'POST',
   600→        body: JSON.stringify({ deviceId: device?.id, action: 'stop' }),
   601→      }),
   602→    onSettled: () => clearLiveFrame(null),
   603→  })
   604→
   605→  const handleClose = (o: boolean) => {
   606→    if (!o && open) stop.mutate()
   607→    onOpenChange(o)
   608→  }
   609→
   610→  return (
   611→    <Dialog open={open} onOpenChange={handleClose}>
   612→      <DialogContent className="glass-strong sm:max-w-4xl">
   613→        <DialogHeader>
   614→          <DialogTitle className="flex items-center gap-2">
   615→            <span className="relative flex h-2.5 w-2.5">
   616→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-rose-500 opacity-70" />
   617→              <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-rose-500" />
   618→            </span>
   619→            Live view — {device?.hostname ?? 'device'}
   620→          </DialogTitle>
   621→          <DialogDescription>
   622→            Realtime one-way stream over the agent socket · never persisted · encrypted transport
   623→          </DialogDescription>
   624→        </DialogHeader>
   625→
   626→        <div className="overflow-hidden rounded-xl ring-1 ring-border/60">
   627→          {frame ? (
   628→            <img
   629→              src={frame.dataUri}
   630→              alt={`Live screen of ${device?.hostname ?? 'device'}`}
   631→              className="w-full object-contain"
   632→            />
   633→          ) : (
   634→            <div className="terminal-glass flex h-72 w-full flex-col items-center justify-center gap-3">
   635→              <Loader2 className="h-6 w-6 animate-spin text-primary" />

   637→            </div>
   638→          )}
   639→        </div>
   640→
   641→        <div className="flex flex-wrap items-center justify-between gap-3 text-xs text-muted-foreground">
   642→          <div className="flex flex-wrap items-center gap-2">
   643→            <Badge variant="outline" className="border-rose-500/40 bg-rose-500/10 text-rose-600 dark:text-rose-400">
   644→              LIVE
   645→            </Badge>
   646→            <Badge variant="secondary" className="font-mono text-[10px]">{fps} fps</Badge>
   647→            <Badge variant="secondary" className="font-mono text-[10px]">
   648→              frame #{frame?.seq ?? 0}
   649→            </Badge>
   650→            <Badge variant="secondary" className="font-mono text-[10px]">1600×900</Badge>
   651→          </div>
   652→          <div className="flex items-center gap-3">
   653→            <span className="font-mono">{fmtElapsed(now - startedAt.current)} elapsed</span>
   654→            <Button

   656→              variant="destructive"
   657→              className="h-9"

   659→              onClick={() => handleClose(false)}

   662→            </Button>
   663→          </div>
   664→        </div>
   665→      </DialogContent>
