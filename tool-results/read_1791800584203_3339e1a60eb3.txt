   291→// ───────────────────────── Device health score ──────────────────────
   292→
   293→/**
   294→ * Composite 0-100 health score from live telemetry (frontend-computed).
   295→ * 100 = idle healthy machine; deductions for CPU/MEM/DISK pressure + penalties
   296→ * for stale telemetry or offline status.
   297→ */
   298→export function computeHealthScore(input: {
   299→  online: boolean;
   300→  cpuPercent?: number | null;
   301→  memoryPercent?: number | null;
   302→  diskPercent?: number | null;
   303→  stale?: boolean;
   304→}): number {
   305→  if (!input.online) return 40;
   306→  let score = 100;
   307→  const cpu = input.cpuPercent ?? 0;
   308→  const mem = input.memoryPercent ?? 0;
   309→  const disk = input.diskPercent ?? 0;
   310→  score -= Math.max(0, cpu - 30) * 0.5; // heavy CPU load hurts most
   311→  score -= Math.max(0, mem - 55) * 0.6; // memory pressure is critical
   312→  score -= Math.max(0, disk - 80) * 0.8; // nearly-full disks are a risk
   313→  if (input.stale) score -= 10;
   314→  return Math.max(0, Math.min(100, Math.round(score)));
   315→}

   317→// ─────────────────────────── Alert rules engine ────────────────────
   318→
   319→export type AlertMetric = "cpu" | "memory" | "disk";
   320→export type AlertRuleOp = "gt" | "lt";
   321→
   322→/**
   323→ * A threshold rule evaluated against every device heartbeat. Rules live in
   324→ * the Setting table under key "alertRules" (JSON array). When a device's
   325→ * telemetry crosses the threshold, an Alert row is created — rate-limited by
   326→ * `cooldownMin` per device+rule and suppressed while maintenance is open.
   327→ */
   328→export interface AlertRule {
   329→  id: string;
   330→  name: string; // "High CPU load" — used in the alert title
   331→  metric: AlertMetric;
   332→  op: AlertRuleOp; // gt = above, lt = below
   333→  threshold: number; // percent 0-100
   334→  severity: "info" | "warning" | "critical";
   335→  cooldownMin: number; // min minutes between repeat alerts for the same device
   336→  enabled: boolean;
   337→  /**
   338→   * Optional tag scope — when set, the rule only evaluates against devices
   339→   * whose tags include this value (e.g. "production"). null/undefined = all
   340→   * devices.
   341→   */
   342→  deviceTag?: string | null;
   343→}

   503→// ───────────────────────── Socket.io contract ──────────────────────
   504→// Hub runs on port 3003 (connect from browser: io("/?XTransformPort=3003", { path: "/" }))
   505→//
   506→// Client (dashboard) → hub:
   507→//   "dashboard:join"            → joins "dashboard" room
   508→// Hub → dashboard:
   509→//   "device:update"   { device: Device }
   510→//   "metrics:update"  { metrics: DeviceMetrics }
   511→//   "command:result"  { command: CommandExecution }
   512→//   "media:update"    { asset: MediaAsset }
   513→//   "network:update"  { capture: NetworkCapture }
   514→//   "alert:new"       { alert: Alert }
   515→//   "screen:frame"    { agentId, dataUri, ts, seq, width, height }  (live screen viewer stream)
   516→// Hub internal (agent ↔ hub):
   517→//   "agent:register"  { info: AgentInfo }            ack → { ok }
   518→//   "agent:heartbeat" { agentId, metrics, location? }
   519→//   "agent:request"   { requestId, type, payload }   (hub → agent)
   520→//   "agent:reply"     { requestId, ok, data?, error? } (agent → hub)
   521→//
   522→// agent request types handled by agents:
   523→//   "exec"        { command }              → { output, exitCode, durationMs }
   524→//   "fs:list"     { path }                 → FsListing
   525→//   "fs:download" { path }                 → FsDownload
   526→//   "fs:upload"   { path, name, contentBase64 } → { path }
   527→//   "fs:delete"   { path }                 → { ok }
   528→//   "fs:rename"   { path, newName }        → { path }
   529→//   "fs:mkdir"    { path }                 → { path }
   530→//   "ps:list"     {}                       → { processes: ProcessInfo[], metrics: DeviceMetrics }
   531→//   "ps:action"   { pid, action }          → { ok }
   532→//   "media:screenshot" { }                 → { dataUri, sizeBytes }
   533→//   "media:record"     { assetId, kind, quality, durationSec, withMic } → ack { started } (async, results via HTTP)
   534→//   "media:stop"       { assetId }          → { ok }
   535→//   "net:start"   { captureId, iface, filterBpf } → ack { started } (stats via HTTP)
   536→//   "net:stop"    { captureId }            → { packets, bytesCaptured }
   537→//   "location:fix" {}                      → { lat, lon, accuracyM, label, source }
   538→//   "screen:live:start" { fps }            → ack { started, fps } (frames via `screen:frame` until stop)
   539→//   "screen:live:stop"  {}                 → { frames }

   575→// Client-side fetch helper contract (frontend should use this)
   576→export async function api<T>(path: string, init?: RequestInit): Promise<T> {
   577→  const res = await fetch(path, {
   578→    headers: { "Content-Type": "application/json" },
   579→    ...init,
   580→  });
   581→  const json = (await res.json()) as ApiResult<T>;
   582→  if (!json.ok) throw new Error(json.error || "Request failed");
   583→  return json.data;
   584→}

   643→/** Socket event `stream:frame` (agent → hub → dashboard). */
   644→export interface StreamFrame {

   646→  streamId: string;
   647→  kind: StreamKind;

   657→// Socket contract additions (hub on :3003, browser via io("/?XTransformPort=3003", { path: "/" })):
   658→//
   659→// Dashboard → hub (hub forwards to the agent as agent:request):
   660→//   agent request "stream:start" { ...StreamDescriptor }        ack { started: true, meta?: { width, height, fps } } | ack error
   661→//   agent request "stream:stop"  { streamId }                   → { frames: number }
   662→// Dashboard → hub socket (frame routing is push-only, no ack):
   663→//   "stream:stop"   { agentId, streamId }   (belt-and-braces stop for orphaned streams)
   664→// Hub → dashboard:
   665→//   "stream:frame"  StreamFrame
   666→//   "stream:ended"  { agentId, streamId, frames, reason? }
   667→//
   668→// Agents MUST stop all streams of a device when the agent disconnects; the
   669→// hub emits "stream:ended" for every active streamId of that agent.

   671→// ─────────────────── Scheduled encrypted recordings ────────────────
   672→// Transparency requirement: the agent announces the schedule to the device
   673→// user (notification) and shows a persistent on-screen indicator while
   674→// recording. These behaviors are non-optional and cannot be disabled by
   675→// config. Recording continues locally while the hub is unreachable.
   676→
   677→export type RecordingState =
   678→  | "armed" // accepted by agent, waiting to start (or counting down)
   679→  | "recording"
   680→  | "completed" // file sealed + uploaded to hub
   681→  | "failed"
   682→  | "canceled";
   683→
   684→export interface RecordingArtifact {
   685→  id: string; // DB id (cuid)
   686→  recordingId: string; // agent-side id (dashboard-generated, shared)
   687→  deviceId: string; // Device.id
   688→  label: string | null;
   689→  durationSec: number;
   690→  /** selected sources, e.g. { monitors:["monitor:0"], cameras:["camera:0"], mics:["mic:default"] } */
   691→  sources: { monitors?: string[]; cameras?: string[]; mics?: string[] } | null;
   692→  state: RecordingState;
   693→  fileName: string | null; // <recordingId>.enc
   694→  sizeBytes: number | null;
   695→  sha256: string | null; // integrity hash of the uploaded .enc file
   696→  encryption: string; // "AES-256-GCM"
   697→  kdf: string; // "PBKDF2-HMAC-SHA256"
   698→  kdfIterations: number; // 200000
   699→  /** first 12 hex chars of sha256(passphrase) — lets the user verify they typed the right key without storing it */
   700→  keyFingerprint: string | null;
   701→  frames: number | null;
   702→  error: string | null;
   703→  startedAt: string | null; // ISO
   704→  endedAt: string | null; // ISO
   705→  createdAt: string; // ISO
   706→  updatedAt: string; // ISO
   707→}
   708→
   709→/**
   710→ * Encrypted artifact format (<recordingId>.enc) — shared by all agents,
   711→ * the hub and tools/decrypt_recording.py:
   712→ *
   713→ *   Line 1 (ASCII, \n-terminated): JSON header
   714→ *     { magic: "SENTRA-REC-1", cipher: "AES-256-GCM", kdf: "PBKDF2-HMAC-SHA256",
   715→ *       iterations: 200000, salt: <b64 16B>, agentId, label?, durationSec,
   716→ *       sources: [...], startedAt: epochMs }
   717→ *   Then a sequence of framed records:
   718→ *     [u32 BE length][12B nonce][ciphertext][16B tag]
   719→ *     plaintext = JSON { t: epochMs, kind: "screen"|"camera"|"mic"|"meta",
   720→ *                        sourceId?, mime?: "image/jpeg", dataB64?, level? }
   721→ *   key = PBKDF2-HMAC-SHA256(passphrase, salt, iterations, 32 bytes)
   722→ *   AAD = first 16 bytes of sha256(recordingId) (binds chunks to the recording)
   723→ */

   726→//
   727→// GET  /api/recordings?deviceId=&limit=          → RecordingArtifact[]
   728→// POST /api/recordings                           body: { deviceId, durationSec, monitors?, cameras?, mics?, passphrase, label? }
   729→//                                                → RecordingArtifact (state "armed"); dispatches record:schedule
   730→// GET  /api/recordings/[id]                      → RecordingArtifact
   731→// DELETE /api/recordings/[id]                    → { ok } (cancels armed/recording via record:cancel, then deletes row + file)
   732→// GET  /api/recordings/[id]/download             → binary stream (application/octet-stream, filename <recordingId>.enc)
   733→//
   734→// Internal (hub → api, x-hub-token):
   735→// POST /api/internal/recording                   body: RecordingArtifact-shaped upsert { recordingId, ... } → { ok }
   736→//
   737→// Socket → dashboard:
   738→//   "recording:update" { recording: RecordingArtifact }
   739→//
   740→// Agent request types (hub → agent):
   741→//   "record:schedule" { recordingId, durationSec, monitors?, cameras?, mics?, passphrase, label?, noticeText? }
   742→//                     ack { accepted: true, state } | ack error (agent MUST notify the device user on accept)
   743→//   "record:cancel"   { recordingId }             ack { ok }

   745→//   "record:status"   { recordingId, state, elapsedSec?, sizeBytes?, frames?, error?, startedAt?, endedAt? }
   746→//   "record:chunk"    { recordingId, seq, dataB64 }   ack { ok, bytes } (256KB max payload per chunk, ordered by seq)
