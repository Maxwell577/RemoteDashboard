     1→#!/usr/bin/env node
     2→// ─────────────────────────────────────────────────────────────────────────────
     3→// SentraRMM reference agent v0.2.0
     4→//
     5→// A TRANSPARENT remote-management client. It speaks the exact same socket.io
     6→// protocol as the simulated fleet bundled with the SentraRMM hub:
     7→//
     8→//   agent:register { info }                    → registers the device
     9→//   agent:heartbeat { agentId, metrics, loc }  → telemetry every N seconds
    10→//   agent:request { requestId, type, payload } → hub-initiated operations
    11→//   agent:reply    { requestId, ok, data }     → operation result
    12→//   agent:event    { name, payload }           → screen:frame live stream
    13→//
    14→// Supported request types (real implementations, no simulation):
    15→//   exec                 — run a shell command (PowerShell on Windows)
    16→//   ps:list / ps:action  — running processes + kill/suspend/resume
    17→//   fs:list / fs:download / fs:upload / fs:delete / fs:rename / fs:mkdir
    18→//                        — file operations, scoped to --root (default: home)
    19→//   media:screenshot     — real screen capture on Windows (PowerShell + GDI+)
    20→//   screen:live:start|stop — periodic screenshot frames over agent:event
    21→//
    22→// HONEST LIMITATIONS (the agent reports an explicit error, never fakes data):
    23→//   media:record, net:start/stop and location:fix need native capture stack
    24→//   (WASAPI / pktcap / OS geolocation) that this reference agent does not ship.
    25→//
    26→// This agent is designed to be SEEN: it prints a visible banner, keeps a
    27→// console window, and briefly flashes a PowerShell window when remote
    28→// commands run (disable with --hidden-exec). It registers itself in the
    29→// operator's device list and the device user can close it at any time.
    30→// ─────────────────────────────────────────────────────────────────────────────

    46→// ── CLI + config file ────────────────────────────────────────────────────
    47→function parseArgs(argv) {
    48→  const out = {};
    49→  for (let i = 2; i < argv.length; i++) {
    50→    const a = argv[i];
    51→    const next = () => argv[++i];
    52→    if (a === "--hub") out.hub = next();
    53→    else if (a === "--name" || a === "--friendly-name") out.friendlyName = next();
    54→    else if (a === "--agent-id") out.agentId = next();
    55→    else if (a === "--tags") out.tags = next();
    56→    else if (a === "--interval") out.heartbeatSec = Number(next());
    57→    else if (a === "--root") out.fsRoot = next();
    58→    else if (a === "--type") out.deviceType = next();
    59→    else if (a === "--config") out.configFile = next();
    60→    else if (a === "--hidden-exec") out.hiddenExec = true;
    61→    else if (a === "--print-info") out.printInfo = true;
    62→    else if (a === "--help" || a === "-h") out.help = true;
    63→  }
    64→  return out;
    65→}

    68→  let fileCfg = {};
    69→  const cfgPath =
    70→    args.configFile ??
    71→    path.join(path.dirname(new URL(import.meta.url).pathname.replace(/^\/([A-Za-z]:)/, "$1")), "sentra-agent.config.json");

    73→    fileCfg = JSON.parse(fs.readFileSync(cfgPath, "utf8"));
    74→  } catch {
    75→    /* no config file — CLI args/env are enough */
    76→  }
    77→  const merged = { ...fileCfg, ...args };

    79→    hub: merged.hub ?? process.env.SENTRA_HUB ?? "http://127.0.0.1:3003",
    80→    friendlyName: merged.friendlyName ?? process.env.SENTRA_NAME ?? null,
    81→    agentId: merged.agentId ?? process.env.SENTRA_AGENT_ID ?? null,

    93→  };

   124→  const ifaces = os.networkInterfaces();

   126→    for (const i of list ?? []) {

   129→  }
   130→  return { ip: "127.0.0.1", mac: "00:00:00:00:00:00" };

   134→  try {

   139→    /* not linux or no os-release */

   144→/** Windows only — friendly edition string via WMI, cached forever, 4s timeout. */
   145→let wmiCaptionCache;
   146→function windowsEdition() {
   147→  if (wmiCaptionCache !== undefined) return Promise.resolve(wmiCaptionCache);
   148→  return new Promise((resolve) => {
   149→    const child = spawn(
   150→      "powershell.exe",
   151→      ["-NoProfile", "-NonInteractive", "-Command", "(Get-CimInstance Win32_OperatingSystem).Caption"],
   152→      { windowsHide: true, timeout: 4000 },
   153→    );
   154→    let out = "";
   155→    child.stdout.on("data", (c) => (out += c));
   156→    const done = (val) => {
   157→      wmiCaptionCache = val;
   158→      resolve(val);
   159→    };
   160→    child.on("error", () => done(null));
   161→    child.on("close", (code) => done(code === 0 ? out.trim() || null : null));
   162→  });
   163→}

   169→    cfg.agentId ??

   196→// ── metrics sampler ──────────────────────────────────────────────────────
   197→function cpuTimes() {
   198→  return os.cpus().reduce((acc, c) => {
   199→    acc.idle += c.times.idle;
   200→    acc.total += c.times.user + c.times.nice + c.times.sys + c.times.idle + c.times.irq;
   201→    return acc;
   202→  }, { idle: 0, total: 0 });
   203→}

   205→class NetCounter {
   206→  constructor() {
   207→    this.lastBytes = null;
   208→    this.lastTs = 0;
   209→    this.winCache = { rows: null, at: 0 };
   210→  }
   211→  async linuxBytes() {
   212→    const txt = await fsp.readFile("/proc/net/dev", "utf8").catch(() => null);
   213→    if (!txt) return null;
   214→    let rx = 0;
   215→    let tx = 0;
   216→    for (const line of txt.split("\n").slice(2)) {
   217→      const [name, rest] = line.split(":");
   218→      if (!rest || name.trim() === "lo") continue;
   219→      const cols = rest.trim().split(/\s+/);
   220→      rx += Number(cols[0]) || 0;
   221→      tx += Number(cols[8]) || 0;
   222→    }
   223→    return { rx, tx };
   224→  }
   225→  async windowsBytes() {
   226→    if (Date.now() - this.winCache.at < 10_000 && this.winCache.rows) return this.winCache.rows;
   227→    const rows = await new Promise((resolve) => {
   228→      const child = spawn(
   229→        "powershell.exe",
   230→        [
   231→          "-NoProfile",
   232→          "-NonInteractive",
   233→          "-Command",
   234→          "Get-NetAdapterStatistics | ForEach-Object { \"{0} {1} {2}\" -f $_.Name, $_.ReceivedBytes, $_.SentBytes }",
   235→        ],
   236→        { windowsHide: true, timeout: 5000 },
   237→      );
   238→      let out = "";
   239→      child.stdout.on("data", (c) => (out += c));
   240→      child.on("error", () => resolve(null));
   241→      child.on("close", () => {
   242→        let rx = 0;
   243→        let tx = 0;
   244→        for (const line of out.split("\n")) {
   245→          const cols = line.trim().split(/\s+/);
   246→          if (cols.length >= 3) {
   247→            rx += Number(cols[1]) || 0;
   248→            tx += Number(cols[2]) || 0;
   249→          }
   250→        }
   251→        resolve({ rx, tx });
   252→      });
   253→    });
   254→    this.winCache = { rows, at: Date.now() };
   255→    return rows;
   256→  }
   257→  /** KB/s since the previous call (per-heartbeat delta). */
   258→  async rates() {
   259→    const now = Date.now();
   260→    const cur =
   261→      process.platform === "linux" ? await this.linuxBytes() : IS_WIN ? await this.windowsBytes() : null;
   262→    if (!cur) return { up: 0, down: 0 };
   263→    if (!this.lastBytes || now <= this.lastTs) {
   264→      this.lastBytes = cur;
   265→      this.lastTs = now;
   266→      return { up: 0, down: 0 };
   267→    }
   268→    const dt = (now - this.lastTs) / 1000;
   269→    const down = Math.max(0, (cur.rx - this.lastBytes.rx) / 1024 / dt);
   270→    const up = Math.max(0, (cur.tx - this.lastBytes.tx) / 1024 / dt);
   271→    this.lastBytes = cur;
   272→    this.lastTs = now;
   273→    return { up: Math.round(up), down: Math.round(down) };
   274→  }
   275→}

   277→class Metrics {
   278→  constructor() {
   279→    this.net = new NetCounter();
   280→    this.cpuPrev = cpuTimes();
   281→    this.cpuPrevTs = Date.now();
   282→  }
   283→  cpuPercent() {
   284→    const cur = cpuTimes();
   285→    const now = Date.now();
   286→    const dt = Math.max(1, now - this.cpuPrevTs);
   287→    const idle = cur.idle - this.cpuPrev.idle;
   288→    const total = cur.total - this.cpuPrev.total;
   289→    this.cpuPrev = cur;
   290→    this.cpuPrevTs = now;
   291→    if (total <= 0) return 0;
   292→    return round1(Math.min(100, Math.max(0, (1 - idle / total) * 100)));
   293→  }
   294→  memoryPercent() {
   295→    const total = os.totalmem();
   296→    if (!total) return 0;
   297→    return round1(Math.min(100, ((total - os.freemem()) / total) * 100));
   298→  }
   299→  async diskPercent() {
   300→    try {
   301→      const s = await fsp.statfs(os.homedir());
   302→      const total = Number(s.blocks);
   303→      const free = Number(s.bavail);
   304→      if (!total) return 0;
   305→      return round1(Math.min(100, ((total - free) / total) * 100));
   306→    } catch {
   307→      return 0;
   308→    }
   309→  }
   310→  async activeUsers() {
   311→    if (!IS_WIN) {
   312→      try {
   313→        const out = await sh("who", 2000);
   314→        return out.trim() === "" ? 0 : out.trim().split("\n").length;
   315→      } catch {
   316→        return 1;
   317→      }
   318→    }
   319→    return 1; // an interactive Windows session is running this agent
   320→  }
   321→  async snapshot(agentId, uptimeSec) {
   322→    const [disk, users, net] = await Promise.all([this.diskPercent(), this.activeUsers(), this.net.rates()]);
   323→    return {
   324→      deviceId: agentId,
   325→      cpuPercent: this.cpuPercent(),
   326→      memoryPercent: this.memoryPercent(),
   327→      diskPercent: disk,
   328→      uptimeSec,
   329→      activeUsers: users,
   330→      netUpKBps: net.up,
   331→      netDownKBps: net.down,
   332→      capturedAt: new Date().toISOString(),
   333→    };
   334→  }
   335→}

   339→// ── shell helpers ────────────────────────────────────────────────────────
   340→function sh(command, timeoutMs = EXEC_TIMEOUT_MS) {
   341→  return new Promise((resolve, reject) => {
   342→    const child = spawn(command, { shell: true, windowsHide: true, timeout: timeoutMs });
   343→    let out = "";
   344→    child.stdout.on("data", (c) => {
   345→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   346→    });
   347→    child.on("error", reject);
   348→    child.on("close", (code) => (code === 0 ? resolve(out) : reject(new Error(`exit ${code}: ${out.trim().slice(0, 200)}`))));
   349→  });
   350→}

   352→/** Run a remote command exactly like the terminal view expects. */
   353→function execCommand(command, hiddenExec) {
   354→  return new Promise((resolve) => {
   355→    const started = Date.now();
   356→    const child = IS_WIN
   357→      ? spawn("powershell.exe", ["-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-Command", command], {
   358→          // Transparency: the PowerShell window is visible on the device by
   359→          // default so users SEE remote activity. --hidden-exec disables it.
   360→          windowsHide: hiddenExec === true,
   361→        })
   362→      : spawn("/bin/bash", ["-lc", command]);
   363→
   364→    let out = "";
   365→    let killed = false;
   366→    const timer = setTimeout(() => {
   367→      killed = true;
   368→      try {
   369→        if (IS_WIN) spawn("taskkill", ["/PID", String(child.pid), "/T", "/F"], { windowsHide: true });
   370→        else child.kill("SIGKILL");
   371→      } catch {
   372→        /* already gone */
   373→      }
   374→    }, EXEC_TIMEOUT_MS);
   375→
   376→    child.stdout.on("data", (c) => {
   377→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   378→    });
   379→    child.stderr.on("data", (c) => {
   380→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   381→    });
   382→    child.on("error", (err) => {
   383→      clearTimeout(timer);
   384→      resolve({ output: out + `\n[spawn error] ${err.message}`, exitCode: 1, durationMs: Date.now() - started });
   385→    });
   386→    child.on("close", (code) => {
   387→      clearTimeout(timer);
   388→      const text = out
   389→        .toString("utf8")
   390→        .replace(/\r\n/g, "\n")
   391→        .slice(0, MAX_OUTPUT_BYTES);
   392→      resolve({
   393→        output: killed ? text + `\n[timeout after ${EXEC_TIMEOUT_MS / 1000}s — process terminated]` : text,
   394→        exitCode: killed ? 124 : (code ?? 0),
   395→        durationMs: Date.now() - started,
   396→      });
   397→    });
   398→  });
   399→}

   401→// ── process listing ──────────────────────────────────────────────────────
   402→async function listProcessesWindows() {
   403→  const out = await sh(
   404→    "powershell.exe -NoProfile -NonInteractive -Command \"Get-Process | Select-Object Id,ProcessName,CPU,WorkingSet64,Path | ConvertTo-Json -Compress -Depth 2\"",
   405→    8000,
   406→  );
   407→  let arr;
   408→  try {
   409→    const parsed = JSON.parse(out);
   410→    arr = Array.isArray(parsed) ? parsed : [parsed];
   411→  } catch {
   412→    throw new Error("Could not parse Get-Process output");
   413→  }
   414→  return arr.slice(0, 200).map((p) => ({
   415→    pid: Number(p.Id) || 0,
   416→    name: String(p.ProcessName ?? "unknown"),
   417→    cpuPercent: 0, // Get-Process reports cumulative CPU seconds, not a rate
   418→    memoryMB: Math.round((Number(p.WorkingSet64) || 0) / (1024 * 1024)),
   419→    status: "running",
   420→    user: "",
   421→    path: String(p.Path ?? ""),
   422→  }));
   423→}

   425→async function listProcessesPosix() {
   426→  const out = await sh("ps -eo pid,pcpu,rss,stat,user,comm --no-headers", 5000);
   427→  return out
   428→    .trim()
   429→    .split("\n")
   430→    .filter(Boolean)
   431→    .slice(0, 200)
   432→    .map((line) => {
   433→      const [pid, pcpu, rss, stat, user, comm] = line.trim().split(/\s+/);
   434→      const status = stat?.startsWith("T") ? "stopped" : stat?.startsWith("S") || stat?.startsWith("R") ? "running" : "running";
   435→      return {
   436→        pid: Number(pid) || 0,
   437→        name: path.basename(comm ?? "unknown"),
   438→        cpuPercent: Math.min(100, Number(pcpu) || 0),
   439→        memoryMB: Math.round((Number(rss) || 0) / 1024),
   440→        status,
   441→        user: user ?? "",
   442→        path: comm ?? "",
   443→      };
   444→    });
   445→}

   447→async function processAction(pid, action) {
   448→  if (action === "kill") {
   449→    if (IS_WIN) await sh(`taskkill /PID ${pid} /T /F`, 5000);
   450→    else process.kill(pid, "SIGKILL");
   451→    return { ok: true, action: "kill", pid };
   452→  }
   453→  if (action === "suspend") {
   454→    if (IS_WIN) throw new Error("suspend is not supported on Windows by the reference agent (needs pssuspend)");
   455→    process.kill(pid, "SIGSTOP");
   456→    return { ok: true, action: "suspend", pid };
   457→  }
   458→  if (action === "resume") {
   459→    if (IS_WIN) throw new Error("resume is not supported on Windows by the reference agent (needs pssuspend)");
   460→    process.kill(pid, "SIGCONT");
   461→    return { ok: true, action: "resume", pid };
   462→  }
   463→  throw new Error(`Unknown process action '${action}'`);
   464→}

   466→// ── file operations (scoped to root) ─────────────────────────────────────
   467→class FileScope {
   468→  constructor(root) {
   469→    this.root = path.resolve(root);
   470→    fs.mkdirSync(this.root, { recursive: true });
   471→  }
   472→  /** Resolve a requested path inside the root; throws when it escapes. */
   473→  resolveIn(p) {
   474→    const abs = path.resolve(this.root, String(p ?? "").replace(/^([A-Za-z]:)?[\\/]+/, "/").slice(1) || ".");
   475→    const rel = path.relative(this.root, abs);
   476→    if (rel.startsWith("..") || path.isAbsolute(rel)) throw new Error("Path escapes the agent file scope");
   477→    return abs;
   478→  }
   479→  async list(dirPath) {
   480→    const abs = this.resolveIn(dirPath || ".");
   481→    const dirents = await fsp.readdir(abs, { withFileTypes: true });
   482→    const entries = await Promise.all(
   483→      dirents.slice(0, 500).map(async (d) => {
   484→        const full = path.join(abs, d.name);
   485→        let size = 0;
   486→        let mtime = new Date().toISOString();
   487→        try {
   488→          const st = await fsp.stat(full);
   489→          size = Number(st.size);
   490→          mtime = st.mtime.toISOString();
   491→        } catch {
   492→          /* stat raced a delete — report defaults */
   493→        }
   494→        return {
   495→          name: d.name,
   496→          path: path.relative(this.root, full) || ".",
   497→          type: d.isDirectory() ? "directory" : "file",
   498→          sizeBytes: size,
   499→          modifiedAt: mtime,
   500→          hidden: d.name.startsWith("."),
   501→          ext: d.isDirectory() ? undefined : path.extname(d.name).replace(".", "") || undefined,
   502→        };
   503→      }),
   504→    );
   505→    entries.sort((a, b) => (a.type === b.type ? a.name.localeCompare(b.name) : a.type === "directory" ? -1 : 1));
   506→    const rel = path.relative(this.root, abs);
   507→    return {
   508→      path: rel || ".",
   509→      parent: rel ? path.dirname(rel) === "." ? "." : path.dirname(rel) : null,
   510→      drives: IS_WIN ? await windowsDrives() : ["/"],
   511→      entries,
   512→    };
   513→  }
   514→  async download(p) {
   515→    const abs = this.resolveIn(p);
   516→    const st = await fsp.stat(abs);
   517→    if (!st.isFile()) throw new Error("Not a file");
   518→    if (st.size > MAX_DOWNLOAD_BYTES) throw new Error(`File too large for inline transfer (>${Math.round(MAX_DOWNLOAD_BYTES / 1024 / 1024)}MB)`);
   519→    const buf = await fsp.readFile(abs);
   520→    return {
   521→      path: p,
   522→      name: path.basename(abs),
   523→      sizeBytes: st.size,
   524→      contentBase64: buf.toString("base64"),
   525→      encryptedNote: "transferred over the hub's socket; stored encrypted at rest by the console",
   526→    };
   527→  }
   528→  async upload(dirPath, name, contentBase64) {
   529→    const absDir = this.resolveIn(dirPath || ".");
   530→    const target = path.join(absDir, path.basename(String(name ?? "upload.bin")));
   531→    await fsp.writeFile(target, Buffer.from(String(contentBase64 ?? ""), "base64"));
   532→    return { ok: true, path: path.relative(this.root, target), name: path.basename(target) };
   533→  }
   534→  async remove(p) {
   535→    const abs = this.resolveIn(p);
   536→    const st = await fsp.stat(abs);
   537→    if (abs === this.root) throw new Error("Refusing to delete the file-scope root");
   538→    await fsp.rm(abs, { recursive: true });
   539→    return { ok: true, removed: path.basename(abs) };
   540→  }
   541→  async rename(p, newName) {
   542→    const abs = this.resolveIn(p);
   543→    const target = path.join(path.dirname(abs), path.basename(String(newName ?? "")));
   544→    if (!target.startsWith(this.root)) throw new Error("Path escapes the agent file scope");
   545→    await fsp.rename(abs, target);
   546→    return { ok: true, path: path.relative(this.root, target) };
   547→  }
   548→  async mkdir(p) {
   549→    const abs = this.resolveIn(p);
   550→    await fsp.mkdir(abs, { recursive: true });
   551→    return { ok: true, path: path.relative(this.root, abs) || "." };
   552→  }
   553→}

   555→let driveCache = { at: 0, drives: ["C:\\"] };
   556→async function windowsDrives() {
   557→  if (Date.now() - driveCache.at < 60_000) return driveCache.drives;
   558→  const found = [];
   559→  for (let i = 65; i <= 90; i++) {
   560→    const d = `${String.fromCharCode(i)}:\\`;
   561→    try {
   562→      await fsp.stat(d);
   563→      found.push(d);
   564→    } catch {
   565→      /* drive absent */
   566→    }
   567→  }
   568→  driveCache = { at: Date.now(), drives: found.length ? found : ["C:\\"] };
   569→  return driveCache.drives;
   570→}

   572→// ── screen capture (real on Windows; graceful error elsewhere) ───────────
   573→let shotSeq = 0;
   574→async function captureScreen() {
   575→  if (IS_WIN) {
   576→    const ps = `
   577→Add-Type -AssemblyName System.Windows.Forms,System.Drawing
   578→$b = [System.Windows.Forms.SystemInformation]::VirtualScreen
   579→$bmp = New-Object System.Drawing.Bitmap $b.Width, $b.Height
   580→$g = [System.Drawing.Graphics]::FromImage($bmp)
   581→$g.CopyFromScreen($b.X, $b.Y, 0, 0, $bmp.Size)
   582→$ms = New-Object System.IO.MemoryStream
   583→$bmp.Save($ms, [System.Drawing.Imaging.ImageFormat]::Png)
   584→[Convert]::ToBase64String($ms.ToArray())`;
   585→    const b64 = (await sh(`powershell.exe -NoProfile -NonInteractive -Command "${ps.replace(/\n/g, "; ").replace(/"/g, '\\"')}"`, 12_000)).trim();
   586→    if (!b64) throw new Error("Screen capture returned no data");
   587→    return { dataUri: `data:image/png;base64,${b64}`, sizeBytes: Math.round((b64.length * 3) / 4) };
   588→  }
   589→  // Non-Windows: try ImageMagick's `import` when an X display exists.
   590→  if (process.env.DISPLAY) {
   591→    try {
   592→      const b64 = (await sh("import -window root png:- | base64 -w0", 8000)).trim();
   593→      if (b64) return { dataUri: `data:image/png;base64,${b64}`, sizeBytes: Math.round((b64.length * 3) / 4) };
   594→    } catch {
   595→      /* fall through to honest error */
   596→    }
   597→  }
   598→  throw new Error("Screen capture is implemented for Windows (GDI+); on this platform no capture backend was found");
   599→}

   601→// ── main ─────────────────────────────────────────────────────────────────
   602→async function main() {
   603→  const cfg = loadConfig(parseArgs(process.argv));
   604→  if (cfg.help) {
   605→    console.log(HELP);
   606→    return;
   607→  }
   608→  const info = await buildAgentInfo(cfg);
   609→  if (cfg.printInfo) {
   610→    console.log(JSON.stringify(info, null, 2));
   611→    return;
   612→  }
   613→
   614→  const metrics = new Metrics();
   615→  const files = new FileScope(cfg.fsRoot);
   616→  const bootTime = Date.now() - os.uptime() * 1000;
   617→
   618→  console.log("┌──────────────────────────────────────────────────────────┐");
   619→  console.log("│  SentraRMM agent — THIS AGENT IS VISIBLE BY DESIGN       │");
   620→  console.log("│  It registers on the operator console and the device     │");
   621→  console.log("│  user can stop it at any time with Ctrl+C.               │");
   622→  console.log("└──────────────────────────────────────────────────────────┘");
   623→  console.log(`[AGENT] v${AGENT_VERSION} → hub ${cfg.hub}`);
   624→  console.log(`[AGENT] device ${info.hostname} (${info.osName} ${info.osVersion}, ${info.cpuCores} cores, ${info.totalMemoryMB}MB)`);
   625→  console.log(`[AGENT] agentId ${info.agentId}  tags=[${info.tags.join(", ")}]  fileScope=${files.root}`);
   626→
   627→  const socket = io(cfg.hub, {
   628→    path: "/",
   629→    reconnectionDelayMax: 10_000,
   630→    transports: ["websocket", "polling"],
   631→  });
   632→
   633→  socket.on("connect", () => {
   634→    console.log(`[AGENT] connected to hub (socket ${socket.id})`);
   635→    socket.emit("agent:register", { info }, (ack) => {
   636→      if (ack?.ok) console.log("[AGENT] registered — device should now appear in the console");
   637→      else console.error("[AGENT] registration rejected:", ack?.error ?? "unknown error");
   638→    });
   639→  });
   640→  socket.on("disconnect", (reason) => console.log(`[AGENT] disconnected (${reason}) — reconnecting`));
   641→  socket.on("connect_error", (err) => console.error(`[AGENT] hub connection error: ${err.message}`));
   642→
   643→  // heartbeat loop
   644→  const beat = async () => {
   645→    if (!socket.connected) return;
   646→    const snap = await metrics.snapshot(info.agentId, Math.floor((Date.now() - bootTime) / 1000));
   647→    socket.emit("agent:heartbeat", { agentId: info.agentId, metrics: snap, location: null });
   648→    console.log(
   649→      `[AGENT] heartbeat cpu=${snap.cpuPercent}% mem=${snap.memoryPercent}% disk=${snap.diskPercent}% net=↑${snap.netUpKBps}KBps ↓${snap.netDownKBps}KBps`,
   650→    );
   651→  };
   652→  const beatTimer = setInterval(() => void beat(), cfg.heartbeatSec * 1000);
   653→  void beat();
   654→
   655→  // request handlers
   656→  const handlers = {
   657→    exec: (p) => {
   658→      const cmd = String(p.command ?? "");
   659→      console.log(`[AGENT] exec: ${cmd}`);
   660→      return execCommand(cmd, cfg.hiddenExec);
   661→    },
   662→    "ps:list": async () => ({
   663→      processes: IS_WIN ? await listProcessesWindows() : await listProcessesPosix(),
   664→      metrics: await metrics.snapshot(info.agentId, Math.floor((Date.now() - bootTime) / 1000)),
   665→    }),
   666→    "ps:action": (p) => processAction(Number(p.pid), String(p.action)),
   667→    "fs:list": (p) => files.list(String(p.path ?? ".")),
   668→    "fs:download": (p) => files.download(String(p.path)),
   669→    "fs:upload": (p) => files.upload(String(p.path), String(p.name), String(p.contentBase64)),
   670→    "fs:delete": (p) => files.remove(String(p.path)),
   671→    "fs:rename": (p) => files.rename(String(p.path), String(p.newName)),
   672→    "fs:mkdir": (p) => files.mkdir(String(p.path)),
   673→    "media:screenshot": async () => {
   674→      const shot = await captureScreen();
   675→      console.log(`[AGENT] screenshot captured (${Math.round(shot.sizeBytes / 1024)}KB)`);
   676→      return shot;
   677→    },
   678→    "location:fix": () => {
   679→      throw new Error("location:fix is not implemented in the reference agent (needs an OS/edge geolocation source)");
   680→    },
   681→    "media:record": () => {
   682→      throw new Error("media:record needs a native capture stack (WASAPI/MediaFoundation) — not shipped in the reference agent");
   683→    },
   684→    "net:start": () => {
   685→      throw new Error("net:start needs a packet-capture backend (pktcap/pcap) — not shipped in the reference agent");
   686→    },
   687→    "net:stop": () => {
   688→      throw new Error("net:stop needs a packet-capture backend (pktcap/pcap) — not shipped in the reference agent");
   689→    },
   690→  };
   691→
   692→  let live = null; // { timer, seq }
   693→  socket.on("agent:request", (msg) => {
   694→    if (!msg || typeof msg.requestId !== "string" || typeof msg.type !== "string") return;
   695→    const payload = msg.payload ?? {};
   696→    Promise.resolve()
   697→      .then(() => {
   698→        if (msg.type === "screen:live:start") {
   699→          if (live) throw new Error("A live screen stream is already running");
   700→          const fps = Math.min(3, Math.max(0.5, Number(payload.fps) > 0 ? Number(payload.fps) : 1));
   701→          live = { seq: 0, timer: setInterval(() => void pushFrame(fps), Math.round(1000 / fps)) };
   702→          console.log(`[AGENT] live screen started (${fps} fps)`);
   703→          return { started: true, fps };
   704→        }
   705→        if (msg.type === "screen:live:stop") {
   706→          if (!live) return { frames: 0 };
   707→          clearInterval(live.timer);
   708→          console.log(`[AGENT] live screen stopped (${live.seq} frames)`);
   709→          const frames = live.seq;
   710→          live = null;
   711→          return { frames };
   712→        }
   713→        const h = handlers[msg.type];
   714→        if (!h) throw new Error(`Unknown request type '${msg.type}'`);
   715→        return h(payload);
   716→      })
   717→      .then((data) => socket.emit("agent:reply", { requestId: msg.requestId, ok: true, data }))
   718→      .catch((err) => {
   719→        socket.emit("agent:reply", { requestId: msg.requestId, ok: false, error: err instanceof Error ? err.message : String(err) });
   720→        console.error(`[AGENT] ${msg.type} failed: ${err instanceof Error ? err.message : String(err)}`);
   721→      });
   722→  });
   723→
   724→  const pushFrame = async (fps) => {
   725→    if (!live || !socket.connected) return;
   726→    try {
   727→      shotSeq += 1;
   728→      live.seq += 1;
   729→      const shot = await captureScreen();
   730→      socket.emit("agent:event", {
   731→        name: "screen:frame",
   732→        payload: { agentId: info.agentId, dataUri: shot.dataUri, ts: Date.now(), seq: live.seq, width: 0, height: 0 },
   733→      });
   734→    } catch (err) {
   735→      console.error(`[AGENT] live frame failed: ${err instanceof Error ? err.message : err}`);
   736→      clearInterval(live.timer);
   737→      live = null;
   738→    }
   739→  };
   740→
   741→  const shutdown = (signal) => {
   742→    console.log(`[AGENT] ${signal} received — disconnecting from hub`);
   743→    clearInterval(beatTimer);
   744→    if (live?.timer) clearInterval(live.timer);
   745→    socket.disconnect();
   746→    process.exit(0);
   747→  };
   748→  process.on("SIGINT", () => shutdown("SIGINT"));
   749→  process.on("SIGTERM", () => shutdown("SIGTERM"));
   750→}
   751→
   752→main().catch((err) => {
   753→  console.error("[AGENT] fatal:", err);
   754→  process.exit(1);
   755→});
