   146→   146→function fmtElapsed(ms: number): string {
   147→   147→  const s = Math.max(0, Math.floor(ms / 1000))
   148→   148→  const h = Math.floor(s / 3600)
   149→   149→  const m = Math.floor((s % 3600) / 60)
   150→   150→  const sec = s % 60
   151→   151→  const pad = (n: number) => String(n).padStart(2, '0')
   152→   152→  return h > 0 ? `${h}:${pad(m)}:${pad(sec)}` : `${m}:${pad(sec)}`
   153→   153→}

   155→   155→function fmtDurationLabel(durationSec: number | null): string {

   158→   158→  if (durationSec < 3600) return `${Math.floor(durationSec / 60)} min`
   159→   159→  return `${Math.floor(durationSec / 3600)} h`

   219→   219→/**
   220→   220→ * Animated input-level meter driven by `stream:frame` level (0..1, ~8 Hz).
   221→   221→ * The fill eases via CSS transition; a peak-hold marker decays slowly so
   222→   222→ * transient spikes stay readable.

   224→   224→function MicLevelMeter({ level }: { level: number }) {
   225→   225→  const peakRef = useRef(0)

   228→   228→  useEffect(() => {

   247→   247→        />

   252→   252→        aria-valuemax={100}

   259→   259→          style={{ width: `${Math.max(2, pct)}%` }}
   260→   260→        />
   261→   261→        {/* peak-hold marker */}
   262→   262→        <span
   263→   263→          aria-hidden
   264→   264→          className="absolute top-0 h-full w-0.5 rounded-full bg-foreground/60 transition-[left] duration-150 ease-out"

   269→   269→        {pct}% input level · live
   270→   270→      </p>
   271→   271→    </div>
   272→   272→  )
   273→   273→}
