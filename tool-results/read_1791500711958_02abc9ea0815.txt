    65→function fmtUptime(sec: number): string {
    66→  const d = Math.floor(sec / 86400)
    67→  const h = Math.floor((sec % 86400) / 3600)
    68→  const min = Math.floor((sec % 3600) / 60)
    69→  if (d > 0) return `${d}d ${h}h`
    70→  if (h > 0) return `${h}h ${min}m`
    71→  return `${min}m`
    72→}

   121→/** SVG donut showing a 0-100 health score with a tinted arc. */
   122→function HealthRing({ score, size = 64 }: { score: number; size?: number }) {
   123→  const stroke = 6
   124→  const r = (size - stroke) / 2
   125→  const c = 2 * Math.PI * r
   126→  const pct = Math.min(100, Math.max(0, score))
   127→  const color =
   128→    score >= 80
   129→      ? 'oklch(0.72 0.14 168)' // emerald
   130→      : score >= 55
   131→        ? 'oklch(0.8 0.14 85)' // amber
   132→        : 'oklch(0.7 0.19 22)' // rose
   133→  return (
   134→    <div
   135→      className="relative shrink-0"
   136→      style={{ width: size, height: size }}
   137→      role="img"
   138→      aria-label={`Health score ${score} of 100`}
   139→    >
   140→      <svg width={size} height={size} className="-rotate-90">
   141→        <circle
   142→          cx={size / 2}
   143→          cy={size / 2}
   144→          r={r}
   145→          fill="none"
   146→          stroke="currentColor"
   147→          className="text-border/60"
   148→          strokeWidth={stroke}
   149→        />
   150→        <circle
   151→          cx={size / 2}
   152→          cy={size / 2}
   153→          r={r}
   154→          fill="none"
   155→          stroke={color}
   156→          strokeWidth={stroke}
   157→          strokeLinecap="round"
   158→          strokeDasharray={`${(pct / 100) * c} ${c}`}
   159→          style={{ transition: 'stroke-dasharray 600ms cubic-bezier(0.22,1,0.36,1)' }}
   160→        />
   161→      </svg>
   162→      <span
   163→        className="absolute inset-0 flex items-center justify-center font-mono text-sm font-semibold"
   164→        style={{ color }}
   165→      >
   166→        {score}
   167→      </span>
   168→    </div>
   169→  )
   170→}

   172→/** Tiny inline CPU% sparkline from the live telemetry buffer. */
   173→function CpuSparkline({ history, className }: { history: number[]; className?: string }) {
   174→  const w = 220
   175→  const h = 48
   176→  if (history.length < 2) {
   177→    return (
   178→      <div className={cn('flex items-center justify-center text-[11px] text-muted-foreground', className)} style={{ height: h }}>
   179→        collecting samples…
   180→      </div>
   181→    )
   182→  }
   183→  const pts = history
   184→    .map((v, i) => {
   185→      const x = (i / (history.length - 1)) * w
   186→      const y = h - (Math.min(100, Math.max(0, v)) / 100) * (h - 6) - 3
   187→      return `${x.toFixed(1)},${y.toFixed(1)}`
   188→    })
   189→    .join(' ')
   190→  const last = history[history.length - 1]
   191→  return (
   192→    <svg viewBox={`0 0 ${w} ${h}`} className={cn('w-full', className)} style={{ height: h }} aria-hidden>
   193→      <polyline
   194→        points={pts}
   195→        fill="none"
   196→        stroke="currentColor"
   197→        className="text-primary"
   198→        strokeWidth="2"
   199→        strokeLinejoin="round"
   200→        strokeLinecap="round"
   201→      />
   202→      <circle
   203→        cx={w}
   204→        cy={h - (Math.min(100, Math.max(0, last)) / 100) * (h - 6) - 3}
   205→        r="3"
   206→        className="fill-primary"
   207→      />
   208→    </svg>
   209→  )
   210→}

   212→function healthTone(score: number): string {
   213→  if (score >= 80) return 'border-emerald-500/30 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   214→  if (score >= 55) return 'border-amber-500/30 bg-amber-500/10 text-amber-600 dark:text-amber-400'
   215→  return 'border-rose-500/30 bg-rose-500/10 text-rose-600 dark:text-rose-400'
   216→}
