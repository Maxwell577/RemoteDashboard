     1→     1→#!/usr/bin/env node
     2→     2→// ─────────────────────────────────────────────────────────────────────────────
     3→     3→// SentraRMM reference agent v0.2.0
     4→     4→//
     5→     5→// A TRANSPARENT remote-management client. It speaks the exact same socket.io
     6→     6→// protocol as the simulated fleet bundled with the SentraRMM hub:
     7→     7→//
     8→     8→//   agent:register { info }                    → registers the device
     9→     9→//   agent:heartbeat { agentId, metrics, loc }  → telemetry every N seconds
    10→    10→//   agent:request { requestId, type, payload } → hub-initiated operations
    11→    11→//   agent:reply    { requestId, ok, data }     → operation result
    12→    12→//   agent:event    { name, payload }           → screen:frame live stream
    13→    13→//
    14→    14→// Supported request types (real implementations, no simulation):
    15→    15→//   exec                 — run a shell command (PowerShell on Windows)
    16→    16→//   ps:list / ps:action  — running processes + kill/suspend/resume
    17→    17→//   fs:list / fs:download / fs:upload / fs:delete / fs:rename / fs:mkdir
    18→    18→//                        — file operations, scoped to --root (default: home)
    19→    19→//   media:screenshot     — real screen capture on Windows (PowerShell + GDI+)
    20→    20→//   screen:live:start|stop — periodic screenshot frames over agent:event
    21→    21→//
    22→    22→// HONEST LIMITATIONS (the agent reports an explicit error, never fakes data):
    23→    23→//   media:record, net:start/stop and location:fix need native capture stack
    24→    24→//   (WASAPI / pktcap / OS geolocation) that this reference agent does not ship.
    25→    25→//
    26→    26→// This agent is designed to be SEEN: it prints a visible banner, keeps a
    27→    27→// console window, and briefly flashes a PowerShell window when remote
    28→    28→// commands run (disable with --hidden-exec). It registers itself in the
    29→    29→// operator's device list and the device user can close it at any time.
    30→    30→// ─────────────────────────────────────────────────────────────────────────────

    47→    47→function parseArgs(argv) {
    48→    48→  const out = {};
    49→    49→  for (let i = 2; i < argv.length; i++) {
    50→    50→    const a = argv[i];
    51→    51→    const next = () => argv[++i];
    52→    52→    if (a === "--hub") out.hub = next();
    53→    53→    else if (a === "--name" || a === "--friendly-name") out.friendlyName = next();
    54→    54→    else if (a === "--agent-id") out.agentId = next();
    55→    55→    else if (a === "--tags") out.tags = next();
    56→    56→    else if (a === "--interval") out.heartbeatSec = Number(next());
    57→    57→    else if (a === "--root") out.fsRoot = next();
    58→    58→    else if (a === "--type") out.deviceType = next();
    59→    59→    else if (a === "--config") out.configFile = next();
    60→    60→    else if (a === "--hidden-exec") out.hiddenExec = true;
    61→    61→    else if (a === "--print-info") out.printInfo = true;
    62→    62→    else if (a === "--help" || a === "-h") out.help = true;
    63→    63→  }
    64→    64→  return out;
    65→    65→}

   196→   196→// ── metrics sampler ──────────────────────────────────────────────────────
   197→   197→function cpuTimes() {
   198→   198→  return os.cpus().reduce((acc, c) => {
   199→   199→    acc.idle += c.times.idle;
   200→   200→    acc.total += c.times.user + c.times.nice + c.times.sys + c.times.idle + c.times.irq;
   201→   201→    return acc;
   202→   202→  }, { idle: 0, total: 0 });
   203→   203→}

   205→   205→class NetCounter {
   206→   206→  constructor() {
   207→   207→    this.lastBytes = null;
   208→   208→    this.lastTs = 0;
   209→   209→    this.winCache = { rows: null, at: 0 };
   210→   210→  }
   211→   211→  async linuxBytes() {
   212→   212→    const txt = await fsp.readFile("/proc/net/dev", "utf8").catch(() => null);
   213→   213→    if (!txt) return null;
   214→   214→    let rx = 0;
   215→   215→    let tx = 0;
   216→   216→    for (const line of txt.split("\n").slice(2)) {
   217→   217→      const [name, rest] = line.split(":");
   218→   218→      if (!rest || name.trim() === "lo") continue;
   219→   219→      const cols = rest.trim().split(/\s+/);
   220→   220→      rx += Number(cols[0]) || 0;
   221→   221→      tx += Number(cols[8]) || 0;
   222→   222→    }
   223→   223→    return { rx, tx };
   224→   224→  }
   225→   225→  async windowsBytes() {
   226→   226→    if (Date.now() - this.winCache.at < 10_000 && this.winCache.rows) return this.winCache.rows;
   227→   227→    const rows = await new Promise((resolve) => {
   228→   228→      const child = spawn(
   229→   229→        "powershell.exe",
   230→   230→        [
   231→   231→          "-NoProfile",
   232→   232→          "-NonInteractive",
   233→   233→          "-Command",
   234→   234→          "Get-NetAdapterStatistics | ForEach-Object { \"{0} {1} {2}\" -f $_.Name, $_.ReceivedBytes, $_.SentBytes }",
   235→   235→        ],
   236→   236→        { windowsHide: true, timeout: 5000 },
   237→   237→      );
   238→   238→      let out = "";
   239→   239→      child.stdout.on("data", (c) => (out += c));
   240→   240→      child.on("error", () => resolve(null));
   241→   241→      child.on("close", () => {
   242→   242→        let rx = 0;
   243→   243→        let tx = 0;
   244→   244→        for (const line of out.split("\n")) {
   245→   245→          const cols = line.trim().split(/\s+/);
   246→   246→          if (cols.length >= 3) {
   247→   247→            rx += Number(cols[1]) || 0;
   248→   248→            tx += Number(cols[2]) || 0;
   249→   249→          }
   250→   250→        }
   251→   251→        resolve({ rx, tx });
   252→   252→      });
   253→   253→    });
   254→   254→    this.winCache = { rows, at: Date.now() };
   255→   255→    return rows;
   256→   256→  }
   257→   257→  /** KB/s since the previous call (per-heartbeat delta). */
   258→   258→  async rates() {
   259→   259→    const now = Date.now();
   260→   260→    const cur =
   261→   261→      process.platform === "linux" ? await this.linuxBytes() : IS_WIN ? await this.windowsBytes() : null;
   262→   262→    if (!cur) return { up: 0, down: 0 };
   263→   263→    if (!this.lastBytes || now <= this.lastTs) {
   264→   264→      this.lastBytes = cur;
   265→   265→      this.lastTs = now;
   266→   266→      return { up: 0, down: 0 };
   267→   267→    }
   268→   268→    const dt = (now - this.lastTs) / 1000;
   269→   269→    const down = Math.max(0, (cur.rx - this.lastBytes.rx) / 1024 / dt);
   270→   270→    const up = Math.max(0, (cur.tx - this.lastBytes.tx) / 1024 / dt);
   271→   271→    this.lastBytes = cur;
   272→   272→    this.lastTs = now;
   273→   273→    return { up: Math.round(up), down: Math.round(down) };
   274→   274→  }
   275→   275→}

   277→   277→class Metrics {
   278→   278→  constructor() {
   279→   279→    this.net = new NetCounter();
   280→   280→    this.cpuPrev = cpuTimes();
   281→   281→    this.cpuPrevTs = Date.now();
   282→   282→  }
   283→   283→  cpuPercent() {
   284→   284→    const cur = cpuTimes();
   285→   285→    const now = Date.now();
   286→   286→    const dt = Math.max(1, now - this.cpuPrevTs);
   287→   287→    const idle = cur.idle - this.cpuPrev.idle;
   288→   288→    const total = cur.total - this.cpuPrev.total;
   289→   289→    this.cpuPrev = cur;
   290→   290→    this.cpuPrevTs = now;
   291→   291→    if (total <= 0) return 0;
   292→   292→    return round1(Math.min(100, Math.max(0, (1 - idle / total) * 100)));
   293→   293→  }
   294→   294→  memoryPercent() {
   295→   295→    const total = os.totalmem();
   296→   296→    if (!total) return 0;
   297→   297→    return round1(Math.min(100, ((total - os.freemem()) / total) * 100));
   298→   298→  }
   299→   299→  async diskPercent() {
   300→   300→    try {
   301→   301→      const s = await fsp.statfs(os.homedir());
   302→   302→      const total = Number(s.blocks);
   303→   303→      const free = Number(s.bavail);
   304→   304→      if (!total) return 0;
   305→   305→      return round1(Math.min(100, ((total - free) / total) * 100));
   306→   306→    } catch {
   307→   307→      return 0;
   308→   308→    }
   309→   309→  }
   310→   310→  async activeUsers() {
   311→   311→    if (!IS_WIN) {
   312→   312→      try {
   313→   313→        const out = await sh("who", 2000);
   314→   314→        return out.trim() === "" ? 0 : out.trim().split("\n").length;
   315→   315→      } catch {
   316→   316→        return 1;
   317→   317→      }
   318→   318→    }
   319→   319→    return 1; // an interactive Windows session is running this agent
   320→   320→  }
   321→   321→  async snapshot(agentId, uptimeSec) {
   322→   322→    const [disk, users, net] = await Promise.all([this.diskPercent(), this.activeUsers(), this.net.rates()]);
   323→   323→    return {

   326→   326→      memoryPercent: this.memoryPercent(),
   327→   327→      diskPercent: disk,
   328→   328→      uptimeSec,
   329→   329→      activeUsers: users,
   330→   330→      netUpKBps: net.up,
   331→   331→      netDownKBps: net.down,
   332→   332→      capturedAt: new Date().toISOString(),
   333→   333→    };
   334→   334→  }
   335→   335→}

   340→   340→function sh(command, timeoutMs = EXEC_TIMEOUT_MS) {
   341→   341→  return new Promise((resolve, reject) => {
   342→   342→    const child = spawn(command, { shell: true, windowsHide: true, timeout: timeoutMs });
   343→   343→    let out = "";
   344→   344→    child.stdout.on("data", (c) => {
   345→   345→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   346→   346→    });
   347→   347→    child.on("error", reject);
   348→   348→    child.on("close", (code) => (code === 0 ? resolve(out) : reject(new Error(`exit ${code}: ${out.trim().slice(0, 200)}`))));
   349→   349→  });
   350→   350→}

   352→   352→/** Run a remote command exactly like the terminal view expects. */
   353→   353→function execCommand(command, hiddenExec) {
   354→   354→  return new Promise((resolve) => {
   355→   355→    const started = Date.now();
   356→   356→    const child = IS_WIN
   357→   357→      ? spawn("powershell.exe", ["-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-Command", command], {
   358→   358→          // Transparency: the PowerShell window is visible on the device by
   359→   359→          // default so users SEE remote activity. --hidden-exec disables it.
   360→   360→          windowsHide: hiddenExec === true,
   361→   361→        })
   362→   362→      : spawn("/bin/bash", ["-lc", command]);
   363→   363→
   364→   364→    let out = "";
   365→   365→    let killed = false;
   366→   366→    const timer = setTimeout(() => {
   367→   367→      killed = true;
   368→   368→      try {
   369→   369→        if (IS_WIN) spawn("taskkill", ["/PID", String(child.pid), "/T", "/F"], { windowsHide: true });
   370→   370→        else child.kill("SIGKILL");
   371→   371→      } catch {
   372→   372→        /* already gone */
   373→   373→      }
   374→   374→    }, EXEC_TIMEOUT_MS);
   375→   375→
   376→   376→    child.stdout.on("data", (c) => {
   377→   377→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   378→   378→    });
   379→   379→    child.stderr.on("data", (c) => {
   380→   380→      if (out.length < MAX_OUTPUT_BYTES) out += c;
   381→   381→    });
   382→   382→    child.on("error", (err) => {
   383→   383→      clearTimeout(timer);
   384→   384→      resolve({ output: out + `\n[spawn error] ${err.message}`, exitCode: 1, durationMs: Date.now() - started });
   385→   385→    });
   386→   386→    child.on("close", (code) => {
   387→   387→      clearTimeout(timer);
   388→   388→      const text = out
   389→   389→        .toString("utf8")
   390→   390→        .replace(/\r\n/g, "\n")
   391→   391→        .slice(0, MAX_OUTPUT_BYTES);
   392→   392→      resolve({
   393→   393→        output: killed ? text + `\n[timeout after ${EXEC_TIMEOUT_MS / 1000}s — process terminated]` : text,
   394→   394→        exitCode: killed ? 124 : (code ?? 0),
   395→   395→        durationMs: Date.now() - started,
   396→   396→      });
   397→   397→    });
   398→   398→  });
   399→   399→}

   402→   402→async function listProcessesWindows() {

   404→   404→    "powershell.exe -NoProfile -NonInteractive -Command \"Get-Process | Select-Object Id,ProcessName,CPU,WorkingSet64,Path | ConvertTo-Json -Compress -Depth 2\"",

   411→   411→  } catch {
   412→   412→    throw new Error("Could not parse Get-Process output");

   415→   415→    pid: Number(p.Id) || 0,
   416→   416→    name: String(p.ProcessName ?? "unknown"),
   417→   417→    cpuPercent: 0, // Get-Process reports cumulative CPU seconds, not a rate
   418→   418→    memoryMB: Math.round((Number(p.WorkingSet64) || 0) / (1024 * 1024)),
   419→   419→    status: "running",
   420→   420→    user: "",
   421→   421→    path: String(p.Path ?? ""),
   422→   422→  }));
   423→   423→}

   425→   425→async function listProcessesPosix() {
   426→   426→  const out = await sh("ps -eo pid,pcpu,rss,stat,user,comm --no-headers", 5000);
   427→   427→  return out
   428→   428→    .trim()
   429→   429→    .split("\n")
   430→   430→    .filter(Boolean)
   431→   431→    .slice(0, 200)
   432→   432→    .map((line) => {
   433→   433→      const [pid, pcpu, rss, stat, user, comm] = line.trim().split(/\s+/);
   434→   434→      const status = stat?.startsWith("T") ? "stopped" : stat?.startsWith("S") || stat?.startsWith("R") ? "running" : "running";
   435→   435→      return {
   436→   436→        pid: Number(pid) || 0,
   437→   437→        name: path.basename(comm ?? "unknown"),
   438→   438→        cpuPercent: Math.min(100, Number(pcpu) || 0),
   439→   439→        memoryMB: Math.round((Number(rss) || 0) / 1024),
   440→   440→        status,
   441→   441→        user: user ?? "",
   442→   442→        path: comm ?? "",
   443→   443→      };
   444→   444→    });
   445→   445→}

   447→   447→async function processAction(pid, action) {
   448→   448→  if (action === "kill") {
   449→   449→    if (IS_WIN) await sh(`taskkill /PID ${pid} /T /F`, 5000);
   450→   450→    else process.kill(pid, "SIGKILL");
   451→   451→    return { ok: true, action: "kill", pid };
   452→   452→  }
   453→   453→  if (action === "suspend") {
   454→   454→    if (IS_WIN) throw new Error("suspend is not supported on Windows by the reference agent (needs pssuspend)");
   455→   455→    process.kill(pid, "SIGSTOP");
   456→   456→    return { ok: true, action: "suspend", pid };
   457→   457→  }
   458→   458→  if (action === "resume") {
   459→   459→    if (IS_WIN) throw new Error("resume is not supported on Windows by the reference agent (needs pssuspend)");
   460→   460→    process.kill(pid, "SIGCONT");
   461→   461→    return { ok: true, action: "resume", pid };
   462→   462→  }
   463→   463→  throw new Error(`Unknown process action '${action}'`);
   464→   464→}

   466→   466→// ── file operations (scoped to root) ─────────────────────────────────────
   467→   467→class FileScope {
   468→   468→  constructor(root) {
   469→   469→    this.root = path.resolve(root);
   470→   470→    fs.mkdirSync(this.root, { recursive: true });
   471→   471→  }
   472→   472→  /** Resolve a requested path inside the root; throws when it escapes. */
   473→   473→  resolveIn(p) {
   474→   474→    const abs = path.resolve(this.root, String(p ?? "").replace(/^([A-Za-z]:)?[\\/]+/, "/").slice(1) || ".");
   475→   475→    const rel = path.relative(this.root, abs);
   476→   476→    if (rel.startsWith("..") || path.isAbsolute(rel)) throw new Error("Path escapes the agent file scope");
   477→   477→    return abs;
   478→   478→  }
   479→   479→  async list(dirPath) {
   480→   480→    const abs = this.resolveIn(dirPath || ".");
   481→   481→    const dirents = await fsp.readdir(abs, { withFileTypes: true });
   482→   482→    const entries = await Promise.all(
   483→   483→      dirents.slice(0, 500).map(async (d) => {
   484→   484→        const full = path.join(abs, d.name);
   485→   485→        let size = 0;
   486→   486→        let mtime = new Date().toISOString();
   487→   487→        try {
   488→   488→          const st = await fsp.stat(full);
   489→   489→          size = Number(st.size);
   490→   490→          mtime = st.mtime.toISOString();
   491→   491→        } catch {
   492→   492→          /* stat raced a delete — report defaults */
   493→   493→        }
   494→   494→        return {
   495→   495→          name: d.name,
   496→   496→          path: path.relative(this.root, full) || ".",
   497→   497→          type: d.isDirectory() ? "directory" : "file",
   498→   498→          sizeBytes: size,
   499→   499→          modifiedAt: mtime,
   500→   500→          hidden: d.name.startsWith("."),
   501→   501→          ext: d.isDirectory() ? undefined : path.extname(d.name).replace(".", "") || undefined,
   502→   502→        };
   503→   503→      }),
   504→   504→    );
   505→   505→    entries.sort((a, b) => (a.type === b.type ? a.name.localeCompare(b.name) : a.type === "directory" ? -1 : 1));
   506→   506→    const rel = path.relative(this.root, abs);
   507→   507→    return {
   508→   508→      path: rel || ".",
   509→   509→      parent: rel ? path.dirname(rel) === "." ? "." : path.dirname(rel) : null,
   510→   510→      drives: IS_WIN ? await windowsDrives() : ["/"],
   511→   511→      entries,
   512→   512→    };
   513→   513→  }
   514→   514→  async download(p) {
   515→   515→    const abs = this.resolveIn(p);
   516→   516→    const st = await fsp.stat(abs);
   517→   517→    if (!st.isFile()) throw new Error("Not a file");
   518→   518→    if (st.size > MAX_DOWNLOAD_BYTES) throw new Error(`File too large for inline transfer (>${Math.round(MAX_DOWNLOAD_BYTES / 1024 / 1024)}MB)`);
   519→   519→    const buf = await fsp.readFile(abs);
   520→   520→    return {
   521→   521→      path: p,
   522→   522→      name: path.basename(abs),
   523→   523→      sizeBytes: st.size,
   524→   524→      contentBase64: buf.toString("base64"),
   525→   525→      encryptedNote: "transferred over the hub's socket; stored encrypted at rest by the console",
   526→   526→    };
   527→   527→  }
   528→   528→  async upload(dirPath, name, contentBase64) {
   529→   529→    const absDir = this.resolveIn(dirPath || ".");
   530→   530→    const target = path.join(absDir, path.basename(String(name ?? "upload.bin")));
   531→   531→    await fsp.writeFile(target, Buffer.from(String(contentBase64 ?? ""), "base64"));
   532→   532→    return { ok: true, path: path.relative(this.root, target), name: path.basename(target) };
   533→   533→  }
   534→   534→  async remove(p) {
   535→   535→    const abs = this.resolveIn(p);
   536→   536→    const st = await fsp.stat(abs);
   537→   537→    if (abs === this.root) throw new Error("Refusing to delete the file-scope root");
   538→   538→    await fsp.rm(abs, { recursive: true });
   539→   539→    return { ok: true, removed: path.basename(abs) };
   540→   540→  }
   541→   541→  async rename(p, newName) {
   542→   542→    const abs = this.resolveIn(p);
   543→   543→    const target = path.join(path.dirname(abs), path.basename(String(newName ?? "")));
   544→   544→    if (!target.startsWith(this.root)) throw new Error("Path escapes the agent file scope");
   545→   545→    await fsp.rename(abs, target);
   546→   546→    return { ok: true, path: path.relative(this.root, target) };
   547→   547→  }
   548→   548→  async mkdir(p) {
   549→   549→    const abs = this.resolveIn(p);
   550→   550→    await fsp.mkdir(abs, { recursive: true });
   551→   551→    return { ok: true, path: path.relative(this.root, abs) || "." };
   552→   552→  }
   553→   553→}

   555→   555→let driveCache = { at: 0, drives: ["C:\\"] };
   556→   556→async function windowsDrives() {
   557→   557→  if (Date.now() - driveCache.at < 60_000) return driveCache.drives;
   558→   558→  const found = [];
   559→   559→  for (let i = 65; i <= 90; i++) {
   560→   560→    const d = `${String.fromCharCode(i)}:\\`;
   561→   561→    try {
   562→   562→      await fsp.stat(d);
   563→   563→      found.push(d);
   564→   564→    } catch {
   565→   565→      /* drive absent */
   566→   566→    }
   567→   567→  }
   568→   568→  driveCache = { at: Date.now(), drives: found.length ? found : ["C:\\"] };
   569→   569→  return driveCache.drives;
   570→   570→}

   572→   572→// ── screen capture (real on Windows; graceful error elsewhere) ───────────

   574→   574→async function captureScreen() {
   575→   575→  if (IS_WIN) {
   576→   576→    const ps = `
   577→   577→Add-Type -AssemblyName System.Windows.Forms,System.Drawing
   578→   578→$b = [System.Windows.Forms.SystemInformation]::VirtualScreen
   579→   579→$bmp = New-Object System.Drawing.Bitmap $b.Width, $b.Height
   580→   580→$g = [System.Drawing.Graphics]::FromImage($bmp)
   581→   581→$g.CopyFromScreen($b.X, $b.Y, 0, 0, $bmp.Size)
   582→   582→$ms = New-Object System.IO.MemoryStream
   583→   583→$bmp.Save($ms, [System.Drawing.Imaging.ImageFormat]::Png)
   584→   584→[Convert]::ToBase64String($ms.ToArray())`;
   585→   585→    const b64 = (await sh(`powershell.exe -NoProfile -NonInteractive -Command "${ps.replace(/\n/g, "; ").replace(/"/g, '\\"')}"`, 12_000)).trim();
   586→   586→    if (!b64) throw new Error("Screen capture returned no data");
   587→   587→    return { dataUri: `data:image/png;base64,${b64}`, sizeBytes: Math.round((b64.length * 3) / 4) };
   588→   588→  }
   589→   589→  // Non-Windows: try ImageMagick's `import` when an X display exists.
   590→   590→  if (process.env.DISPLAY) {
   591→   591→    try {
   592→   592→      const b64 = (await sh("import -window root png:- | base64 -w0", 8000)).trim();
   593→   593→      if (b64) return { dataUri: `data:image/png;base64,${b64}`, sizeBytes: Math.round((b64.length * 3) / 4) };
   594→   594→    } catch {
   595→   595→      /* fall through to honest error */
   596→   596→    }
   597→   597→  }
   598→   598→  throw new Error("Screen capture is implemented for Windows (GDI+); on this platform no capture backend was found");
   599→   599→}

   602→   602→async function main() {
   603→   603→  const cfg = loadConfig(parseArgs(process.argv));

   605→   605→    console.log(HELP);

   681→   681→    "media:record": () => {
   682→   682→      throw new Error("media:record needs a native capture stack (WASAPI/MediaFoundation) — not shipped in the reference agent");

   684→   684→    "net:start": () => {
   685→   685→      throw new Error("net:start needs a packet-capture backend (pktcap/pcap) — not shipped in the reference agent");
   686→   686→    },
   687→   687→    "net:stop": () => {
   688→   688→      throw new Error("net:stop needs a packet-capture backend (pktcap/pcap) — not shipped in the reference agent");
   689→   689→    },
   690→   690→  };
   691→   691→
   692→   692→  let live = null; // { timer, seq }
   693→   693→  socket.on("agent:request", (msg) => {
   694→   694→    if (!msg || typeof msg.requestId !== "string" || typeof msg.type !== "string") return;
   695→   695→    const payload = msg.payload ?? {};
   696→   696→    Promise.resolve()
   697→   697→      .then(() => {
   698→   698→        if (msg.type === "screen:live:start") {
   699→   699→          if (live) throw new Error("A live screen stream is already running");
   700→   700→          const fps = Math.min(3, Math.max(0.5, Number(payload.fps) > 0 ? Number(payload.fps) : 1));
   701→   701→          live = { seq: 0, timer: setInterval(() => void pushFrame(fps), Math.round(1000 / fps)) };
   702→   702→          console.log(`[AGENT] live screen started (${fps} fps)`);
   703→   703→          return { started: true, fps };
   704→   704→        }
   705→   705→        if (msg.type === "screen:live:stop") {
   706→   706→          if (!live) return { frames: 0 };
   707→   707→          clearInterval(live.timer);
   708→   708→          console.log(`[AGENT] live screen stopped (${live.seq} frames)`);
   709→   709→          const frames = live.seq;
   710→   710→          live = null;
   711→   711→          return { frames };
   712→   712→        }
   713→   713→        const h = handlers[msg.type];
   714→   714→        if (!h) throw new Error(`Unknown request type '${msg.type}'`);
   715→   715→        return h(payload);
   716→   716→      })

   720→   720→        console.error(`[AGENT] ${msg.type} failed: ${err instanceof Error ? err.message : String(err)}`);
   721→   721→      });
   722→   722→  });

   724→   724→  const pushFrame = async (fps) => {
   725→   725→    if (!live || !socket.connected) return;

   731→   731→        name: "screen:frame",
   732→   732→        payload: { agentId: info.agentId, dataUri: shot.dataUri, ts: Date.now(), seq: live.seq, width: 0, height: 0 },
   733→   733→      });
   734→   734→    } catch (err) {
   735→   735→      console.error(`[AGENT] live frame failed: ${err instanceof Error ? err.message : err}`);

   737→   737→      live = null;
   738→   738→    }
   739→   739→  };

   744→   744→    if (live?.timer) clearInterval(live.timer);
   745→   745→    socket.disconnect();
   746→   746→    process.exit(0);
   747→   747→  };
   748→   748→  process.on("SIGINT", () => shutdown("SIGINT"));
   749→   749→  process.on("SIGTERM", () => shutdown("SIGTERM"));
   750→   750→}
