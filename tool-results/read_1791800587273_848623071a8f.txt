   291→   291→// ───────────────────────── Device health score ──────────────────────
   292→   292→
   293→   293→/**
   294→   294→ * Composite 0-100 health score from live telemetry (frontend-computed).
   295→   295→ * 100 = idle healthy machine; deductions for CPU/MEM/DISK pressure + penalties
   296→   296→ * for stale telemetry or offline status.
   297→   297→ */
   298→   298→export function computeHealthScore(input: {
   299→   299→  online: boolean;
   300→   300→  cpuPercent?: number | null;
   301→   301→  memoryPercent?: number | null;
   302→   302→  diskPercent?: number | null;
   303→   303→  stale?: boolean;
   304→   304→}): number {
   305→   305→  if (!input.online) return 40;
   306→   306→  let score = 100;
   307→   307→  const cpu = input.cpuPercent ?? 0;
   308→   308→  const mem = input.memoryPercent ?? 0;
   309→   309→  const disk = input.diskPercent ?? 0;
   310→   310→  score -= Math.max(0, cpu - 30) * 0.5; // heavy CPU load hurts most
   311→   311→  score -= Math.max(0, mem - 55) * 0.6; // memory pressure is critical
   312→   312→  score -= Math.max(0, disk - 80) * 0.8; // nearly-full disks are a risk
   313→   313→  if (input.stale) score -= 10;
   314→   314→  return Math.max(0, Math.min(100, Math.round(score)));
   315→   315→}

   317→   317→// ─────────────────────────── Alert rules engine ────────────────────
   318→   318→
   319→   319→export type AlertMetric = "cpu" | "memory" | "disk";
   320→   320→export type AlertRuleOp = "gt" | "lt";
   321→   321→
   322→   322→/**
   323→   323→ * A threshold rule evaluated against every device heartbeat. Rules live in
   324→   324→ * the Setting table under key "alertRules" (JSON array). When a device's
   325→   325→ * telemetry crosses the threshold, an Alert row is created — rate-limited by
   326→   326→ * `cooldownMin` per device+rule and suppressed while maintenance is open.
   327→   327→ */
   328→   328→export interface AlertRule {
   329→   329→  id: string;
   330→   330→  name: string; // "High CPU load" — used in the alert title
   331→   331→  metric: AlertMetric;
   332→   332→  op: AlertRuleOp; // gt = above, lt = below
   333→   333→  threshold: number; // percent 0-100
   334→   334→  severity: "info" | "warning" | "critical";
   335→   335→  cooldownMin: number; // min minutes between repeat alerts for the same device
   336→   336→  enabled: boolean;
   337→   337→  /**
   338→   338→   * Optional tag scope — when set, the rule only evaluates against devices
   339→   339→   * whose tags include this value (e.g. "production"). null/undefined = all
   340→   340→   * devices.
   341→   341→   */
   342→   342→  deviceTag?: string | null;
   343→   343→}

   672→   672→// Transparency requirement: the agent announces the schedule to the device

   677→   677→export type RecordingState =
   678→   678→  | "armed" // accepted by agent, waiting to start (or counting down)
   679→   679→  | "recording"

   684→   684→export interface RecordingArtifact {
   685→   685→  id: string; // DB id (cuid)
   686→   686→  recordingId: string; // agent-side id (dashboard-generated, shared)
   687→   687→  deviceId: string; // Device.id

   689→   689→  durationSec: number;
   690→   690→  /** selected sources, e.g. { monitors:["monitor:0"], cameras:["camera:0"], mics:["mic:default"] } */
   691→   691→  sources: { monitors?: string[]; cameras?: string[]; mics?: string[] } | null;
   692→   692→  state: RecordingState;
   693→   693→  fileName: string | null; // <recordingId>.enc
   694→   694→  sizeBytes: number | null;
   695→   695→  sha256: string | null; // integrity hash of the uploaded .enc file
   696→   696→  encryption: string; // "AES-256-GCM"
   697→   697→  kdf: string; // "PBKDF2-HMAC-SHA256"
   698→   698→  kdfIterations: number; // 200000
   699→   699→  /** first 12 hex chars of sha256(passphrase) — lets the user verify they typed the right key without storing it */
   700→   700→  keyFingerprint: string | null;
   701→   701→  frames: number | null;
   702→   702→  error: string | null;
   703→   703→  startedAt: string | null; // ISO
   704→   704→  endedAt: string | null; // ISO
   705→   705→  createdAt: string; // ISO
   706→   706→  updatedAt: string; // ISO
   707→   707→}

   709→   709→/**
   710→   710→ * Encrypted artifact format (<recordingId>.enc) — shared by all agents,
   711→   711→ * the hub and tools/decrypt_recording.py:
   712→   712→ *
   713→   713→ *   Line 1 (ASCII, \n-terminated): JSON header
   714→   714→ *     { magic: "SENTRA-REC-1", cipher: "AES-256-GCM", kdf: "PBKDF2-HMAC-SHA256",
   715→   715→ *       iterations: 200000, salt: <b64 16B>, agentId, label?, durationSec,
   716→   716→ *       sources: [...], startedAt: epochMs }
   717→   717→ *   Then a sequence of framed records:
   718→   718→ *     [u32 BE length][12B nonce][ciphertext][16B tag]
   719→   719→ *     plaintext = JSON { t: epochMs, kind: "screen"|"camera"|"mic"|"meta",
   720→   720→ *                        sourceId?, mime?: "image/jpeg", dataB64?, level? }
   721→   721→ *   key = PBKDF2-HMAC-SHA256(passphrase, salt, iterations, 32 bytes)
   722→   722→ *   AAD = first 16 bytes of sha256(recordingId) (binds chunks to the recording)
   723→   723→ */
