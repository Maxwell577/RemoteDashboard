    59→function fmtUptime(sec: number): string {
    60→  const d = Math.floor(sec / 86400)
    61→  const h = Math.floor((sec % 86400) / 3600)
    62→  const min = Math.floor((sec % 3600) / 60)
    63→  if (d > 0) return `${d}d ${h}h`
    64→  if (h > 0) return `${h}h ${min}m`
    65→  return `${min}m`
    66→}

   115→/** SVG donut showing a 0-100 health score with a tinted arc. */
   116→function HealthRing({ score, size = 64 }: { score: number; size?: number }) {
   117→  const stroke = 6
   118→  const r = (size - stroke) / 2
   119→  const c = 2 * Math.PI * r
   120→  const pct = Math.min(100, Math.max(0, score))
   121→  const color =
   122→    score >= 80
   123→      ? 'oklch(0.72 0.14 168)' // emerald
   124→      : score >= 55
   125→        ? 'oklch(0.8 0.14 85)' // amber
   126→        : 'oklch(0.7 0.19 22)' // rose
   127→  return (
   128→    <div
   129→      className="relative shrink-0"
   130→      style={{ width: size, height: size }}
   131→      role="img"
   132→      aria-label={`Health score ${score} of 100`}
   133→    >
   134→      <svg width={size} height={size} className="-rotate-90">
   135→        <circle
   136→          cx={size / 2}
   137→          cy={size / 2}
   138→          r={r}
   139→          fill="none"
   140→          stroke="currentColor"
   141→          className="text-border/60"
   142→          strokeWidth={stroke}
   143→        />
   144→        <circle
   145→          cx={size / 2}
   146→          cy={size / 2}
   147→          r={r}
   148→          fill="none"
   149→          stroke={color}
   150→          strokeWidth={stroke}
   151→          strokeLinecap="round"
   152→          strokeDasharray={`${(pct / 100) * c} ${c}`}
   153→          style={{ transition: 'stroke-dasharray 600ms cubic-bezier(0.22,1,0.36,1)' }}
   154→        />
   155→      </svg>
   156→      <span
   157→        className="absolute inset-0 flex items-center justify-center font-mono text-sm font-semibold"
   158→        style={{ color }}
   159→      >
   160→        {score}
   161→      </span>
   162→    </div>
   163→  )
   164→}

   166→/** Tiny inline CPU% sparkline from the live telemetry buffer. */
   167→function CpuSparkline({ history, className }: { history: number[]; className?: string }) {
   168→  const w = 220
   169→  const h = 48
   170→  if (history.length < 2) {
   171→    return (
   172→      <div className={cn('flex items-center justify-center text-[11px] text-muted-foreground', className)} style={{ height: h }}>
   173→        collecting samples…
   174→      </div>
   175→    )
   176→  }
   177→  const pts = history
   178→    .map((v, i) => {
   179→      const x = (i / (history.length - 1)) * w
   180→      const y = h - (Math.min(100, Math.max(0, v)) / 100) * (h - 6) - 3
   181→      return `${x.toFixed(1)},${y.toFixed(1)}`
   182→    })
   183→    .join(' ')
   184→  const last = history[history.length - 1]
   185→  return (
   186→    <svg viewBox={`0 0 ${w} ${h}`} className={cn('w-full', className)} style={{ height: h }} aria-hidden>
   187→      <polyline
   188→        points={pts}
   189→        fill="none"
   190→        stroke="currentColor"
   191→        className="text-primary"
   192→        strokeWidth="2"
   193→        strokeLinejoin="round"
   194→        strokeLinecap="round"
   195→      />
   196→      <circle
   197→        cx={w}
   198→        cy={h - (Math.min(100, Math.max(0, last)) / 100) * (h - 6) - 3}
   199→        r="3"
   200→        className="fill-primary"
   201→      />
   202→    </svg>
   203→  )
   204→}

   206→function healthTone(score: number): string {
   207→  if (score >= 80) return 'border-emerald-500/30 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   208→  if (score >= 55) return 'border-amber-500/30 bg-amber-500/10 text-amber-600 dark:text-amber-400'
   209→  return 'border-rose-500/30 bg-rose-500/10 text-rose-600 dark:text-rose-400'
   210→}

   560→          <div className="flex items-baseline justify-between gap-3">
   561→            <dt className="shrink-0 text-muted-foreground">{online ? 'Status' : 'Last seen'}</dt>
   562→            <dd
   563→              className={cn(
   564→                'truncate text-right font-medium',
   565→                online && !m && 'text-muted-foreground',
   566→                online && m && 'text-emerald-600 dark:text-emerald-400',
   567→              )}
   568→            >
   569→              {online
   570→                ? m
   571→                  ? `live · CPU ${Math.round(m.cpuPercent)}% · MEM ${Math.round(m.memoryPercent)}%`
   572→                  : 'waiting for telemetry…'
   573→                : timeAgo(device.lastSeen)}
   574→            </dd>
   575→          </div>
   576→          {online && m ? (
   577→            <div className="flex items-baseline justify-between gap-3">
   578→              <dt className="flex shrink-0 items-center gap-1 text-muted-foreground">
   579→                <Timer className="size-3" aria-hidden />
   580→                Uptime
   581→              </dt>
   582→              <dd
   583→                className="truncate text-right font-medium tabular-nums"
   584→                title={`Last boot ${new Date(Date.now() - m.uptimeSec * 1000).toLocaleString()}`}
   585→              >
   586→                {fmtUptime(m.uptimeSec)}
   587→              </dd>
   588→            </div>
   589→          ) : null}
   590→        </dl>
   591→
   592→        {device.tags.length > 0 ? (
   593→          <div className="mt-3 flex flex-wrap gap-1">
   594→            {device.tags.map((t) => (
   595→              <Badge key={t} variant="secondary" className="text-[10px]">
   596→                {t}
   597→              </Badge>
   598→            ))}
   599→          </div>
   600→        ) : null}
   601→
   602→        <div className="mt-4 flex items-center gap-2 border-t border-border/50 pt-3">
   603→          <Button size="sm" className="flex-1" onClick={() => go('terminal')}>
   604→            <SquareTerminal className="size-3.5" aria-hidden /> Manage
   605→          </Button>
   606→          <Button
   607→            size="sm"
   608→            variant="outline"
   609→            className="flex-1"
   610→            onClick={() => setDetailsOpen(true)}
   611→          >
   612→            <Info className="size-3.5" aria-hidden /> Details
   613→          </Button>
   614→          <Button size="sm" variant="outline" className="flex-1" onClick={() => go('location')}>
   615→            <MapPin className="size-3.5" aria-hidden /> Locate
   616→          </Button>
   617→        </div>
   618→      </Panel>
   619→
   620→      <DeviceDetailsDialog
   621→        device={device}
   622→        open={detailsOpen}
   623→        onOpenChange={setDetailsOpen}
   624→      />
   625→    </>
   626→  )
   627→}

   668→

   684→      if (tagFilter && !d.tags.includes(tagFilter)) return false
   685→      if (!q) return true

   691→        d.ipAddress?.toLowerCase().includes(q) ||
