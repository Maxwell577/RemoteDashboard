     1→"""Scheduled encrypted recordings (v0.3 ``record:schedule`` family).
     2→
     3→Flow (state machine ``armed → recording → completed|failed|canceled``):
     4→
     5→1.  ``schedule`` validates durationSec (≤ maxRecordingMin*60) and free disk
     6→    space (≥ minFreeMB), NOTIFIES THE DEVICE USER (mandatory transparency,
     7→    using ``noticeText`` when provided) and acks ``{accepted: true, state}``.
     8→2.  A daemon job thread captures the selected sources — screen monitors at
     9→    2 fps (mss), cameras at 5 fps (OpenCV), mic levels at 4 Hz — and appends
    10→    plaintext JSON records to memory buffers.
    11→3.  ENCRYPT-THEN-WRITE: records are grouped into ≤64KB plaintext groups,
    12→    each group is AES-256-GCM encrypted and framed as
    13→    ``[u32BE len][12B nonce][ct][16B tag]`` (SENTRA-REC-1). Frames are BOTH
    14→    appended to the local ``<recordingId>.enc`` file (the offline-safe copy)
    15→    AND packed into ``record:chunk`` events (≤256KB payload, ordered seq,
    16→    ack'd; on failure/nack they go to a local spool flushed on reconnect).
    17→4.  On completion the file is sealed, sha256 computed, and ``record:manifest``
    18→    emitted. ``record:cancel`` stops early (state ``canceled``).
    19→
    20→The job runs in a daemon thread: losing the hub NEVER stops a recording.
    21→The passphrase is used for key derivation only and is never logged.
    22→"""
    23→from __future__ import annotations
    24→
    25→import base64
    26→import hashlib
    27→import io
    28→import json

    38→MAGIC = "SENTRA-REC-1"
    39→CIPHER = "AES-256-GCM"
    40→KDF = "PBKDF2-HMAC-SHA256"
    41→KDF_ITERATIONS = 200000
    42→
    43→RECORD_FPS_SCREEN = 2.0
    44→RECORD_FPS_CAMERA = 5.0
    45→MIC_RECORD_HZ = 4
    46→GROUP_BYTES = 64 * 1024          # plaintext group size before encrypting
    47→CHUNK_TARGET = 240 * 1024        # pack frames into chunks up to this (b64)
    48→CHUNK_MAX = 256 * 1024           # hard contract cap per record:chunk payload
    49→MAX_JPEG_BYTES = 160 * 1024      # keeps base64(frame) safely inside a chunk
    50→ACK_TIMEOUT_S = 8.0
    51→MAX_CONCURRENT = 2

    54→def _aad_for(recording_id: str) -> bytes:
    55→    return hashlib.sha256(recording_id.encode("utf-8")).digest()[:16]

    58→def _derive_key(passphrase: str, salt: bytes) -> bytes:
    59→    return hashlib.pbkdf2_hmac(
    60→        "sha256", passphrase.encode("utf-8"), salt, KDF_ITERATIONS, dklen=32,
    61→    )

    64→def _fit_jpeg_b64_bgra(raw: Any, width: int, height: int) -> str:
    65→    """JPEG-encode an mss BGRA frame, shrinking quality/size to stay ≤ cap."""
    66→    import cv2  # type: ignore
    67→    import numpy as np  # type: ignore
    68→
    69→    arr = np.frombuffer(raw, dtype=np.uint8).reshape((height, width, 4))
    70→    img = cv2.cvtColor(arr, cv2.COLOR_BGRA2BGR)
    71→    return _fit_jpeg_img(img)

    73→
    74→def _fit_jpeg_img(img: Any) -> str:
    75→    """Encode a BGR image, lowering quality then scaling until ≤ MAX_JPEG_BYTES."""
    76→    import cv2  # type: ignore
    77→
    78→    for quality in (70, 55, 40):
    79→        ok, buf = cv2.imencode(".jpg", img, [int(cv2.IMWRITE_JPEG_QUALITY), quality])
    80→        if ok and len(buf) <= MAX_JPEG_BYTES:
    81→            return base64.b64encode(buf.tobytes()).decode("ascii")
    82→    h, w = img.shape[:2]
    83→    scale = 0.75
    84→    while scale >= 0.3:
    85→        small = cv2.resize(img, (max(320, int(w * scale)), max(240, int(h * scale))))
    86→        ok, buf = cv2.imencode(".jpg", small, [int(cv2.IMWRITE_JPEG_QUALITY), 45])
    87→        if ok and len(buf) <= MAX_JPEG_BYTES:
    88→            return base64.b64encode(buf.tobytes()).decode("ascii")
    89→        scale -= 0.15
    90→    # last resort — accept whatever we got at the smallest size
    91→    ok, buf = cv2.imencode(".jpg", small, [int(cv2.IMWRITE_JPEG_QUALITY), 35])
    92→    return base64.b64encode((buf.tobytes() if ok else b"")).decode("ascii")

    95→class _RecordSink:
    96→    """Thread-safe record collector feeding the encryptor."""
    97→
    98→    def __init__(self) -> None:
    99→        self._pending: List[Dict[str, Any]] = []
   100→        self._lock = threading.Lock()
   101→
   102→    def add(self, record: Dict[str, Any]) -> None:
   103→        with self._lock:
   104→            self._pending.append(record)
   105→
   106→    def drain(self) -> List[Dict[str, Any]]:
   107→        with self._lock:
   108→            out = self._pending
   109→            self._pending = []
   110→            return out

   113→class _ChunkUploader:
   114→    """Sends record:chunk events with acks; spools failures for later flush."""
   115→
   116→    def __init__(self, recordings_dir: str,
   117→                 is_connected: Callable[[], bool],
   118→                 emit_chunk: Callable[[str, Dict[str, Any]], Tuple[bool, str]],
   119→                 emit_status: Callable[[Dict[str, Any]], None]) -> None:
   120→        self.spool_dir = os.path.join(recordings_dir, "spool")
   121→        try:
   122→            os.makedirs(self.spool_dir, exist_ok=True)
   123→        except OSError:
   124→            pass
   125→        self.is_connected = is_connected
   126→        self.emit_chunk = emit_chunk          # → (ok, error)
   127→        self.emit_status = emit_status
   128→        self._queue: "list[Tuple[str, int, bytes]]" = []
   129→        self._cv = threading.Condition()
   130→        self._stop = threading.Event()
   131→        self._thread = threading.Thread(target=self._loop, name="sentra-chunk-up", daemon=True)
   132→        self._thread.start()
   133→
   134→    # ------------------------------------------------------------------ api
   135→
   136→    def submit(self, recording_id: str, seq: int, payload: bytes) -> None:
   137→        with self._cv:
   138→            self._queue.append((recording_id, seq, payload))
   139→            self._cv.notify()
   140→
   141→    def flush_spool(self) -> int:
   142→        """Push all spooled chunks (ascending seq per recording). Returns sent count."""
   143→        sent = 0
   144→        try:
   145→            recs = sorted(os.listdir(self.spool_dir))
   146→        except OSError:
   147→            return 0
   148→        for rec_dir in recs:
   149→            rec_path = os.path.join(self.spool_dir, rec_dir)
   150→            if not os.path.isdir(rec_path):
   151→                continue
   152→            try:
   153→                chunks = sorted(os.listdir(rec_path))
   154→            except OSError:
   155→                continue
   156→            for name in chunks:
   157→                if self._stop.is_set():
   158→                    return sent
   159→                path = os.path.join(rec_path, name)
   160→                try:
   161→                    with open(path, "rb") as fh:
   162→                        payload = fh.read()
   163→                    seq = int(name.split(".")[0])
   164→                except (OSError, ValueError):
   165→                    continue
   166→                ok, _err = self.emit_chunk(rec_dir, {"recordingId": rec_dir, "seq": seq,
   167→                                                     "dataB64": base64.b64encode(payload).decode("ascii")})
   168→                if ok:
   169→                    try:
   170→                        os.remove(path)
   171→                        sent += 1
   172→                    except OSError:
   173→                        pass
   174→                else:
   175→                    return sent  # still offline — stop early, stay spooled
   176→        return sent
   177→
   178→    def stop(self) -> None:
   179→        self._stop.set()
   180→        with self._cv:
   181→            self._cv.notify()
   182→        self._thread.join(timeout=3.0)
   183→
   184→    # ----------------------------------------------------------------- loop
   185→
   186→    def _spool(self, recording_id: str, seq: int, payload: bytes) -> None:
   187→        rec_path = os.path.join(self.spool_dir, recording_id)
   188→        try:
   189→            os.makedirs(rec_path, exist_ok=True)
   190→            with open(os.path.join(rec_path, f"{seq:08d}.chunk"), "wb") as fh:
   191→                fh.write(payload)
   192→        except OSError as exc:
   193→            log.error("chunk spool write failed (%s seq=%d): %s", recording_id, seq, exc)
   194→
   195→    def _loop(self) -> None:
   196→        while not self._stop.is_set():
   197→            with self._cv:
   198→                if not self._queue:
   199→                    self._cv.wait(timeout=0.5)
   200→                item = self._queue.pop(0) if self._queue else None
   201→            if item is None:
   202→                continue
   203→            recording_id, seq, payload = item
   204→            if not self.is_connected():
   205→                self._spool(recording_id, seq, payload)
   206→                continue
   207→            ok, err = self.emit_chunk(recording_id, {
   208→                "recordingId": recording_id, "seq": seq,
   209→                "dataB64": base64.b64encode(payload).decode("ascii"),
   210→            })
   211→            if ok:
   212→                continue
   213→            log.warning("record:chunk %s seq=%d rejected (%s) — spooled for retry", recording_id, seq, err)
   214→            self._spool(recording_id, seq, payload)

   217→class RecordingJob(threading.Thread):
   218→    """One recording: capture threads → sink → encrypt → file + chunks."""
   219→
   220→    def __init__(self, spec: Dict[str, Any], agent_id: str, recordings_dir: str,
   221→                 uploader: _ChunkUploader, emit_status: Callable[[Dict[str, Any]], None],
   222→                 emit_manifest: Callable[[Dict[str, Any]], Tuple[bool, str]],
   223→                 transparency: Any, stop_event: threading.Event) -> None:
   224→        super().__init__(name=f"sentra-rec-{spec.get('recordingId', '?')[:8]}", daemon=True)
   225→        self.spec = spec
   226→        self.recording_id = str(spec["recordingId"])
   227→        self.duration_sec = int(spec.get("durationSec") or 0)
   228→        self.agent_id = agent_id
   229→        self.recordings_dir = recordings_dir
   230→        self.uploader = uploader
   231→        self.emit_status = emit_status
   232→        self.emit_manifest = emit_manifest
   233→        self.transparency = transparency
   234→        self.global_stop = stop_event
   235→
   236→        self.state = "armed"
   237→        self.error: Optional[str] = None
   238→        self.frames = 0
   239→        self.size_bytes = 0
   240→        self.started_at_ms: Optional[int] = None
   241→        self.ended_at_ms: Optional[int] = None
   242→        self._cancel = threading.Event()
   243→        self.sink = _RecordSink()

   246→        self.salt = os.urandom(16)

   252→    def status_msg(self) -> Dict[str, Any]:
   253→        msg: Dict[str, Any] = {
   254→            "recordingId": self.recording_id,
   255→            "state": self.state,
   256→            "frames": self.frames,

   258→        if self.state == "recording" and self.started_at_ms:

   260→        if self.size_bytes:
   261→            msg["sizeBytes"] = self.size_bytes

   263→            msg["startedAt"] = self.started_at_ms
   264→        if self.ended_at_ms:
   265→            msg["endedAt"] = self.ended_at_ms
   266→        if self.error:
   267→            msg["error"] = self.error

   275→    def _selected(self, kind: str) -> List[str]:

   283→    def _default_monitors(self, caps: Dict[str, Any]) -> List[str]:
   284→        mons = self._selected("monitors")
   285→        if mons:  # explicit selection (possibly empty → none)
   286→            return mons
   287→        for mon in caps.get("monitors", []):
   288→            if mon.get("primary"):
   289→                return [str(mon["id"])]

   292→    def _capture_screen(self, monitor_ids: List[str]) -> None:
   293→        try:
   294→            import mss  # type: ignore
   295→        except Exception as exc:  # noqa: BLE001
   296→            self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   297→                           "event": "screen-unavailable", "error": str(exc)})
   298→            return
   299→        interval = 1.0 / RECORD_FPS_SCREEN
   300→        try:
   301→            with mss.mss() as sct:
   302→                monitors = list(sct.monitors)
   303→                for mon_id in monitor_ids:
   304→                    try:
   305→                        idx = int(str(mon_id).split(":", 1)[1])
   306→                    except (ValueError, IndexError):
   307→                        idx = 0
   308→                    mon = monitors[idx + 1] if idx + 1 < len(monitors) else (monitors[1] if len(monitors) > 1 else monitors[0])
   309→                    next_t = time.monotonic()
   310→                    while not self._cancel.is_set() and not self.global_stop.is_set() and self.state == "recording":
   311→                        shot = sct.grab(mon)
   312→                        data_b64 = _fit_jpeg_b64_bgra(shot.bgra, shot.width, shot.height)
   313→                        self.sink.add({
   314→                            "t": int(time.time() * 1000), "kind": "screen",
   315→                            "sourceId": mon_id, "mime": "image/jpeg", "dataB64": data_b64,
   316→                        })
   317→                        next_t += interval
   318→                        sleep = next_t - time.monotonic()
   319→                        if sleep > 0:
   320→                            time.sleep(sleep)
   321→                        else:
   322→                            next_t = time.monotonic()
   323→        except Exception as exc:  # noqa: BLE001 - degrade to honest meta note
   324→            self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   325→                           "event": "screen-capture-error", "error": str(exc)})

   327→    def _capture_camera(self, camera_ids: List[str]) -> None:
   328→        if not camera_ids:
   329→            return
   330→        try:
   331→            import cv2  # type: ignore
   332→        except Exception as exc:  # noqa: BLE001
   333→            self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   334→                           "event": "camera-unavailable", "error": str(exc)})
   335→            return
   336→        interval = 1.0 / RECORD_FPS_CAMERA
   337→        for cam_id in camera_ids:
   338→            try:
   339→                idx = int(str(cam_id).split(":", 1)[1])
   340→            except (ValueError, IndexError):
   341→                idx = 0
   342→            cap = cv2.VideoCapture(idx, getattr(cv2, "CAP_DSHOW", 0))
   343→            try:
   344→                if not cap.isOpened():
   345→                    self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   346→                                   "event": "camera-open-failed", "sourceId": cam_id})
   347→                    continue
   348→                next_t = time.monotonic()
   349→                while not self._cancel.is_set() and not self.global_stop.is_set() and self.state == "recording":
   350→                    ok, img = cap.read()
   351→                    if not ok:
   352→                        time.sleep(0.1)
   353→                        continue
   354→                    data_b64 = _fit_jpeg_img(img)
   355→                    self.sink.add({
   356→                        "t": int(time.time() * 1000), "kind": "camera",
   357→                        "sourceId": cam_id, "mime": "image/jpeg", "dataB64": data_b64,
   358→                    })
   359→                    next_t += interval
   360→                    sleep = next_t - time.monotonic()
   361→                    if sleep > 0:
   362→                        time.sleep(sleep)
   363→                    else:
   364→                        next_t = time.monotonic()
   365→            except Exception as exc:  # noqa: BLE001
   366→                self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   367→                               "event": "camera-capture-error", "sourceId": cam_id, "error": str(exc)})
   368→            finally:
   369→                try:
   370→                    cap.release()
   371→                except Exception:  # noqa: BLE001
   372→                    pass

   374→    def _capture_mic(self, mic_ids: List[str]) -> None:
   375→        if not mic_ids:
   376→            return
   377→        try:
   378→            import numpy as np  # type: ignore
   379→            import sounddevice as sd  # type: ignore
   380→        except Exception as exc:  # noqa: BLE001
   381→            self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   382→                           "event": "mic-unavailable", "error": str(exc)})
   383→            return
   384→        interval = 1.0 / MIC_RECORD_HZ
   385→        for mic_id in mic_ids:
   386→            try:
   387→                dev_idx: Optional[int] = None
   388→                if str(mic_id).startswith("mic:"):
   389→                    try:
   390→                        dev_idx = int(str(mic_id).split(":", 1)[1])
   391→                    except ValueError:
   392→                        dev_idx = None
   393→                latest = {"rms": 0.0}
   394→
   395→                def cb(indata: Any, frames: int, time_info: Any, status: Any) -> None:
   396→                    try:
   397→                        mono = indata[:, 0].astype("float32")
   398→                        latest["rms"] = min(1.0, max(0.0, float(np.sqrt(np.mean(np.square(mono)))) * 4.0))
   399→                    except Exception:  # noqa: BLE001
   400→                        pass
   401→
   402→                with sd.InputStream(samplerate=16000, channels=1, dtype="float32",
   403→                                    device=dev_idx, blocksize=1024, callback=cb):
   404→                    next_t = time.monotonic()
   405→                    while not self._cancel.is_set() and not self.global_stop.is_set() and self.state == "recording":
   406→                        self.sink.add({"t": int(time.time() * 1000), "kind": "mic",
   407→                                       "sourceId": mic_id, "level": round(latest["rms"], 4)})
   408→                        next_t += interval
   409→                        sleep = next_t - time.monotonic()
   410→                        if sleep > 0:
   411→                            time.sleep(sleep)
   412→                        else:
   413→                            next_t = time.monotonic()
   414→            except Exception as exc:  # noqa: BLE001
   415→                self.sink.add({"t": int(time.time() * 1000), "kind": "meta",
   416→                               "event": "mic-capture-error", "sourceId": mic_id, "error": str(exc)})

   420→    def _write_header(self, fh: io.BufferedWriter, sources: List[str]) -> None:
   421→        header = {
   422→            "magic": MAGIC, "cipher": CIPHER, "kdf": KDF,
   423→            "iterations": KDF_ITERATIONS, "salt": base64.b64encode(self.salt).decode("ascii"),
   424→            "agentId": self.agent_id, "recordingId": self.recording_id,
   425→            "label": self.spec.get("label"),
   426→            "durationSec": self.duration_sec, "sources": sources,
   427→            "startedAt": self.started_at_ms,
   428→        }
   429→        fh.write((json.dumps(header, separators=(",", ":")) + "\n").encode("ascii"))

   431→    def _encrypt_group(self, key: bytes, aad: bytes, plaintext: bytes) -> bytes:
   432→        from cryptography.hazmat.primitives.ciphers.aead import AESGCM
   433→
   434→        nonce = os.urandom(12)
   435→        sealed = AESGCM(key).encrypt(nonce, plaintext, aad)  # ct || tag
   436→        ct, tag = sealed[:-16], sealed[-16:]
   437→        frame = len(nonce) + len(ct) + len(tag)
   438→        return frame.to_bytes(4, "big") + nonce + ct + tag

   440→    # ------------------------------------------------------------------- run
   441→
   442→    def run(self) -> None:
   443→        try:
   444→            self._run_inner()
   445→        except Exception as exc:  # noqa: BLE001 - never let a job kill the process
   446→            self.state = "failed"
   447→            self.error = str(exc)
   448→            log.exception("recording %s failed", self.recording_id)
   449→            try:
   450→                self.emit_status(self.status_msg())
   451→            except Exception:  # noqa: BLE001
   452→                pass
   453→            try:
   454→                self.transparency.recording_stop(self.spec.get("label"), "failed")
   455→            except Exception:  # noqa: BLE001
   456→                pass
   457→
   458→    def _run_inner(self) -> None:
   459→        from .captures import enumerate_captures
   460→
   461→        caps = enumerate_captures()
   462→        monitor_ids = self._default_monitors(caps)
   463→        camera_ids = self._selected("cameras")
   464→        mic_ids = self._selected("mics")
   465→        sources = monitor_ids + camera_ids + mic_ids
   466→
   467→        passphrase = str(self.spec.get("passphrase") or "")
   468→        if not passphrase:
   469→            raise ValueError("record:schedule payload is missing 'passphrase'")
   470→
   471→        # armed: MANDATORY user notification BEFORE any capture starts
   472→        try:
   473→            self.transparency.notifier.recording_armed(
   474→                self.spec.get("label") or "recording", self.duration_sec,
   475→                str(self.spec["noticeText"]) if self.spec.get("noticeText") else None,
   476→            )
   477→            self.transparency.remote_action(
   478→                "record", f"Recording session scheduled ({self.duration_sec}s)"
   479→                + (f" — {self.spec.get('label')}" if self.spec.get("label") else ""),
   480→            )
   481→        except Exception:  # noqa: BLE001 - transparency must never break recording
   482→            log.exception("transparency armed-notify failed")
   483→        self.emit_status(self.status_msg())  # state=armed
   484→        time.sleep(1.0)  # brief armed window so cancel() can land
   485→        if self._cancel.is_set():
   486→            self.state = "canceled"
   487→            self.ended_at_ms = int(time.time() * 1000)
   488→            self.emit_status(self.status_msg())
   489→            try:
   490→                self.transparency.recording_stop(self.spec.get("label"), "canceled")
   491→            except Exception:  # noqa: BLE001
   492→                pass
   493→            return
   494→
   495→        self.state = "recording"
   496→        self.started_at_ms = int(time.time() * 1000)
   497→        # MANDATORY on-screen overlay + red tray dot + toast (non-disableable)
   498→        try:
   499→            self.transparency.recording_start(
   500→                self.spec.get("label") or None,
   501→                str(self.spec["noticeText"]) if self.spec.get("noticeText") else None,
   502→            )
   503→        except Exception:  # noqa: BLE001
   504→            log.exception("transparency recording_start failed")
   505→
   506→        key = _derive_key(passphrase, self.salt)
   507→        del passphrase  # do not keep it around
   508→        aad = _aad_for(self.recording_id)
   509→
   510→        try:
   511→            fh = open(self.file_path, "wb")
   512→        except OSError as exc:
   513→            raise RuntimeError(f"cannot create recording file: {exc}") from exc
   514→
   515→        group: List[bytes] = []
   516→        group_size = 0
   517→        chunk_buf: List[bytes] = []
   518→        chunk_size = 0
   519→
   520→        def flush_group() -> None:
   521→            nonlocal group, group_size, chunk_buf, chunk_size
   522→            if not group:
   523→                return
   524→            plaintext = b"\n".join(group) + b"\n"
   525→            frame = self._encrypt_group(key, aad, plaintext)
   526→            fh.write(frame)
   527→            self.size_bytes += len(frame)
   528→            self.frames += len(group)
   529→            group = []
   530→            group_size = 0
   531→            chunk_buf.append(frame)
   532→            chunk_size += len(frame)
   533→            if chunk_size >= CHUNK_TARGET:
   534→                self._emit_chunk(chunk_buf)
   535→                chunk_buf = []
   536→                chunk_size = 0
   537→
   538→        # start capture threads (daemon — die with the process)
   539→        for target, args in (
   540→            (self._capture_screen, (monitor_ids,)),
   541→            (self._capture_camera, (camera_ids,)),
   542→            (self._capture_mic, (mic_ids,)),
   543→        ):
   544→            t = threading.Thread(target=target, args=args, daemon=True)
   545→            t.start()
   546→            self.capture_threads.append(t)
   547→
   548→        end_at = time.monotonic() + self.duration_sec
   549→        last_progress = time.monotonic()
   550→        try:
   551→            self._write_header(fh, sources)
   552→            self.size_bytes = fh.tell()
   553→            while True:
   554→                time.sleep(0.15)
   555→                records = self.sink.drain()
   556→                for rec in records:
   557→                    line = json.dumps(rec, separators=(",", ":")).encode("utf-8")
   558→                    group.append(line)
   559→                    group_size += len(line)
   560→                    if group_size >= GROUP_BYTES:
   561→                        flush_group()
   562→                now = time.monotonic()
   563→                if now - last_progress >= 15.0:
   564→                    last_progress = now
   565→                    self.emit_status(self.status_msg())
   566→                if self._cancel.is_set() or self.global_stop.is_set() or now >= end_at:
   567→                    break
   568→            flush_group()
   569→            if chunk_buf:
   570→                self._emit_chunk(chunk_buf)
   571→        finally:
   572→            try:
   573→                fh.flush()
   574→                os.fsync(fh.fileno())
   575→            except OSError:
   576→                pass
   577→            fh.close()
   578→
   579→        self.ended_at_ms = int(time.time() * 1000)
   580→        self.state = "canceled" if self._cancel.is_set() else "completed"
   581→        self.emit_status(self.status_msg())
   582→        try:
   583→            self.transparency.recording_stop(self.spec.get("label"), self.state)
   584→        except Exception:  # noqa: BLE001
   585→            log.exception("transparency recording_stop failed")
   586→
   587→        if self.state == "completed":
   588→            sha = self._sha256_file(self.file_path)
   589→            ok, _err = self.emit_manifest({
   590→                "recordingId": self.recording_id,
   591→                "fileName": os.path.basename(self.file_path),
   592→                "sizeBytes": self.size_bytes,
   593→                "sha256": sha,
   594→            })
   595→            log.info("recording %s completed: %s bytes, manifest ack=%s",
   596→                     self.recording_id, self.size_bytes, ok)
   597→
   598→    def _emit_chunk(self, frames: List[bytes]) -> None:
   599→        payload = b"".join(frames)
   600→        self.seq += 1
   601→        if len(payload) > CHUNK_MAX:
   602→            # contract cap violated (single huge frame) — split is impossible in
   603→            # this format, so log loudly; _fit_jpeg_b64_bgra prevents this path.
   604→            log.error("chunk %s seq=%d exceeds %dKB (%d) — dropping", self.recording_id,
   605→                      self.seq, CHUNK_MAX // 1024, len(payload))
   606→            return
   607→        self.uploader.submit(self.recording_id, self.seq, payload)
   608→
   609→    @staticmethod
   610→    def _sha256_file(path: str) -> str:
   611→        h = hashlib.sha256()
   612→        with open(path, "rb") as fh:
   613→            for block in iter(lambda: fh.read(65536), b""):
   614→                h.update(block)
   615→        return h.hexdigest()
   616→

   618→class RecordingManager:
   619→    """Handles record:schedule / record:cancel / record:status for the hub."""
   620→
   621→    def __init__(self, agent_id: str, cfg: Dict[str, Any], recordings_dir: str,
   622→                 transparency: Any, is_connected: Callable[[], bool],
   623→                 emit_chunk: Callable[[str, Dict[str, Any]], Tuple[bool, str]],
   624→                 emit_status: Callable[[Dict[str, Any]], None],
   625→                 emit_manifest: Callable[[Dict[str, Any]], Tuple[bool, str]]) -> None:
   626→        self.agent_id = agent_id
   627→        self.cfg = cfg
   628→        self.recordings_dir = recordings_dir
   629→        self.transparency = transparency
   630→        self.uploader = _ChunkUploader(recordings_dir, is_connected, emit_chunk, emit_status)
   631→        self.jobs: Dict[str, RecordingJob] = {}
   632→        self._lock = threading.Lock()
   633→        self._stop = threading.Event()
   634→        self.emit_status_fn = emit_status
   635→        self.emit_manifest_fn = emit_manifest
   636→
   637→    # ------------------------------------------------------------- handlers
   638→
   639→    def schedule(self, payload: Dict[str, Any]) -> Dict[str, Any]:
   640→        """Validate + accept a record:schedule. Raises ValueError on bad input."""
   641→        recording_id = str(payload.get("recordingId") or "").strip()
   642→        if not recording_id:
   643→            raise ValueError("recordingId is required")
   644→        duration_sec = int(payload.get("durationSec") or 0)
   645→        max_sec = int(self.cfg.get("maxRecordingMin", 60)) * 60
   646→        if duration_sec <= 0:
   647→            raise ValueError("durationSec must be positive")
   648→        if duration_sec > max_sec:
   649→            raise ValueError(
   650→                f"durationSec {duration_sec} exceeds this agent's maxRecordingMin cap "
   651→                f"({max_sec}s) — adjust maxRecordingMin in config.json"
   652→            )
   653→        usage = shutil.disk_usage(self.recordings_dir)
   654→        free_mb = int(usage.free / (1024 * 1024))
   655→        if free_mb < int(self.cfg.get("minFreeMB", 500)):
   656→            raise ValueError(
   657→                f"insufficient disk space for recording ({free_mb}MB free < "
   658→                f"minFreeMB {self.cfg.get('minFreeMB')}MB)"
   659→            )
   660→        with self._lock:
   661→            alive = [j for j in self.jobs.values() if j.is_alive()]
   662→            if len(alive) >= MAX_CONCURRENT:
   663→                raise ValueError(f"agent already has {len(alive)} recordings in progress (cap {MAX_CONCURRENT})")
   664→
   665→        job = RecordingJob(
   666→            dict(payload), self.agent_id, self.recordings_dir, self.uploader,
   667→            self.emit_status_fn, self.emit_manifest_fn,
   668→            self.transparency, self._stop,
   669→        )
   670→        with self._lock:
   671→            self.jobs[recording_id] = job
   672→        job.start()
   673→        return {"accepted": True, "state": job.state}
   674→
   675→    def cancel(self, payload: Dict[str, Any]) -> Dict[str, Any]:
   676→        recording_id = str(payload.get("recordingId") or "")
   677→        with self._lock:
   678→            job = self.jobs.get(recording_id)
   679→        if job is None:
   680→            raise ValueError(f"unknown recordingId '{recording_id}'")
   681→        job.cancel()
   682→        return {"ok": True}
   683→
   684→    def status(self) -> Dict[str, Any]:
   685→        with self._lock:
   686→            jobs = [j for j in self.jobs.values() if j.is_alive() or j.state in ("armed", "recording")]
   687→        return {"recordings": [j.status_msg() for j in jobs]}
   688→
   689→    # ------------------------------------------------------------------ misc
   690→
   691→    def on_reconnect(self) -> None:
   692→        """Flush spooled chunks after the hub link comes back."""
   693→        try:
   694→            sent = self.uploader.flush_spool()
   695→            if sent:
   696→                log.info("flushed %d spooled recording chunk(s) after reconnect", sent)
   697→        except Exception:  # noqa: BLE001
   698→            log.exception("spool flush failed")
   699→
   700→    def stop_all(self) -> None:
   701→        self._stop.set()
   702→        with self._lock:
   703→            jobs = list(self.jobs.values())
   704→        for job in jobs:
   705→            job.cancel()
   706→        for job in jobs:
   707→            job.join(timeout=5.0)
   708→        self.uploader.stop()
