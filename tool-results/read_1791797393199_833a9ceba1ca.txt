    80→    80→  constructor(readonly def: DeviceDef) {
    81→    81→    this.vfs = seedDeviceFs(def);
    82→    82→    this.procTable = new ProcTable(def);
    83→    83→    this.bootTime = Date.now() - def.bootDaysAgo * 86_400_000 - rndInt(0, 3_600_000);
    84→    84→    this.cmdCwd = `C:\\Users\\${def.userName}`;
    85→    85→    this.socket = io(`http://127.0.0.1:${CONFIG.port}`, {
    86→    86→      path: "/",
    87→    87→      reconnectionDelayMax: 5000,
    88→    88→    });
    89→    89→  }
    90→    90→
    91→    91→  start(): void {
    92→    92→    const s = this.socket;
    93→    93→    s.on("connect", () => {
    94→    94→      this.log(`connected to hub (socket ${s.id})`);
    95→    95→      s.emit(
    96→    96→        "agent:register",
    97→    97→        { info: { ...this.def.info, captures: this.def.captures } },
    98→    98→        (ack?: { ok?: boolean }) => {
    99→    99→          if (ack?.ok) this.log("registered (captures: monitors/cameras/mics/shells reported)");
   100→   100→          else this.log("registration rejected by hub");
   101→   101→        },
   102→   102→      );
   103→   103→      this.ensureHeartbeat();
   104→   104→    });
   105→   105→    s.on("disconnect", (reason: string) => {
   106→   106→      this.log(`disconnected from hub (${reason}) — will reconnect`);
   107→   107→      // v0.3 contract: agents stop streaming when they disconnect. Recordings
   108→   108→      // keep running and buffer chunks until the hub is reachable again.
   109→   109→      this.stopAllStreams();
   110→   110→    });
   111→   111→    s.on("connect_error", (err: Error) => {
   112→   112→      this.log(`hub connect error: ${err.message} — retrying`);
   113→   113→    });
   114→   114→    s.on("agent:request", (msg: RequestMsg) => {
   115→   115→      void this.onRequest(msg);
   116→   116→    });
   117→   117→  }
   118→   118→
   119→   119→  stop(): void {
   120→   120→    this.stopped = true;
   121→   121→    if (this.heartbeatTimer) {
   122→   122→      clearInterval(this.heartbeatTimer);
   123→   123→      this.heartbeatTimer = null;
   124→   124→    }
   125→   125→    for (const cap of this.media.values()) {
   126→   126→      if (cap.timer) clearTimeout(cap.timer);
   127→   127→    }
   128→   128→    this.media.clear();
   129→   129→    for (const cap of this.net.values()) {
   130→   130→      if (cap.timer) clearInterval(cap.timer);
   131→   131→    }
   132→   132→    this.net.clear();
   133→   133→    if (this.live?.timer) clearInterval(this.live.timer);
   134→   134→    this.live = null;
   135→   135→    this.stopAllStreams();
   136→   136→    if (this.recorder && !this.recorder.isDone()) this.recorder.cancel();
   137→   137→    this.recorder = null;
   138→   138→    this.socket.disconnect();
   139→   139→  }

   141→   141→  // ── telemetry ─────────────────────────────────────────────────────────
   142→   142→
   143→   143→  private ensureHeartbeat(): void {
   144→   144→    if (this.heartbeatTimer || this.stopped) return;
   145→   145→    this.heartbeatTimer = setInterval(() => this.beat(), CONFIG.heartbeatSec * 1000);
   146→   146→  }
   147→   147→
   148→   148→  private beat(): void {
   149→   149→    if (this.stopped || !this.socket.connected) return;
   150→   150→    const metrics = this.buildMetrics();
   151→   151→    this.socket.emit("agent:heartbeat", {
   152→   152→      agentId: this.def.info.agentId,
   153→   153→      metrics,
   154→   154→      location: this.locationFix(),
   155→   155→    });
   156→   156→    this.log(
   157→   157→      `heartbeat cpu=${metrics.cpuPercent}% mem=${metrics.memoryPercent}% disk=${metrics.diskPercent}% net=↑${metrics.netUpKBps}KBps ↓${metrics.netDownKBps}KBps`,
   158→   158→    );
   159→   159→  }
   160→   160→
   161→   161→  private buildMetrics(): DeviceMetrics {
   162→   162→    const d = this.def;
   163→   163→    const m: DeviceMetrics = {
   164→   164→      deviceId: d.info.agentId,
   165→   165→      cpuPercent: round1(clamp(jitter(d.cpuBase, d.cpuSpread), 3, 85)),
   166→   166→      memoryPercent: round1(clamp(jitter(d.memBase, d.memSpread), 8, 96)),
   167→   167→      diskPercent: d.diskPercent,
   168→   168→      uptimeSec: Math.floor((Date.now() - this.bootTime) / 1000),
   169→   169→      activeUsers: pick(d.activeUsers),
   170→   170→      netUpKBps: Math.max(1, Math.round(jitter(d.netUpBase, d.netUpSpread))),
   171→   171→      netDownKBps: Math.max(1, Math.round(jitter(d.netDownBase, d.netDownSpread))),
   172→   172→      capturedAt: new Date().toISOString(),
   173→   173→    };
   174→   174→    this.lastMetrics = m;
   175→   175→    this.cpuHistory.push(m.cpuPercent);
   176→   176→    if (this.cpuHistory.length > 64) this.cpuHistory.shift();
   177→   177→    return m;
   178→   178→  }

   180→   180→  private execCtx(): ExecContext {
   181→   181→    const uptimeSec = Math.floor((Date.now() - this.bootTime) / 1000);
   182→   182→    return {
   183→   183→      def: this.def,
   184→   184→      procTable: this.procTable,
   185→   185→      vfs: this.vfs,
   186→   186→      uptimeSec,
   187→   187→      cpuPercent: this.lastMetrics?.cpuPercent ?? this.def.cpuBase,
   188→   188→      memPercent: this.lastMetrics?.memoryPercent ?? this.def.memBase,
   189→   189→      cwd: this.cmdCwd,
   190→   190→    };
   191→   191→  }

   193→   193→  // ── request handling ──────────────────────────────────────────────────

   195→   195→  private async onRequest(msg: RequestMsg): Promise<void> {
   196→   196→    if (this.stopped || !msg || typeof msg.requestId !== "string" || typeof msg.type !== "string") return;
   197→   197→    const payload = (msg.payload ?? {}) as Record<string, unknown>;
   198→   198→    try {
   199→   199→      const data = await this.handle(msg.type, payload);
   200→   200→      this.socket.emit("agent:reply", { requestId: msg.requestId, ok: true, data });
   201→   201→    } catch (err) {
   202→   202→      const error = err instanceof Error ? err.message : String(err);
   203→   203→      this.socket.emit("agent:reply", { requestId: msg.requestId, ok: false, error });
   204→   204→      this.log(`${msg.type} failed: ${error}`);
   205→   205→    }
   206→   206→  }
   207→   207→
   208→   208→  private async handle(type: string, p: Record<string, unknown>): Promise<unknown> {
   209→   209→    switch (type) {
   210→   210→      case "exec": {
   211→   211→        const command = str(p.command);
   212→   212→        const wanted = str(p.shell) || "powershell";
   213→   213→        const effective = wanted === "cmd" ? "cmd" : "powershell"; // sims always have powershell
   214→   214→        const r = effective === "cmd" ? await runCommandCmd(command, this.execCtx()) : await runCommand(command, this.execCtx());
   215→   215→        this.log(
   216→   216→          `exec [${effective}] '${command}' ${r.exitCode === 0 ? "ok" : `exit ${r.exitCode}`} (${r.durationMs}ms)`,
   217→   217→        );
   218→   218→        return r;
   219→   219→      }
   220→   220→      case "fs:list":
   221→   221→        return this.vfs.list(str(p.path));
   222→   222→      case "fs:download":
   223→   223→        return this.vfs.download(str(p.path));
   224→   224→      case "fs:upload":
   225→   225→        return this.vfs.upload(str(p.path), str(p.name), str(p.contentBase64));
   226→   226→      case "fs:delete":
   227→   227→        return this.vfs.remove(str(p.path));
   228→   228→      case "fs:rename":
   229→   229→        return this.vfs.rename(str(p.path), str(p.newName));
   230→   230→      case "fs:mkdir":
   231→   231→        return this.vfs.mkdir(str(p.path));
   232→   232→      case "ps:list":
   233→   233→        return { processes: this.procTable.list(), metrics: this.buildMetrics() };
   234→   234→      case "ps:action": {
   235→   235→        const res = this.procTable.action(Number(p.pid), str(p.action));
   236→   236→        this.log(`ps:action ${str(p.action)} pid=${p.pid} ok`);
   237→   237→        return res;
   238→   238→      }
   239→   239→      case "media:screenshot":
   240→   240→        return this.screenshot();
   241→   241→      case "media:record":
   242→   242→        return this.startMediaRecord(p);
   243→   243→      case "media:stop":
   244→   244→        return this.stopMediaRecord(str(p.assetId));
   245→   245→      case "net:start":
   246→   246→        return this.startNetCapture(p);
   247→   247→      case "net:stop":
   248→   248→        return this.stopNetCapture(str(p.captureId));
   249→   249→      case "location:fix":
   250→   250→        return this.locationFix();
   251→   251→      case "screen:live:start":
   252→   252→        return this.startLiveStream(p);
   253→   253→      case "screen:live:stop":
   254→   254→        return this.stopLiveStream();
   255→   255→      case "stream:start":
   256→   256→        return this.startStream(p);
   257→   257→      case "stream:stop":
   258→   258→        return this.stopStream(str(p.streamId));
   259→   259→      case "record:schedule":
   260→   260→        return this.scheduleRecording(p);
   261→   261→      case "record:cancel":
   262→   262→        return this.cancelRecording(str(p.recordingId));
   263→   263→      default:
   264→   264→        throw new Error(`Unknown request type '${type}'`);
   265→   265→    }
   266→   266→  }

   299→   299→        elapsedSec,
   300→   300→        palette: this.def.palette,
   301→   301→        accent: this.def.accent,

   304→   304→      }),

   308→   308→  private startMediaRecord(p: Record<string, unknown>): { started: true } {
   309→   309→    const assetId = str(p.assetId) || `asset-${Date.now()}`;
   310→   310→    const kind = str(p.kind) || "screen_record";
   311→   311→    const quality = str(p.quality) || "720p";
   312→   312→    const withMic = p.withMic === true;
   313→   313→    const durationSec = typeof p.durationSec === "number" ? p.durationSec : 0;
   314→   314→    if (this.media.has(assetId)) throw new Error(`Capture '${assetId}' is already running`);
   315→   315→
   316→   316→    const bytesPerSec =
   317→   317→      kind === "mic_record"
   318→   318→        ? 16_000
   319→   319→        : kind === "camera_stream" || kind === "call"
   320→   320→          ? 260_000
   321→   321→          : quality === "1080p"
   322→   322→            ? 760_000
   323→   323→            : quality === "480p"
   324→   324→              ? 190_000
   325→   325→              : quality === "source"
   326→   326→                ? 1_150_000
   327→   327→                : 390_000;
   328→   328→
   329→   329→    const cap: MediaCapture = { assetId, kind, quality, withMic, startedAt: Date.now(), bytesPerSec, timer: null };
   330→   330→    this.media.set(assetId, cap);
   331→   331→    this.emitEvent("media:update", { id: assetId, status: "capturing", kind, quality, withMic });
   332→   332→    this.log(`media:record ${assetId} capturing (${kind}, ${quality}, ${durationSec > 0 ? `${durationSec}s` : "indefinite"})`);
   333→   333→
   334→   334→    if (durationSec > 0) {
   335→   335→      cap.timer = setTimeout(() => {
   336→   336→        void this.finishMedia(cap, "available");
   337→   337→      }, Math.min(durationSec, 3600) * 1000);
   338→   338→    }
   339→   339→    return { started: true };
   340→   340→  }
   341→   341→
   342→   342→  private stopMediaRecord(assetId: string): { ok: true } {
   343→   343→    const cap = this.media.get(assetId);
   344→   344→    if (!cap) {
   345→   345→      this.log(`media:stop ${assetId}: no active capture (already finished)`);
   346→   346→      return { ok: true };
   347→   347→    }
   348→   348→    void this.finishMedia(cap, "stopped");
   349→   349→    return { ok: true };
   350→   350→  }
   351→   351→
   352→   352→  private async finishMedia(cap: MediaCapture, status: "available" | "stopped"): Promise<void> {
   353→   353→    if (cap.timer) {
   354→   354→      clearTimeout(cap.timer);
   355→   355→      cap.timer = null;
   356→   356→    }
   357→   357→    if (!this.media.has(cap.assetId) || this.stopped) return;
   358→   358→    this.media.delete(cap.assetId);
   359→   359→    const elapsed = (Date.now() - cap.startedAt) / 1000;
   360→   360→    const sizeBytes = Math.max(24_000, Math.round(cap.bytesPerSec * elapsed * (0.85 + Math.random() * 0.3)));
   361→   361→    const dataRef = this.poster(cap, elapsed);
   362→   362→    this.log(`media:record ${cap.assetId} ${status} (${fmtBytes(sizeBytes)}, ${Math.round(elapsed)}s)`);
   363→   363→    this.emitEvent("media:update", { id: cap.assetId, status, sizeBytes });
   364→   364→    await postJson(
   365→   365→      nextApi("/api/internal/media/result"),
   366→   366→      { assetId: cap.assetId, status, sizeBytes, dataRef },
   367→   367→      { "x-hub-token": CONFIG.hubToken },
   368→   368→    );
   369→   369→  }

   373→   373→  private startLiveStream(p: Record<string, unknown>): { started: true; fps: number } {
   374→   374→    if (this.live) throw new Error("A live screen stream is already running");
   375→   375→    const fps = Math.min(3, Math.max(0.5, typeof p.fps === "number" && p.fps > 0 ? p.fps : 1));
   376→   376→    const stream: LiveStream = { seq: 0, fps, startedAt: Date.now(), timer: null };
   377→   377→    this.live = stream;
   378→   378→    const interval = Math.round(1000 / fps);
   379→   379→    stream.timer = setInterval(() => {
   380→   380→      if (!this.live || this.stopped || !this.socket.connected) return;
   381→   381→      stream.seq += 1;
   382→   382→      const shot = this.screenshot();
   383→   383→      this.socket.emit("agent:event", {
   384→   384→        name: "screen:frame",
   385→   385→        payload: {
   386→   386→          agentId: this.def.info.agentId,
   387→   387→          dataUri: shot.dataUri,
   388→   388→          ts: Date.now(),
   389→   389→          seq: stream.seq,
   390→   390→          width: 1600,
   391→   391→          height: 900,
   392→   392→        },
   393→   393→      });
   394→   394→    }, interval);
   395→   395→    this.log(`screen:live started (${fps} fps)`);
   396→   396→    return { started: true, fps };
   397→   397→  }
   398→   398→
   399→   399→  private stopLiveStream(): { frames: number } {
   400→   400→    const stream = this.live;
   401→   401→    if (!stream) return { frames: 0 };
   402→   402→    if (stream.timer) clearInterval(stream.timer);
   403→   403→    this.live = null;
   404→   404→    this.log(`screen:live stopped (${stream.seq} frames streamed)`);
   405→   405→    return { frames: stream.seq };
   406→   406→  }

   410→   410→  private num(v: unknown): number | null {
   411→   411→    return typeof v === "number" && Number.isFinite(v) ? v : null;

   414→   414→  private startStream(p: Record<string, unknown>): { started: true; meta: Record<string, unknown> } {
   415→   415→    const streamId = str(p.streamId);
   416→   416→    if (!streamId) throw new Error("streamId is required");
   417→   417→    const kind = str(p.kind);
   418→   418→    if (kind !== "screen" && kind !== "camera" && kind !== "mic") {
   419→   419→      throw new Error("kind must be one of: screen, camera, mic");
   420→   420→    }
   421→   421→    if (this.streams.has(streamId)) throw new Error(`Stream '${streamId}' is already running`);
   422→   422→    const caps = this.def.captures;
   423→   423→
   424→   424→    let stream: SimStream;
   425→   425→    let meta: Record<string, unknown>;
   426→   426→    if (kind === "screen") {
   427→   427→      const sourceId = str(p.sourceId) || caps.monitors[0]?.id || "monitor:0";
   428→   428→      const idx = caps.monitors.findIndex((m) => m.id === sourceId);
   429→   429→      if (idx < 0) {
   430→   430→        throw new Error(
   431→   431→          `Unknown monitor '${sourceId}' — available: ${caps.monitors.map((m) => m.id).join(", ") || "none"}`,
   432→   432→        );
   433→   433→      }
   434→   434→      const mon = caps.monitors[idx]!;
   435→   435→      const fps = clamp(this.num(p.fps) ?? 2, 0.5, 5);
   436→   436→      stream = { streamId, kind, sourceId, fps, width: mon.width, height: mon.height, seq: 0, timer: null, startedAt: Date.now() };
   437→   437→      meta = { width: mon.width, height: mon.height, fps, sourceId, name: mon.name, primary: mon.primary };
   438→   438→    } else if (kind === "camera") {
   439→   439→      const sourceId = str(p.sourceId) || caps.cameras[0]?.id || "camera:0";
   440→   440→      const idx = caps.cameras.findIndex((c) => c.id === sourceId);
   441→   441→      if (idx < 0) {
   442→   442→        throw new Error(`Unknown camera '${sourceId}' — available: ${caps.cameras.map((c) => c.id).join(", ") || "none"}`);
   443→   443→      }
   444→   444→      const fps = clamp(this.num(p.fps) ?? 10, 1, 15);
   445→   445→      stream = { streamId, kind, sourceId, fps, width: 1280, height: 720, seq: 0, timer: null, startedAt: Date.now() };
   446→   446→      meta = { width: 1280, height: 720, fps, sourceId, name: caps.cameras[idx]!.name };
   447→   447→    } else {
   448→   448→      const sourceId = str(p.sourceId) || caps.mics[0]?.id || "mic:0";
   449→   449→      const idx = caps.mics.findIndex((m) => m.id === sourceId);
   450→   450→      if (idx < 0) {
   451→   451→        throw new Error(`Unknown mic '${sourceId}' — available: ${caps.mics.map((m) => m.id).join(", ") || "none"}`);
   452→   452→      }
   453→   453→      stream = { streamId, kind, sourceId, fps: 8, width: 0, height: 0, seq: 0, timer: null, startedAt: Date.now() };
   454→   454→      meta = { fps: 8, sourceId, name: caps.mics[idx]!.name };
   455→   455→    }
   456→   456→
   457→   457→    this.streams.set(streamId, stream);
   458→   458→    const interval = Math.round(1000 / stream.fps);
   459→   459→    stream.timer = setInterval(() => this.emitStreamFrame(stream), interval);
   460→   460→    this.log(`stream:start ${kind} '${stream.sourceId}' @${stream.fps}fps (streamId ${streamId.slice(0, 8)}…, ${this.streams.size} active)`);
   461→   461→    return { started: true, meta };
   462→   462→  }
   463→   463→
   464→   464→  private emitStreamFrame(stream: SimStream): void {
   465→   465→    if (this.stopped || !this.socket.connected) return;
   466→   466→    stream.seq += 1;
   467→   467→    const frame: Record<string, unknown> = {
   468→   468→      agentId: this.def.info.agentId,
   469→   469→      streamId: stream.streamId,
   470→   470→      kind: stream.kind,
   471→   471→      ts: Date.now(),
   472→   472→      seq: stream.seq,
   473→   473→    };
   474→   474→    if (stream.kind === "screen") {
   475→   475→      const shot = this.renderScreenSource(stream.sourceId, 1600, 900);
   476→   476→      frame.dataUri = shot.dataUri;
   477→   477→      frame.width = shot.width;
   478→   478→      frame.height = shot.height;
   479→   479→    } else if (stream.kind === "camera") {
   480→   480→      const idx = Math.max(0, this.def.captures.cameras.findIndex((c) => c.id === stream.sourceId));
   481→   481→      const shot = this.renderCameraSource(stream.sourceId, idx);
   482→   482→      frame.dataUri = shot.dataUri;
   483→   483→      frame.width = 1280;
   484→   484→      frame.height = 720;
   485→   485→    } else {
   486→   486→      frame.level = this.micLevel();
   487→   487→    }
   488→   488→    this.socket.emit("agent:event", { name: "stream:frame", payload: frame });
   489→   489→  }
   490→   490→
   491→   491→  private stopStream(streamId: string): { frames: number } {
   492→   492→    const stream = this.streams.get(streamId);
   493→   493→    if (!stream) return { frames: 0 };
   494→   494→    if (stream.timer) clearInterval(stream.timer);
   495→   495→    this.streams.delete(streamId);
   496→   496→    this.log(`stream:stop ${stream.kind} '${stream.sourceId}' (${stream.seq} frames streamed, ${this.streams.size} active)`);
   497→   497→    return { frames: stream.seq };
   498→   498→  }
   499→   499→
   500→   500→  private stopAllStreams(): void {
   501→   501→    for (const stream of this.streams.values()) {
   502→   502→      if (stream.timer) clearInterval(stream.timer);
   503→   503→    }
   504→   504→    if (this.streams.size > 0) {
   505→   505→      this.log(`all ${this.streams.size} active stream(s) stopped (agent disconnect)`);
   506→   506→    }
   507→   507→    this.streams.clear();
   508→   508→  }

   511→   511→  private renderScreenSource(sourceId: string, w: number, h: number): { dataUri: string; width: number; height: number } {
   512→   512→    const idx = Math.max(0, this.def.captures.monitors.findIndex((m) => m.id === sourceId));
   513→   513→    const m = this.lastMetrics;
   514→   514→    const svg = svgDesktop({
   515→   515→      hostname: this.def.info.hostname,
   516→   516→      ip: this.def.info.localIp,
   517→   517→      osLabel: `${this.def.info.osName} ${this.def.info.osVersion}`,
   518→   518→      agentVersion: this.def.info.agentVersion,
   519→   519→      cpu: m?.cpuPercent ?? this.def.cpuBase,
   520→   520→      mem: m?.memoryPercent ?? this.def.memBase,
   521→   521→      disk: this.def.diskPercent,
   522→   522→      netUpKBps: Math.max(4, Math.round(jitter(this.def.netUpBase, this.def.netUpSpread))),
   523→   523→      netDownKBps: Math.max(16, Math.round(jitter(this.def.netDownBase, this.def.netDownSpread))),
   524→   524→      uptimeSec: Math.floor((Date.now() - this.bootTime) / 1000),
   525→   525→      cpuHistory: this.cpuHistory,
   526→   526→      palette: this.def.palette,
   527→   527→      accent: this.def.accent,
   528→   528→      now: new Date(),
   529→   529→      w,
   530→   530→      h,
   531→   531→      variant: idx === 0 ? 0 : 1,
   532→   532→    });
   533→   533→    return { dataUri: toDataUri(svg), width: w, height: h };
   534→   534→  }

   550→   550→  /** Mic input level 0.05..0.9 (sinusoid + noise), 8 Hz cadence. */
   551→   551→  private micLevel(): number {
   552→   552→    const t = Date.now() / 1000;
   553→   553→    const base = 0.475 + 0.32 * Math.sin(t * 2.2) + 0.1 * Math.sin(t * 5.7);
   554→   554→    const noise = (Math.random() * 2 - 1) * 0.08;
   555→   555→    return Number(clamp(base + noise, 0.05, 0.9).toFixed(3));
   556→   556→  }

   558→   558→  // ── v0.3 scheduled recordings ─────────────────────────────────────────

   560→   560→  private scheduleRecording(p: Record<string, unknown>): { accepted: true; state: "armed" } {
   561→   561→    const recordingId = str(p.recordingId);
   562→   562→    if (!recordingId) throw new Error("recordingId is required");
   563→   563→    if (this.recorder && !this.recorder.isDone()) {
   564→   564→      throw new Error("A recording is already active on this device");
   565→   565→    }
   566→   566→    const passphrase = str(p.passphrase);
   567→   567→    if (!passphrase) throw new Error("passphrase is required");
   568→   568→    const durationSec = clamp(Math.round(this.num(p.durationSec) ?? 0), 5, 120); // sim hard cap: 120s
   569→   569→    const list = (v: unknown): string[] | undefined =>
   570→   570→      Array.isArray(v) && v.length > 0 ? v.filter((x): x is string => typeof x === "string") : undefined;
   571→   571→    const sources = {
   572→   572→      ...(list(p.monitors) ? { monitors: list(p.monitors) } : {}),
   573→   573→      ...(list(p.cameras) ? { cameras: list(p.cameras) } : {}),
   574→   574→      ...(list(p.mics) ? { mics: list(p.mics) } : {}),
   575→   575→    };
   576→   576→    const label = str(p.label) || null;
   577→   577→    this.log(
   578→   578→      `record:schedule ${recordingId} armed (${durationSec}s) — on-device notice shown, indicator will be visible while recording`,
   579→   579→    );
   580→   580→    this.recorder = new SimRecorder({
   581→   581→      socket: this.socket,
   582→   582→      agentId: this.def.info.agentId,
   583→   583→      hostname: this.def.info.hostname,
   584→   584→      recordingId,
   585→   585→      durationSec,
   586→   586→      passphrase,
   587→   587→      label,
   588→   588→      sources,
   589→   589→      renderScreen: (sourceId) => this.renderScreenSource(sourceId, 1280, 720),
   590→   590→      renderCamera: (sourceId) =>
   591→   591→        this.renderCameraSource(sourceId, Math.max(0, this.def.captures.cameras.findIndex((c) => c.id === sourceId))),
   592→   592→      micLevel: () => this.micLevel(),
   593→   593→      log: (msg) => this.log(msg),
   594→   594→    });
   595→   595→    return { accepted: true, state: "armed" };
   596→   596→  }
   597→   597→
   598→   598→  private cancelRecording(recordingId: string): { ok: true; state: string } {
   599→   599→    if (!this.recorder || this.recorder.isDone()) {
   600→   600→      this.log(`record:cancel ${recordingId}: no active recording`);
   601→   601→      return { ok: true, state: "idle" };
   602→   602→    }
   603→   603→    this.recorder.cancel();
   604→   604→    return { ok: true, state: "canceled" };
   605→   605→  }

   607→   607→  // ── network capture ───────────────────────────────────────────────────
   608→   608→
   609→   609→  private startNetCapture(p: Record<string, unknown>): { started: true } {
   610→   610→    const captureId = str(p.captureId) || `cap-${Date.now()}`;
   611→   611→    if (this.net.has(captureId)) throw new Error(`Capture '${captureId}' is already running`);
   612→   612→    const cap: NetCapture = {
   613→   613→      captureId,
   614→   614→      iface: str(p.iface) || "Ethernet",
   615→   615→      filterBpf: str(p.filterBpf) || null,
   616→   616→      packets: 0,
   617→   617→      bytesCaptured: 0,
   618→   618→      timer: null,
   619→   619→    };
   620→   620→    this.net.set(captureId, cap);
   621→   621→    cap.timer = setInterval(() => {
   622→   622→      cap.packets += rndInt(20, 80);
   623→   623→      cap.bytesCaptured += rndInt(2_048, 61_440);
   624→   624→      this.emitEvent("network:update", {
   625→   625→        id: captureId,
   626→   626→        status: "running",
   627→   627→        packets: cap.packets,
   628→   628→        bytesCaptured: cap.bytesCaptured,
   629→   629→      });
   630→   630→      void postJson(
   631→   631→        nextApi("/api/internal/network/stats"),
   632→   632→        { captureId, packets: cap.packets, bytesCaptured: cap.bytesCaptured },
   633→   633→        { "x-hub-token": CONFIG.hubToken },
   634→   634→      );
   635→   635→    }, 3000);
   636→   636→    this.log(`net:capture ${captureId} started on ${cap.iface}${cap.filterBpf ? ` (bpf: ${cap.filterBpf})` : ""}`);
   637→   637→    return { started: true };
   638→   638→  }
   639→   639→
   640→   640→  private stopNetCapture(captureId: string): { packets: number; bytesCaptured: number } {
   641→   641→    const cap = this.net.get(captureId);
   642→   642→    if (!cap) throw new Error(`No active capture '${captureId}'`);
   643→   643→    if (cap.timer) {
   644→   644→      clearInterval(cap.timer);
   645→   645→      cap.timer = null;
   646→   646→    }
   647→   647→    this.net.delete(captureId);
   648→   648→    this.log(`net:capture ${captureId} stopped (${cap.packets} packets, ${fmtBytes(cap.bytesCaptured)})`);
   649→   649→    this.emitEvent("network:update", {
   650→   650→      id: captureId,
   651→   651→      status: "stopped",
   652→   652→      packets: cap.packets,
   653→   653→      bytesCaptured: cap.bytesCaptured,
   654→   654→    });
   655→   655→    return { packets: cap.packets, bytesCaptured: cap.bytesCaptured };
   656→   656→  }
