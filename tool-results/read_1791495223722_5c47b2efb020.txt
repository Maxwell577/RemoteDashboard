    55→function fmtUptime(sec: number): string {
    56→  const d = Math.floor(sec / 86400)
    57→  const h = Math.floor((sec % 86400) / 3600)
    58→  const min = Math.floor((sec % 3600) / 60)
    59→  if (d > 0) return `${d}d ${h}h`
    60→  if (h > 0) return `${h}h ${min}m`
    61→  return `${min}m`
    62→}

   111→/** SVG donut showing a 0-100 health score with a tinted arc. */
   112→function HealthRing({ score, size = 64 }: { score: number; size?: number }) {
   113→  const stroke = 6
   114→  const r = (size - stroke) / 2
   115→  const c = 2 * Math.PI * r
   116→  const pct = Math.min(100, Math.max(0, score))
   117→  const color =
   118→    score >= 80
   119→      ? 'oklch(0.72 0.14 168)' // emerald
   120→      : score >= 55
   121→        ? 'oklch(0.8 0.14 85)' // amber
   122→        : 'oklch(0.7 0.19 22)' // rose
   123→  return (
   124→    <div
   125→      className="relative shrink-0"
   126→      style={{ width: size, height: size }}
   127→      role="img"
   128→      aria-label={`Health score ${score} of 100`}
   129→    >
   130→      <svg width={size} height={size} className="-rotate-90">
   131→        <circle
   132→          cx={size / 2}
   133→          cy={size / 2}
   134→          r={r}
   135→          fill="none"
   136→          stroke="currentColor"
   137→          className="text-border/60"
   138→          strokeWidth={stroke}
   139→        />
   140→        <circle
   141→          cx={size / 2}
   142→          cy={size / 2}
   143→          r={r}
   144→          fill="none"
   145→          stroke={color}
   146→          strokeWidth={stroke}
   147→          strokeLinecap="round"
   148→          strokeDasharray={`${(pct / 100) * c} ${c}`}
   149→          style={{ transition: 'stroke-dasharray 600ms cubic-bezier(0.22,1,0.36,1)' }}
   150→        />
   151→      </svg>
   152→      <span
   153→        className="absolute inset-0 flex items-center justify-center font-mono text-sm font-semibold"
   154→        style={{ color }}
   155→      >
   156→        {score}
   157→      </span>
   158→    </div>
   159→  )
   160→}

   162→/** Tiny inline CPU% sparkline from the live telemetry buffer. */
   163→function CpuSparkline({ history, className }: { history: number[]; className?: string }) {
   164→  const w = 220
   165→  const h = 48
   166→  if (history.length < 2) {
   167→    return (
   168→      <div className={cn('flex items-center justify-center text-[11px] text-muted-foreground', className)} style={{ height: h }}>
   169→        collecting samples…
   170→      </div>
   171→    )
   172→  }
   173→  const pts = history
   174→    .map((v, i) => {
   175→      const x = (i / (history.length - 1)) * w
   176→      const y = h - (Math.min(100, Math.max(0, v)) / 100) * (h - 6) - 3
   177→      return `${x.toFixed(1)},${y.toFixed(1)}`
   178→    })
   179→    .join(' ')
   180→  const last = history[history.length - 1]
   181→  return (
   182→    <svg viewBox={`0 0 ${w} ${h}`} className={cn('w-full', className)} style={{ height: h }} aria-hidden>
   183→      <polyline
   184→        points={pts}
   185→        fill="none"
   186→        stroke="currentColor"
   187→        className="text-primary"
   188→        strokeWidth="2"
   189→        strokeLinejoin="round"
   190→        strokeLinecap="round"
   191→      />
   192→      <circle
   193→        cx={w}
   194→        cy={h - (Math.min(100, Math.max(0, last)) / 100) * (h - 6) - 3}
   195→        r="3"
   196→        className="fill-primary"
   197→      />
   198→    </svg>
   199→  )
   200→}

   202→function healthTone(score: number): string {
   203→  if (score >= 80) return 'border-emerald-500/30 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   204→  if (score >= 55) return 'border-amber-500/30 bg-amber-500/10 text-amber-600 dark:text-amber-400'
   205→  return 'border-rose-500/30 bg-rose-500/10 text-rose-600 dark:text-rose-400'
   206→}

   557→            <dt className="shrink-0 text-muted-foreground">{online ? 'Status' : 'Last seen'}</dt>
   558→            <dd
   559→              className={cn(
   560→                'truncate text-right font-medium',
   561→                online && !m && 'text-muted-foreground',
   562→                online && m && 'text-emerald-600 dark:text-emerald-400',
   563→              )}
   564→            >
   565→              {online
   566→                ? m
   567→                  ? `live · CPU ${Math.round(m.cpuPercent)}% · MEM ${Math.round(m.memoryPercent)}%`
   568→                  : 'waiting for telemetry…'
   569→                : timeAgo(device.lastSeen)}
   570→            </dd>
   571→          </div>
   572→          {online && m ? (
   573→            <div className="flex items-baseline justify-between gap-3">
   574→              <dt className="flex shrink-0 items-center gap-1 text-muted-foreground">
   575→                <Timer className="size-3" aria-hidden />
   576→                Uptime
   577→              </dt>
   578→              <dd
   579→                className="truncate text-right font-medium tabular-nums"
   580→                title={`Last boot ${new Date(Date.now() - m.uptimeSec * 1000).toLocaleString()}`}
   581→              >
   582→                {fmtUptime(m.uptimeSec)}
   583→              </dd>
   584→            </div>
   585→          ) : null}
   586→        </dl>
   587→
   588→        {device.tags.length > 0 ? (
   589→          <div className="mt-3 flex flex-wrap gap-1">
   590→            {device.tags.map((t) => (
   591→              <Badge key={t} variant="secondary" className="text-[10px]">
   592→                {t}
   593→              </Badge>
   594→            ))}
   595→          </div>
   596→        ) : null}
   597→
   598→        <div className="mt-4 flex items-center gap-2 border-t border-border/50 pt-3">
   599→          <Button size="sm" className="flex-1" onClick={() => go('terminal')}>
   600→            <SquareTerminal className="size-3.5" aria-hidden /> Manage
   601→          </Button>
   602→          <Button
   603→            size="sm"
   604→            variant="outline"
   605→            className="flex-1"
   606→            onClick={() => setDetailsOpen(true)}
   607→          >
   608→            <Info className="size-3.5" aria-hidden /> Details
   609→          </Button>
   610→          <Button size="sm" variant="outline" className="flex-1" onClick={() => go('location')}>
   611→            <MapPin className="size-3.5" aria-hidden /> Locate
   612→          </Button>
   613→        </div>
   614→      </Panel>
   615→
   616→      <DeviceDetailsDialog
   617→        device={device}
   618→        open={detailsOpen}
   619→        onOpenChange={setDetailsOpen}
   620→      />
   621→    </>
   622→  )
   623→}
