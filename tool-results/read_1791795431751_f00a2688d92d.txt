    24→// ── small text helpers ───────────────────────────────────────────────────
    25→const w = (s: string, n: number): string => (s.length > n ? s.slice(0, n) : s.padEnd(n));
    26→const wl = (s: string, n: number): string => (s.length > n ? s : s.padStart(n));
    27→const num = (n: number): string => Math.round(n).toLocaleString("en-US");

    29→function fmtLong(d: Date): string {
    30→  return d.toLocaleString("en-US", {
    31→    year: "numeric",
    32→    month: "numeric",
    33→    day: "numeric",
    34→    hour: "numeric",
    35→    minute: "2-digit",
    36→    second: "2-digit",
    37→    hour12: true,
    38→  });
    39→}
    40→
    41→function fmtDirDate(d: Date): string {
    42→  const time = d.toLocaleTimeString("en-US", { hour: "numeric", minute: "2-digit", hour12: true });
    43→  return `${d.getMonth() + 1}/${d.getDate()}/${d.getFullYear()} ${time}`;
    44→}

    46→/** EUI-64 style link-local IPv6 derived from the device MAC. */
    47→function linkLocal(mac: string): string {
    48→  const hex = mac.replace(/:/g, "").toLowerCase();
    49→  const first = (parseInt(hex.slice(0, 2), 16) ^ 0x02).toString(16).padStart(2, "0");
    50→  return `fe80::${first}${hex.slice(2, 4)}:${hex.slice(4, 6)}ff:fe${hex.slice(6, 8)}:${hex.slice(8, 10)}${hex.slice(10, 12)}`;
    51→}

    53→function notRecognized(cmd: string): string {
    54→  return `${cmd} : The term '${cmd}' is not recognized as a name of a cmdlet, function, operable program, or script file. Check the spelling of the name, or if a path was included, verify that the path is correct and try again.
    55→At line:1 char:1
    56→+ ${cmd}
    57→+ ~~~~~
    58→    + CategoryInfo          : ObjectNotFound: (${cmd}:String) [], CommandNotFoundException
    59→    + FullyQualifiedErrorId : CommandNotFoundException`;
    60→}

   201→function dir(ctx: ExecContext, target: string | null): { output: string; exitCode: number } {
   202→  const fallback = `C:\\Users\\${ctx.def.userName}`;
   203→  const path = target && /^[a-zA-Z]:/.test(target) ? target : fallback;
   204→  let listing;
   205→  try {
   206→    listing = ctx.vfs.list(path);
   207→  } catch {
   208→    return {
   209→      output: `dir : Cannot find path '${path}' because it does not exist.
   210→At line:1 char:1
   211→+ dir${target ? ` ${target}` : ""}
   212→+ ~~~~~
   213→    + CategoryInfo          : ObjectNotFound: (${path}:String) [Get-ChildItem], ItemNotFoundException
   214→    + FullyQualifiedErrorId : DirIOPathNotFound,Microsoft.PowerShell.Commands.GetChildItemCommand`,
   215→      exitCode: 1,
   216→    };
   217→  }
   218→  const rows = [
   219→    "",
   220→    `    Directory: ${listing.path}`,
   221→    "",
   222→    "Mode                 LastWriteTime         Length Name",
   223→    "----                 -------------         ------ ----",
   224→  ];
   225→  for (const e of listing.entries) {
   226→    const mode = e.type === "directory" ? (e.hidden ? "d--h--" : "d-----") : e.hidden ? "-a-h--" : "-a----";
   227→    const dt = fmtDirDate(new Date(e.modifiedAt));
   228→    const len = e.type === "file" ? num(e.sizeBytes) : "";
   229→    rows.push(`${mode}         ${dt.padStart(18)}${len.padStart(13)} ${e.name}`);
   230→  }
   231→  return { output: rows.join("\n"), exitCode: 0 };
   232→}

   259→function netstat(ctx: ExecContext): string {
   260→  const d = ctx.def;
   261→  const ip = d.info.localIp;
   262→  const rows = [
   263→    "",
   264→    "Active Connections",
   265→    "",
   266→    "  Proto  Local Address          Foreign Address        State",
   267→  ];
   268→  const listen = (addr: string, port: number): void => {
   269→    rows.push(`  TCP    ${w(`${addr}:${port}`, 22)} ${w("0.0.0.0:0", 22)}LISTENING`);
   270→  };
   271→  listen("0.0.0.0", 135);
   272→  listen("0.0.0.0", 445);
   273→  listen("0.0.0.0", 5357);
   274→  if (d.profile === "server") {
   275→    listen("0.0.0.0", 3389);
   276→    listen("0.0.0.0", 22);
   277→    listen("0.0.0.0", 5985);
   278→  } else {
   279→    listen("0.0.0.0", 139);
   280→    listen("0.0.0.0", 5040);
   281→  }
   282→  listen("[::]", 135);
   283→  listen("[::]", 445);
   284→  // the hub connection itself
   285→  rows.push(`  TCP    ${w("127.0.0.1:3003", 22)} ${w(`127.0.0.1:${rndInt(49152, 65500)}`, 22)}ESTABLISHED`);
   286→  const foreign = ["104.18.32.115", "142.250.72.68", "52.113.194.132", "13.107.42.14", "1.1.1.1", "93.184.216.34", "20.42.65.92", "151.101.1.69"];
   287→  for (let i = 0; i < rndInt(6, 10); i++) {
   288→    rows.push(`  TCP    ${w(`${ip}:${rndInt(49152, 65500)}`, 22)} ${w(`${pick(foreign)}:443`, 22)}ESTABLISHED`);
   289→  }
   290→  rows.push(`  TCP    ${w(`${ip}:${rndInt(49152, 65500)}`, 22)} ${w(`${pick(foreign)}:443`, 22)}TIME_WAIT`);
   291→  rows.push(`  TCP    ${w(`${ip}:${rndInt(49152, 65500)}`, 22)} ${w(`${pick(foreign)}:80`, 22)}TIME_WAIT`);
   292→  rows.push("  UDP    0.0.0.0:5353           *:*");
   293→  rows.push("  UDP    0.0.0.0:68             *:*");
   294→  rows.push("  UDP    [::]:500               *:*");
   295→  return rows.slice(0, 44).join("\n");
   296→}

   724→export async function runCommand(command: string, ctx: ExecContext): Promise<ExecResult> {
   725→  const trimmed = (command ?? "").trim();
   726→  const lower = trimmed.toLowerCase();
   727→  const t0 = Date.now();
   728→
   729→  let output: string;
   730→  let exitCode = 0;
   731→
   732→  const dirCmd = /^(dir|ls|gci|get-childitem)(\s+(.*))?$/.exec(lower);
   733→  const args = (cmd: string): string => trimmed.slice(cmd.length).trim();
   734→
   735→  if (lower === "sysinfo" || lower === "systeminfo" || lower === "get-computerinfo") {
   736→    output = sysinfo(ctx);
   737→  } else if (lower === "get-date" || lower === "date") {
   738→    output = getDatePs();
   739→  } else if (lower === "hostname") {
   740→    output = ctx.def.info.hostname;
   741→  } else if (lower === "whoami" || lower.startsWith("whoami ")) {
   742→    if (lower.includes("/priv")) output = whoamiPriv(ctx);
   743→    else if (lower.includes("/groups")) output = whoamiGroups(ctx);
   744→    else output = `${ctx.def.info.hostname.toLowerCase()}\\${ctx.def.userName}`;
   745→  } else if (lower.startsWith("echo ") || lower.startsWith("write-host ") || lower.startsWith("write-output ") || lower === "echo") {
   746→    const text = lower === "echo" ? "" : trimmed.slice(trimmed.indexOf(" ") + 1);
   747→    output = text.replace(/^"(.*)"$/, "$1").replace(/^'(.*)'$/, "$1");
   748→  } else if (lower.startsWith("ipconfig")) {
   749→    output = ipconfig(ctx, lower.includes("/all"));
   750→  } else if (lower.startsWith("ping ") || lower.startsWith("test-connection") || lower === "ping") {
   751→    const target = lower.startsWith("ping ") ? args("ping") : lower.startsWith("test-connection ") ? args("test-connection") : "";
   752→    const res = ping(target.replace(/-\w+\s+\S+\s*/g, "").trim());
   753→    output = res.output;
   754→    exitCode = res.exitCode;
   755→  } else if (lower.startsWith("tracert") || lower.startsWith("pathping") || lower.startsWith("trace route")) {
   756→    output = tracert(lower.startsWith("tracert ") ? args("tracert") : lower.startsWith("pathping ") ? args("pathping") : "");
   757→  } else if (lower.startsWith("nslookup")) {
   758→    output = nslookup(lower === "nslookup" ? "" : args("nslookup").replace(/^-\S+\s*/, ""));
   759→  } else if (lower === "arp -a" || lower === "arp" || lower.startsWith("arp ")) {
   760→    output = arpA(ctx);
   761→  } else if (lower.startsWith("route print") || lower === "route") {
   762→    output = routePrint(ctx);
   763→  } else if (lower.startsWith("netstat")) {
   764→    output = netstat(ctx);
   765→  } else if (lower.startsWith("tasklist")) {
   766→    output = tasklist(ctx);
   767→  } else if (lower === "get-process" || lower === "ps" || lower.startsWith("get-process ") || lower.startsWith("ps ")) {
   768→    const argStr = lower.startsWith("get-process ") ? args("get-process") : lower.startsWith("ps ") ? args("ps") : "";
   769→    const nameMatch = /-name\s+([^\s]+|"[^"]+")/.exec(argStr);
   770→    if (nameMatch) {
   771→      const needle = nameMatch[1].replace(/"/g, "").toLowerCase();
   772→      const filtered = ctx.procTable.list().filter((p) => p.name.toLowerCase().includes(needle));
   773→      output = filtered.length === 0 ? `Get-Process : No process matches '${needle}'.` : getProcess(ctx).split("\n").slice(0, 3).concat(filtered.map((p) => ` ${wl(String(rndInt(8, 40)), 7)}${wl((p.memoryMB * 0.82).toFixed(2), 10)}${wl(p.memoryMB.toFixed(2), 10)}${wl((p.cpuPercent * rndInt(20, 220)).toFixed(2), 12)}${wl(String(p.pid), 9)}${wl(p.user === "SYSTEM" ? "0" : "1", 4)} ${p.name.replace(/\.exe$/i, "")}`)).join("\n");
   774→      if (filtered.length === 0) exitCode = 1;
   775→    } else {
   776→      output = getProcess(ctx);
   777→    }
   778→  } else if (/^stop-process(\s|$)/.test(lower)) {
   779→    const idMatch = /-id\s+(\d+)/.exec(lower);
   780→    if (!idMatch) {
   781→      output = "Stop-Process : A positional parameter cannot be found that accepts argument '-Name'. Use Stop-Process -Id <pid>.";
   782→      exitCode = 1;
   783→    } else {
   784→      try {
   785→        ctx.procTable.action(parseInt(idMatch[1], 10), "kill");
   786→        output = "";
   787→      } catch (e) {
   788→        output = `Stop-Process : ${(e as Error).message}`;
   789→        exitCode = 1;
   790→      }
   791→    }
   792→  } else if (lower === "ver" || lower === "cmd /c ver") {
   793→    output = `Microsoft Windows [Version ${ctx.def.osBuild}]`;
   794→  } else if (lower === "get-location" || lower === "pwd" || lower === "gl") {
   795→    output = getLocation(ctx);
   796→  } else if (dirCmd) {
   797→    const target = dirCmd[3] ? dirCmd[3].trim() : null;
   798→    if (target && /^(env|variable)$/i.test(target.replace("env:", "env").split(":")[0])) {
   799→      output = envList(ctx);
   800→    } else {
   801→      const res = dir(ctx, target && target !== "" && !target.startsWith("/") ? target : null);
   802→      output = res.output;
   803→      exitCode = res.exitCode;
   804→    }
   805→  } else if (lower === "get-netipaddress" || lower === "gip") {
   806→    output = getNetIpAddress(ctx);
   807→  } else if (lower.startsWith("get-netadapter")) {
   808→    output = getNetAdapter(ctx);
   809→  } else if (lower === "get-volume") {
   810→    output = getVolume(ctx);
   811→  } else if (lower.startsWith("get-psdrive") || lower.startsWith("gpsdrive")) {
   812→    output = getPsDrive(ctx);
   813→  } else if (lower === "qwinsta" || lower === "query session") {
   814→    output = qwinsta(ctx);
   815→  } else if (lower === "get-service" || lower.startsWith("get-service ") || lower === "gsv") {
   816→    const raw = lower === "get-service" || lower === "gsv" ? "" : args(lower === "gsv" ? "gsv" : "get-service");
   817→    const nameMatch = /-name\s+("[^"]+"|[^\s]+)/.exec(raw);
   818→    const filter = nameMatch ? nameMatch[1].replace(/"/g, "") : raw.replace(/-\w+/, "").trim();
   819→    const res = getService(ctx, filter || null);
   820→    output = res.output;
   821→    exitCode = res.exitCode;
   822→  } else if (lower.startsWith("get-eventlog") || lower.startsWith("get-winevent")) {
   823→    output = getEventLog(ctx);
   824→  } else if (lower.startsWith("get-localuser")) {
   825→    output = getLocalUser(ctx);
   826→  } else if (lower.startsWith("get-hotfix")) {
   827→    output = getHotfix();
   828→  } else if (lower.startsWith("get-scheduledtask")) {
   829→    output = getScheduledTask(ctx);
   830→  } else if (/get-ciminstance|get-wmiobject/.test(lower) && /win32_operatingsystem/.test(lower)) {
   831→    output = getCimOs(ctx);
   832→  } else if (lower === "$psversiontable" || lower.startsWith("$psversiontable") || lower === "get-host" || lower === "$host" || lower.startsWith("get-host")) {
   833→    output = psVersionTable(ctx);
   834→  } else if (lower === "get-random") {
   835→    output = String(rndInt(0, 2147483647));
