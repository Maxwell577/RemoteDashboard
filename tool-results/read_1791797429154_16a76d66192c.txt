    88→/** sha256 hex prefix (12 chars) of the passphrase — matches the agent's keyFingerprint. */
    89→async function sha256HexPrefix(input: string): Promise<string | null> {
    90→  try {
    91→    if (typeof crypto === 'undefined' || !crypto.subtle) return null
    92→    const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(input))
    93→    return Array.from(new Uint8Array(buf))
    94→      .map((b) => b.toString(16).padStart(2, '0'))
    95→      .join('')
    96→      .slice(0, 12)
    97→  } catch {
    98→    return null
    99→  }
   100→}
   101→
   102→function generateStrongPassphrase(): string {
   103→  const pick = <T,>(arr: T[]): T => {
   104→    const buf = new Uint32Array(1)
   105→    crypto.getRandomValues(buf)
   106→    return arr[buf[0] % arr.length]
   107→  }
   108→  const words = [pick(WORDS), pick(WORDS), pick(WORDS), pick(WORDS)]
   109→  const digits = String(crypto.getRandomValues(new Uint16Array(1))[0] % 1000).padStart(3, '0')
   110→  return `${words.join('-')}-${digits}`
   111→}

   144→function fmtElapsedSec(sec: number): string {
   145→  const s = Math.max(0, Math.floor(sec))
   146→  const m = Math.floor(s / 60)
   147→  const r = s % 60
   148→  return `${String(m).padStart(2, '0')}:${String(r).padStart(2, '0')}`
   149→}

   257→function Scheduler({
   258→  device,
   259→  onScheduled,
   260→}: {
   261→  device: Device | null
   262→  onScheduled: (r: RecordingArtifact) => void
   263→}) {
   264→  // derived selection: follows the store's last-selected device until the
   265→  // user explicitly overrides it in this panel (no sync setState-in-effect)
   266→  const [overrideId, setOverrideId] = useState<string | null>(null)
   267→  const devId = overrideId ?? device?.id ?? null
   268→  const [label, setLabel] = useState('')
   269→  const [minutes, setMinutes] = useState('15')
   270→  const [sources, setSources] = useState<SelectedSources>({ monitors: [], cameras: [], mics: [] })
   271→  const [passphrase, setPassphrase] = useState('')
   272→  const [confirmPass, setConfirmPass] = useState('')
   273→  const [showPass, setShowPass] = useState(false)
   274→  const [showConfirm, setShowConfirm] = useState(false)
   275→  const [fingerprint, setFingerprint] = useState<string | null>(null)
   276→
   277→  const caps = device?.captures ?? null
   278→  const hasCaps =
   279→    !!caps && (caps.monitors.length > 0 || caps.cameras.length > 0 || caps.mics.length > 0)
   280→
   281→  // live sha256(passphrase)[0:12] preview so the key can be verified later
   282→  // (async — Web Crypto; render guards on passphrase so a cleared field never
   283→  // shows a stale fingerprint)
   284→  useEffect(() => {
   285→    if (!passphrase) return
   286→    let alive = true
   287→    void sha256HexPrefix(passphrase).then((fp) => {
   288→      if (alive) setFingerprint(fp)
   289→    })
   290→    return () => {
   291→      alive = false
   292→    }
   293→  }, [passphrase])
   294→
   295→  const durationSec = Math.max(1, Math.floor(Number(minutes) || 0)) * 60
   296→  const passOk = passphrase.length >= 8
   297→  const confirmOk = passphrase.length > 0 && passphrase === confirmPass
   298→  const sourceCount = sources.monitors.length + sources.cameras.length + sources.mics.length
   299→  const valid = !!devId && durationSec >= 60 && sourceCount > 0 && passOk && confirmOk
   300→
   301→  const toggle = (kind: keyof SelectedSources, id: string) =>
   302→    setSources((prev) => {
   303→      const has = prev[kind].includes(id)
   304→      return { ...prev, [kind]: has ? prev[kind].filter((x) => x !== id) : [...prev[kind], id] }
   305→    })
   306→
   307→  const schedule = useMutation({
   308→    mutationFn: () =>
   309→      api<RecordingArtifact>('/api/recordings', {
   310→        method: 'POST',
   311→        body: JSON.stringify({
   312→          deviceId: devId,
   313→          durationSec,
   314→          monitors: sources.monitors.length > 0 ? sources.monitors : undefined,
   315→          cameras: sources.cameras.length > 0 ? sources.cameras : undefined,
   316→          mics: sources.mics.length > 0 ? sources.mics : undefined,
   317→          passphrase,
   318→          label: label.trim() || undefined,
   319→        }),
   320→      }),
   321→    onSuccess: (r) => {
   322→      toast.success('Recording scheduled', {
   323→        description: `${device?.hostname ?? 'device'} • ${Math.round(durationSec / 60)} min • armed`,
   324→      })
   325→      onScheduled(r)
   326→      // hygiene: never keep the passphrase around longer than needed
   327→      setPassphrase('')
   328→      setConfirmPass('')
   329→    },
   330→    onError: (e) =>
   331→      toast.error('Failed to schedule recording', {
   332→        description: e instanceof Error ? e.message : 'Unknown error',
   333→      }),
   334→  })
   335→
   336→  return (
   337→    <Panel className="card-sheen glass-hover flex flex-col gap-5 p-5 sm:p-6">
   338→      <div>
   339→        <h3 className="font-semibold">Schedule a recording</h3>
   340→        <p className="text-sm text-muted-foreground">
   341→          Armed on the device, recorded locally, uploaded encrypted when done.
   342→        </p>
   343→      </div>
   344→
   345→      <div className="flex flex-col gap-2">
   346→        <Label className="text-xs uppercase tracking-wide text-muted-foreground">Device</Label>
   347→        <DeviceSelector value={devId} onChange={setOverrideId} className="sm:w-full" />
   348→      </div>
   349→
   350→      <div className="grid gap-3 sm:grid-cols-2">
   351→        <div className="flex flex-col gap-2">
   352→          <Label htmlFor="rec-label" className="text-xs uppercase tracking-wide text-muted-foreground">
   353→            Label
   354→          </Label>
   355→          <Input
   356→            id="rec-label"
   357→            value={label}
   358→            onChange={(e) => setLabel(e.target.value)}
   359→            placeholder="e.g. Nightly audit trail"
   360→            className="h-11 md:h-9"
   361→            maxLength={80}
   362→          />
   363→        </div>
   364→        <div className="flex flex-col gap-2">
   365→          <Label htmlFor="rec-minutes" className="text-xs uppercase tracking-wide text-muted-foreground">
   366→            Duration (minutes)
   367→          </Label>
   368→          <Input
   369→            id="rec-minutes"
   370→            type="number"
   371→            min={1}
   372→            max={1440}
   373→            value={minutes}
   374→            onChange={(e) => setMinutes(e.target.value)}
   375→            className="h-11 md:h-9"
   376→          />
   377→          <div className="flex flex-wrap gap-1.5">
   378→            {DURATION_CHIPS.map((m) => (
   379→              <button
   380→                key={m}
   381→                type="button"
   382→                onClick={() => setMinutes(String(m))}
   383→                className={cn(
   384→                  'min-h-8 rounded-full px-2.5 text-[11px] font-semibold outline-none transition-all focus-visible:ring-2 focus-visible:ring-ring/60',
   385→                  Number(minutes) === m
   386→                    ? 'bg-primary/20 text-primary shadow-[inset_0_0_0_1px] shadow-primary/30'
   387→                    : 'glass-inset text-muted-foreground hover:bg-primary/10 hover:text-foreground',
   388→                )}
   389→              >
   390→                {m} min
   391→              </button>
   392→            ))}
   393→          </div>
   394→        </div>
   395→      </div>
   396→
   397→      <div className="flex flex-col gap-2">
   398→        <Label className="text-xs uppercase tracking-wide text-muted-foreground">
   399→          Sources — at least one
   400→        </Label>
   401→        {hasCaps && caps ? (
   402→          <div className="flex flex-col gap-2">
   403→            {caps.monitors.length > 0 ? (
   404→              <div className="flex flex-wrap gap-1.5">
   405→                {caps.monitors.map((m) => (

   410→                    selected={sources.monitors.includes(m.id)}
   411→                    onToggle={() => toggle('monitors', m.id)}
   412→                  />
   413→                ))}
   414→              </div>
   415→            ) : null}
   416→            {caps.cameras.length > 0 ? (
   417→              <div className="flex flex-wrap gap-1.5">
   418→                {caps.cameras.map((c) => (
   419→                  <SourceChip
   420→                    key={c.id}
   421→                    icon={Camera}
   422→                    label={c.name}
   423→                    selected={sources.cameras.includes(c.id)}
   424→                    onToggle={() => toggle('cameras', c.id)}
   425→                  />
   426→                ))}
   427→              </div>
   428→            ) : null}
   429→            {caps.mics.length > 0 ? (
   430→              <div className="flex flex-wrap gap-1.5">
   431→                {caps.mics.map((m) => (
   432→                  <SourceChip
   433→                    key={m.id}
   434→                    icon={Mic}
   435→                    label={m.name}
   436→                    selected={sources.mics.includes(m.id)}
   437→                    onToggle={() => toggle('mics', m.id)}
   438→                  />
   439→                ))}
   440→              </div>
   441→            ) : null}
   442→          </div>
   443→        ) : (
   444→          <p className="glass-inset flex items-start gap-2 rounded-xl px-3 py-2.5 text-xs text-muted-foreground">
   445→            <Info className="mt-0.5 h-3.5 w-3.5 shrink-0 text-amber-500" aria-hidden />
   446→            This device did not report capture sources — the Python/Node agent is required before
   447→            recording sources can be planned.
   448→          </p>
   449→        )}
   450→      </div>
   451→
   452→      <div className="grid gap-3 sm:grid-cols-2">
   453→        <div className="flex flex-col gap-2">
   454→          <Label htmlFor="rec-pass" className="text-xs uppercase tracking-wide text-muted-foreground">
   455→            Passphrase (min 8)
   456→          </Label>
   457→          <div className="relative">
   458→            <Input
   459→              id="rec-pass"
   460→              type={showPass ? 'text' : 'password'}
   461→              value={passphrase}
   462→              onChange={(e) => setPassphrase(e.target.value)}
   463→              placeholder="••••••••••••"
   464→              autoComplete="new-password"
   465→              className="h-11 pr-20 md:h-9"
   466→            />
   467→            <div className="absolute right-1 top-1/2 flex -translate-y-1/2 items-center">

   512→                'h-11 pr-11 md:h-9',

   515→            />

   517→              type="button"

   740→              size="sm"
